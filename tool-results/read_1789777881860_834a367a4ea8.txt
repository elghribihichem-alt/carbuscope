   281→   281→function PriceHistoryChart({ data }: { data: FuelPricesResponse }) {
   282→   282→  const [visible, setVisible] = useState<Record<FuelType, boolean>>({
   283→   283→    ESSENCE: true, GAZOLE: true, GPL: true,
   284→   284→  })
   285→   285→
   286→   286→  // Série par date de décret : une ligne par carburant, valeurs en DT.
   287→   287→  const series = useMemo(() => {
   288→   288→    const byDate = new Map<string, Record<string, number | string>>()
   289→   289→    for (const h of data.history) {
   290→   290→      const key = h.effectiveAt.slice(0, 10)
   291→   291→      const row = byDate.get(key) ?? { date: key, dateLabel: fmtDateFr(h.effectiveAt) }
   292→   292→      row[h.fuelType] = h.priceMillimes / 1000
   293→   293→      byDate.set(key, row)
   294→   294→    }
   295→   295→    return [...byDate.values()].sort((a, b) => String(a.date).localeCompare(String(b.date)))
   296→   296→  }, [data.history])
   297→   297→
   298→   298→  // Points de changement (pour pastilles) : dates où un carburant visible a bougé.
   299→   299→  const changeDots = useMemo(() => {
   300→   300→    const dots: { date: string; fuel: FuelType; value: number }[] = []
   301→   301→    const prev = new Map<FuelType, number>()
   302→   302→    for (const h of data.history) {
   303→   303→      const p = prev.get(h.fuelType)
   304→   304→      if (p !== undefined && p !== h.priceMillimes && visible[h.fuelType]) {
   305→   305→        dots.push({ date: h.effectiveAt.slice(0, 10), fuel: h.fuelType, value: h.priceMillimes / 1000 })
   306→   306→      }
   307→   307→      prev.set(h.fuelType, h.priceMillimes)
   308→   308→    }
   309→   309→    return dots
   310→   310→  }, [data.history, visible])
   311→   311→
   312→   312→  const anyVisible = Object.values(visible).some(Boolean)
   313→   313→
   314→   314→  return (
   315→   315→    <Card className="h-full">
   316→   316→      <CardHeader className="pb-2">
   317→   317→        <div className="flex flex-wrap items-start justify-between gap-2">
   318→   318→          <div>
   319→   319→            <CardTitle className="text-base leading-tight">Historique des décrets — 2021 → aujourd&apos;hui</CardTitle>
   320→   320→            <CardDescription className="mt-0.5 text-xs">
   321→   321→              Prix pompe par carburant (DT/{`L`}) — chaque marche = un décret ministériel
   322→   322→            </CardDescription>
   323→   323→          </div>
   324→   324→          <div className="flex flex-wrap gap-1.5">
   325→   325→            {(Object.keys(FUEL_META) as FuelType[]).map(f => (
   326→   326→              <button
   327→   327→                key={f}
   328→   328→                type="button"
   329→   329→                onClick={() => setVisible(v => ({ ...v, [f]: !v[f] }))}
   330→   330→                aria-pressed={visible[f]}
   331→   331→                className={cn(
   332→   332→                  "inline-flex items-center gap-1.5 rounded-full border px-2.5 py-1 text-[11px] font-medium transition-all focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-emerald-500",
   333→   333→                  visible[f]
   334→   334→                    ? "border-border bg-background shadow-sm hover:bg-muted/60"
   335→   335→                    : "border-transparent bg-muted/40 text-muted-foreground/60 hover:bg-muted",
   336→   336→                )}
   337→   337→              >
   338→   338→                <span
   339→   339→                  className="h-2 w-2 rounded-full"
   340→   340→                  style={{ background: visible[f] ? FUEL_META[f].color : "#94a3b8" }}
   341→   341→                  aria-hidden
   342→   342→                />
   343→   343→                {FUEL_META[f].label.split(" ")[0]}
   344→   344→              </button>
   345→   345→            ))}
   346→   346→          </div>
   347→   347→        </div>
   348→   348→      </CardHeader>
   349→   349→      <CardContent className="h-[340px] pl-0">
   350→   350→        {anyVisible ? (
   351→   351→          <ResponsiveContainer width="100%" height="100%">
   352→   352→            <ComposedChart data={series} margin={{ top: 10, right: 16, left: 0, bottom: 0 }}>
   353→   353→              <CartesianGrid stroke={GRID_STROKE} strokeDasharray="3 3" vertical={false} />
   354→   354→              <XAxis
   355→   355→                dataKey="dateLabel"
   356→   356→                tick={{ ...AXIS_STYLE, fontSize: 10 }}
   357→   357→                tickLine={false}
   358→   358→                axisLine={false}
   359→   359→                interval="preserveStartEnd"
   360→   360→                minTickGap={24}
   361→   361→              />
   362→   362→              <YAxis
   363→   363→                tick={{ ...AXIS_STYLE, fontSize: 10 }}
   364→   364→                tickLine={false}
   365→   365→                axisLine={false}
   366→   366→                width={44}
   367→   367→                tickFormatter={(v: number) => v.toFixed(2)}
   368→   368→                domain={["auto", "auto"]}
   369→   369→              />
   370→   370→              <Tooltip
   371→   371→                contentStyle={tooltipStyle()}
   372→   372→                formatter={(value: number | string, name: string) => [
   373→   373→                  `${Number(value).toFixed(3).replace(".", ",")} DT`,
   374→   374→                  FUEL_META[name as FuelType]?.label ?? name,
   375→   375→                ]}
   376→   376→              />
   377→   377→              {(Object.keys(FUEL_META) as FuelType[]).filter(f => visible[f]).map(f => (
   378→   378→                <Line
   379→   379→                  key={f}
   380→   380→                  type="stepAfter"
   381→   381→                  dataKey={f}
   382→   382→                  stroke={FUEL_META[f].color}
   383→   383→                  strokeWidth={2.2}
   384→   384→                  dot={false}
   385→   385→                  activeDot={{ r: 4, strokeWidth: 2 }}
   386→   386→                />
   387→   387→              ))}
   388→   388→              {changeDots.map((d, i) => (
   389→   389→                <ReferenceDot
   390→   390→                  key={`${d.date}-${d.fuel}-${i}`}
   391→   391→                  x={d.date}
   392→   392→                  y={d.value}
   393→   393→                  r={3.5}
   394→   394→                  fill={FUEL_META[d.fuel].color}
   395→   395→                  stroke="var(--background)"
   396→   396→                  strokeWidth={1.5}
   397→   397→                  // dataKey inutile ici : les points s'affichent sur leur série
   398→   398→                  ifOverflow="extendDomain"
   399→   399→                />
   400→   400→              ))}
   401→   401→            </ComposedChart>
   402→   402→          </ResponsiveContainer>
   403→   403→        ) : (
   404→   404→          <div className="flex h-full items-center justify-center text-sm text-muted-foreground">
   405→   405→            Sélectionnez au moins un carburant.
   406→   406→          </div>
   407→   407→        )}
   408→   408→      </CardContent>
   409→   409→    </Card>
   410→   410→  )
   411→   411→}

   632→   632→function DecreesTable({ data }: { data: FuelPricesResponse }) {

   634→   634→  const rows = useMemo(() => {

   636→   636→    return [...data.history]
   637→   637→      .sort((a, b) => b.effectiveAt.localeCompare(a.effectiveAt))
   638→   638→      .map(h => {
   639→   639→        const p = prev.get(h.fuelType)
   640→   640→        prev.set(h.fuelType, { price: h.priceMillimes, note: h.decreeNote, at: h.effectiveAt })
   641→   641→        return {
   642→   642→          ...h,
   643→   643→          prevPrice: p?.price ?? null,
   644→   644→          delta: p ? h.priceMillimes - p.price : null,

   675→   675→                return (
   676→   676→                  <tr
   677→   677→                    key={r.id}
   678→   678→                    className={cn(
   679→   679→                      "border-b/60 transition-colors hover:bg-muted/50",
