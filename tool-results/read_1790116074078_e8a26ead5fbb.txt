    85→function EditableCell({ row, field, kind, ariaName, onCommit }: {
    86→  row: DemographicDelegationDTO
    87→  field: string
    88→  kind: EditKind
    89→  ariaName: string
    90→  onCommit: (row: DemographicDelegationDTO, field: string, v: number | null) => void
    91→}) {
    92→  const value = (row as unknown as Record<string, number | null>)[field] ?? null
    93→  const [editing, setEditing] = useState(false)
    94→  const [draft, setDraft] = useState("")
    95→  const inputRef = useRef<HTMLInputElement>(null)
    96→
    97→  useEffect(() => {
    98→    if (editing) {
    99→      inputRef.current?.focus()
   100→      inputRef.current?.select()
   101→    }
   102→  }, [editing])
   103→
   104→  const display = value === null ? "—" : kind === "pct" ? fmtPct(value) : fmtInt(value)
   105→
   106→  const tryCommit = (fromBlur: boolean) => {
   107→    const t = draft.trim()
   108→    if (t === "") {
   109→      setEditing(false)
   110→      if (value !== null) onCommit(row, field, null)
   111→      return
   112→    }
   113→    const v = parseNumFr(t)
   114→    const invalid = !Number.isFinite(v)
   115→      || (kind === "pct" && (v < 0 || v > 100))
   116→      || (kind === "int" && (!Number.isInteger(v) || v < 0))
   117→    if (invalid) {
   118→      if (fromBlur) { setEditing(false); return }
   119→      toast.error(kind === "pct" ? "Le taux doit être un nombre entre 0 et 100" : "Nombre entier positif attendu")
   120→      return
   121→    }
   122→    setEditing(false)
   123→    if (v !== value) onCommit(row, field, v)
   124→  }
   125→
   126→  if (editing) {
   127→    return (
   128→      <input
   129→        ref={inputRef}
   130→        value={draft}
   131→        onChange={e => setDraft(e.target.value)}
   132→        onKeyDown={e => {
   133→          if (e.key === "Enter") { e.preventDefault(); tryCommit(false) }
   134→          if (e.key === "Escape") { e.preventDefault(); setEditing(false) }
   135→        }}
   136→        onBlur={() => tryCommit(true)}
   137→        inputMode={kind === "int" ? "numeric" : "decimal"}
   138→        className="h-8 w-full min-w-0 rounded-md border border-emerald-400 bg-background px-2 text-right font-mono text-xs tabular-nums shadow-sm outline-none focus-visible:ring-2 focus-visible:ring-emerald-500/40 dark:border-emerald-600"
   139→        aria-label={ariaName}
   140→      />
   141→    )
   142→  }
   143→
   144→  return (
   145→    <button
   146→      type="button"
   147→      onClick={() => { setDraft(value === null ? "" : String(value)); setEditing(true) }}
   148→      className={`inline-flex h-8 w-full items-center justify-end rounded-md px-2 font-mono text-xs tabular-nums transition-colors hover:bg-emerald-100/80 hover:font-semibold dark:hover:bg-emerald-950/70 ${value === null ? "text-muted-foreground/50" : "text-foreground"}`}
   149→      title="Cliquer pour modifier"
   150→      aria-label={`${ariaName} : ${value === null ? "non renseigné" : display} — cliquer pour modifier`}
   151→    >
   152→      {display}
   153→    </button>
   154→  )
   155→}

   159→function ChartPie({ items, unit, decimals, total }: {
   160→  items: { label: string; value: number }[]
   161→  unit: string
   162→  decimals: number
   163→  total: number
   164→}) {
   165→  const sorted = [...items].sort((a, b) => b.value - a.value)
   166→  const top = sorted.slice(0, 8)
   167→  const rest = sorted.slice(8)
   168→  const slices = rest.length > 0
   169→    ? [...top, { label: `Autres (${rest.length})`, value: rest.reduce((s, x) => s + x.value, 0) }]
   170→    : top
   171→  const R = 56
   172→  const C = 2 * Math.PI * R
   173→  // Calcul pur des segments (part cumulée, sans mutation) : camembert = donut
   174→  // SVG en traits circulaires, départ à 12 h (rotation -90°).
   175→  const segments = slices.reduce<{ label: string; value: number; frac: number; start: number }[]>((list, s) => {
   176→    const prev = list.length > 0 ? list[list.length - 1] : null
   177→    const start = prev ? prev.start + prev.frac : 0
   178→    const frac = total > 0 ? s.value / total : 0
   179→    return [...list, { label: s.label, value: s.value, frac, start }]
   180→  }, [])
   181→
   182→  return (
   183→    <div className="flex flex-col items-center gap-6 lg:flex-row lg:gap-10">
   184→      <svg viewBox="0 0 160 160" className="h-44 w-44 shrink-0" role="img"
   185→        aria-label={`Camembert : ${slices.length} parts, total ${fmtChart(total, decimals)} ${unit}`}>
   186→        <g transform="rotate(-90 80 80)">
   187→          {segments.map((s, i) => {
   188→            const len = s.frac * C
   189→            if (len <= 0) return null
   190→            return (
   191→              <motion.circle
   192→                key={s.label}
   193→                cx={80} cy={80} r={R}
   194→                fill="none"
   195→                stroke={CHART_COLORS[i % CHART_COLORS.length]}
   196→                strokeWidth={26}
   197→                strokeDasharray={`${len} ${C - len}`}
   198→                strokeDashoffset={-s.start * C}
   199→                initial={{ opacity: 0 }}
   200→                animate={{ opacity: 1 }}
   201→                transition={{ duration: 0.35, delay: i * 0.05 }}
   202→              />
   203→            )
   204→          })}
   205→        </g>
   206→        <text x={80} y={74} textAnchor="middle" className="fill-foreground text-[13px] font-bold">
   207→          {fmtChart(total, decimals)}
   208→        </text>
   209→        <text x={80} y={90} textAnchor="middle" className="fill-muted-foreground text-[9px]">
   210→          {unit}
   211→        </text>
   212→      </svg>
   213→      <ul className="grid w-full max-w-lg grid-cols-1 gap-x-8 gap-y-1.5 sm:grid-cols-2">
   214→        {slices.map((s, i) => (
   215→          <li key={s.label} className="flex items-center gap-2 text-xs">
   216→            <span className="h-2.5 w-2.5 shrink-0 rounded-full" style={{ backgroundColor: CHART_COLORS[i % CHART_COLORS.length] }} aria-hidden />
   217→            <span className="min-w-0 flex-1 truncate" title={s.label}>{s.label}</span>
   218→            <span className="font-mono tabular-nums text-muted-foreground">{fmtChart(s.value, decimals)}</span>
   219→            <span className="w-14 text-right font-mono tabular-nums text-muted-foreground/70">
   220→              {total > 0 ? `${((s.value / total) * 100).toFixed(1).replace(".", ",")} %` : "—"}
   221→            </span>
   222→          </li>
   223→        ))}
   224→      </ul>
   225→    </div>
   226→  )
   227→}

   231→function ChartBars({ items, unit, decimals }: {
   232→  items: { label: string; value: number }[]
   233→  unit: string
   234→  decimals: number
   235→}) {
   236→  const sorted = [...items].sort((a, b) => b.value - a.value)
   237→  const shown = sorted.slice(0, 12)
   238→  const maxV = shown[0]?.value ?? 0
   239→
   240→  return (
   241→    <div className="space-y-2">
   242→      {shown.map((x, i) => (
   243→        <div key={x.label} className="flex items-center gap-3">
   244→          <span className="w-36 shrink-0 truncate text-right text-xs" title={x.label}>{x.label}</span>
   245→          <div className="relative h-5 min-w-0 flex-1 overflow-hidden rounded-md bg-muted">
   246→            <motion.div
   247→              className="h-full rounded-md bg-emerald-600"
   248→              style={{ opacity: 1 - Math.min(i, 8) * 0.08 }}
   249→              initial={{ width: 0 }}
   250→              animate={{ width: `${maxV > 0 ? Math.max((x.value / maxV) * 100, 2) : 0}%` }}
   251→              transition={{ duration: 0.45, delay: i * 0.04, ease: "easeOut" }}
   252→            />
   253→            <span className="absolute inset-y-0 right-2 flex items-center font-mono text-[10px] tabular-nums text-foreground/80">
   254→              {fmtChart(x.value, decimals)} {unit}
   255→            </span>
   256→          </div>
   257→        </div>
   258→      ))}
   259→      {sorted.length > shown.length && (
   260→        <p className="pl-[10.5rem] text-[11px] text-muted-foreground">… et {sorted.length - shown.length} autres</p>
   261→      )}
   262→    </div>
   263→  )
   264→}

   759→/** Résumé FR du patch pour le toast : « population : 52 400 · taux urbain : 78 % ». */
   760→function describePatch(patch: Record<string, number | null>): string {
   761→  return Object.entries(patch)
   762→    .map(([k, v]) => {
   763→      const label = TOAST_LABELS[k] ?? k
   764→      return v === null ? `${label} effacé` : `${label} : ${k.endsWith("Pct") ? fmtPct(v) : fmtInt(v)}`
   765→    })
   766→    .join(" · ")
   767→}
