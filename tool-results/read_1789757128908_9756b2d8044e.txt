    91→let staticPromise: Promise<StaticLayers> | null = null
    92→function loadStatic(): Promise<StaticLayers> {
    93→  if (!staticPromise) {
    94→    staticPromise = (async () => {
    95→      const safe = async (url: string, key: string) => {
    96→        try {
    97→          const r = await fetch(url)
    98→          if (!r.ok) return []
    99→          const j = await r.json()
   100→          return j[key] ?? []
   101→        } catch { return [] }
   102→      }
   103→      const [roads, projects, governorats, services] = await Promise.all([
   104→        safe("/api/roads", "roads"),
   105→        safe("/api/projects", "projects"),
   106→        safe("/api/governorats", "governorats"),
   107→        safe("/api/services", "services"),
   108→      ])
   109→      return { roads, projects, governorats, services }
   110→    })()
   111→  }
   112→  return staticPromise
   113→}

   129→  const wrapRef = useRef<HTMLDivElement>(null)
   130→  const svgRef = useRef<SVGSVGElement>(null)
   131→  const movedRef = useRef(0)
   132→
   133→  const [view, setView] = useState({ k: 1, x: 0, y: 0 })
   134→  const [hoverCode, setHoverCode] = useState<string | null>(null)
   135→  const [selCode, setSelCode] = useState<string | null>(selectedCode ?? null)

   138→    ...(layers ?? {}),
   139→  })
   140→  const [staticData, setStaticData] = useState<StaticLayers | null>(null)
   141→  const [detail, setDetail] = useState<StationDetail | null>(null)
   142→  const [detailCode, setDetailCode] = useState<string | null>(null)
   143→  const [showHistory, setShowHistory] = useState(false)
   144→
   145→  // Fond cartographique réel (OSM) vs schéma vectoriel — persisté par visiteur.
   146→  // Lecture différée (setTimeout 0) : évite tout écart d'hydratation SSR et les setState synchrones en effet.
   147→  // La persistance n'écrit qu'après la lecture initiale (sinon la valeur par défaut écraserait le choix stocké).
   148→  const [realMap, setRealMap] = useState(true)
   149→  const [osmBg, setOsmBg] = useState<OsmBackground>("auto")
   150→  const [bgPrefsLoaded, setBgPrefsLoaded] = useState(false)
   151→  useEffect(() => {
   152→    const id = setTimeout(() => {
   153→      try {
   154→        const stored = localStorage.getItem("carbuscope-map-bg")
   155→        setRealMap(stored ? stored === "real" : true) // fond réel activé par défaut
   156→      } catch { /* ignore */ }
   157→      try {
   158→        const b = localStorage.getItem("carbuscope-map-osmbg")
   159→        if (b && OSM_BG_KEYS.has(b)) setOsmBg(b as OsmBackground)
   160→      } catch { /* ignore */ }
   161→      setBgPrefsLoaded(true)
   162→    }, 0)
   163→    return () => clearTimeout(id)
   164→  }, [])
   165→  useEffect(() => {
   166→    if (!bgPrefsLoaded) return
   167→    try { localStorage.setItem("carbuscope-map-bg", realMap ? "real" : "schema") } catch { /* ignore */ }
   168→  }, [realMap, bgPrefsLoaded])
   169→  useEffect(() => {
   170→    if (!bgPrefsLoaded) return
   171→    try { localStorage.setItem("carbuscope-map-osmbg", osmBg) } catch { /* ignore */ }
   172→  }, [osmBg, bgPrefsLoaded])
   173→
   174→  // outils
   175→  const [tool, setTool] = useState<Tool>(null)
   176→  const [measurePts, setMeasurePts] = useState<[number, number][]>([])
   177→  const [radiusCenter, setRadiusCenter] = useState<[number, number] | null>(null)
   178→  const [radiusKm, setRadiusKm] = useState(10)
   179→  const [polygonPts, setPolygonPts] = useState<[number, number][]>([])
   180→  const [polygonClosed, setPolygonClosed] = useState(false)
   181→
   182→  useEffect(() => { loadStatic().then(setStaticData).catch(() => {}) }, [])
   183→
   184→  // ---------- Conversions écran ↔ SVG (letterbox-safe) ----------
   185→  // métriques de conteneur suivies par ResizeObserver (accessibles au rendu)
   186→  const [mstate, setMstate] = useState<{ scale: number; offX: number; offY: number; w: number; h: number }>({
   187→    scale: 0.5, offX: 0, offY: 0, w: 520, h: 1000,
   188→  })
   189→
   190→  useEffect(() => {
   191→    const el = svgRef.current
   192→    if (!el) return
   193→    const update = () => {
   194→      const r = el.getBoundingClientRect()
   195→      const scale = Math.min(r.width / MAP_W, r.height / MAP_H) || 0.5
   196→      setMstate({
   197→        scale,
   198→        offX: (r.width - MAP_W * scale) / 2,
   199→        offY: (r.height - MAP_H * scale) / 2,
   200→        w: r.width,
   201→        h: r.height,
   202→      })
   203→    }
   204→    const raf = requestAnimationFrame(update)
   205→    const ro = new ResizeObserver(update)
   206→    ro.observe(el)
   207→    return () => { cancelAnimationFrame(raf); ro.disconnect() }
   208→  }, [])
   209→
   210→  const metrics = useCallback(() => {
   211→    const el = svgRef.current
   212→    if (!el) return null
   213→    const r = el.getBoundingClientRect()
   214→    const scale = Math.min(r.width / MAP_W, r.height / MAP_H)
   215→    const offX = (r.width - MAP_W * scale) / 2
   216→    const offY = (r.height - MAP_H * scale) / 2
   217→    return { scale, offX, offY, rect: r }
   218→  }, [])
   219→
   220→  const svgPointFromEvent = useCallback((e: { clientX: number; clientY: number }): [number, number] | null => {
   221→    const m = metrics()
   222→    if (!m) return null
   223→    const sx = (e.clientX - m.rect.left - m.offX) / m.scale
   224→    const sy = (e.clientY - m.rect.top - m.offY) / m.scale
   225→    return [sx, sy]
   226→  }, [metrics])
   227→
   228→  const toScreen = useCallback((p: { x: number; y: number }) => ({
   229→    left: mstate.offX + (p.x * view.k + view.x) * mstate.scale,
   230→    top: mstate.offY + (p.y * view.k + view.y) * mstate.scale,
   231→  }), [mstate, view])
   232→
   233→  // ---------- Zoom / pan ----------
   234→  useEffect(() => {
   235→    const el = svgRef.current
   236→    if (!el) return
   237→    const onWheel = (e: WheelEvent) => {
   238→      e.preventDefault()
   239→      const pt = svgPointFromEvent(e)
   240→      if (!pt) return
   241→      setView(v => {
   242→        const factor = e.deltaY < 0 ? 1.18 : 1 / 1.18
   243→        const k = clamp(v.k * factor, 1, 14)
   244→        const mx = (pt[0] - v.x) / v.k
   245→        const my = (pt[1] - v.y) / v.k
   246→        return { k, x: pt[0] - mx * k, y: pt[1] - my * k }
   247→      })
   248→    }
   249→    el.addEventListener("wheel", onWheel, { passive: false })
   250→    return () => el.removeEventListener("wheel", onWheel)
   251→  }, [svgPointFromEvent])
   252→
   253→  const zoomBy = (factor: number) => setView(v => {
   254→    const k = clamp(v.k * factor, 1, 14)
   255→    const cx = MAP_W / 2, cy = MAP_H / 2
   256→    const mx = (cx - v.x) / v.k, my = (cy - v.y) / v.k
   257→    return { k, x: cx - mx * k, y: cy - my * k }
   258→  })
   259→
   260→  const resetView = () => setView({ k: 1, x: 0, y: 0 })
   261→
   262→  const fitBounds = useCallback((sts: StationDTO[]) => {
   263→    if (!sts.length) { resetView(); return }
   264→    const pts = sts.map(s => project(s.lng, s.lat))
   265→    const minX = Math.min(...pts.map(p => p.x)), maxX = Math.max(...pts.map(p => p.x))
   266→    const minY = Math.min(...pts.map(p => p.y)), maxY = Math.max(...pts.map(p => p.y))
   267→    const bw = Math.max(maxX - minX, 10), bh = Math.max(maxY - minY, 10)
   268→    const k = clamp(Math.min((MAP_W - 80) / bw, (MAP_H - 80) / bh), 1, 6)
   269→    const cx = (minX + maxX) / 2, cy = (minY + maxY) / 2
   270→    setView({ k, x: MAP_W / 2 - cx * k, y: MAP_H / 2 - cy * k })
   271→  }, [])
   272→
   273→  useEffect(() => {
   274→    if (!(autoFitOnLoad && stations.length)) return
   275→    const raf = requestAnimationFrame(() => fitBounds(stations))
   276→    return () => cancelAnimationFrame(raf)
   277→  }, [autoFitOnLoad, stations, fitBounds])
   278→
   279→  useEffect(() => {
   280→    if (!fitTo) return
   281→    const raf = requestAnimationFrame(() => fitBounds(fitTo))
   282→    return () => cancelAnimationFrame(raf)
   283→  }, [fitTo, fitBounds])
   284→
   285→  const selectStation = useCallback((s: StationDTO | null) => {
   286→    setSelCode(s ? s.code : null)
   287→    setShowHistory(false)
   288→    onSelect?.(s)
   289→  }, [onSelect])
   290→
   291→  // ---------- Focus « voir sur la carte » ----------
   292→  useEffect(() => {
   293→    if (!focusCode) return
   294→    const raf = requestAnimationFrame(() => {
   295→      const st = stations.find(s => s.code === focusCode)
   296→      if (st) {
   297→        selectStation(st)
   298→        const p = project(st.lng, st.lat)
   299→        setView({ k: 6, x: MAP_W / 2 - p.x * 6, y: MAP_H / 2 - p.y * 6 })
   300→      }
   301→    })
   302→    return () => cancelAnimationFrame(raf)
   303→  }, [focusCode, stations, selectStation])
   304→
   305→  // ---------- Détail fiche (dérivé : actif seulement si cohérent avec la sélection) ----------
   306→  const activeDetail = selCode && detailCode === selCode ? detail : null
   307→  useEffect(() => {
   308→    if (!selCode) return
   309→    const st = stations.find(s => s.code === selCode)
   310→    if (!st) return
   311→    if (detailCode === selCode) return
   312→    const ctrl = new AbortController()
   313→    fetch(`/api/stations/${st.id}`, { signal: ctrl.signal })
   314→      .then(r => (r.ok ? r.json() : null))
   315→      .then((d: StationDetail | null) => { if (d) { setDetail(d); setDetailCode(selCode) } })
   316→      .catch(() => {})
   317→    return () => ctrl.abort()
   318→  }, [selCode, stations, detailCode])
   319→
   320→  // ---------- Drag pan ----------
   321→  const dragRef = useRef<{ sx: number; sy: number; vx: number; vy: number } | null>(null)
   322→  const onPointerDown = (e: React.PointerEvent) => {
   323→    if (e.button !== 0) return
   324→    movedRef.current = 0
   325→    dragRef.current = { sx: e.clientX, sy: e.clientY, vx: view.x, vy: view.y }
   326→    ;(e.target as Element).setPointerCapture?.(e.pointerId)
   327→  }
   328→  const onPointerMove = (e: React.PointerEvent) => {
   329→    const d = dragRef.current
   330→    if (!d) return
   331→    const m = metrics()
   332→    if (!m) return
   333→    const dx = (e.clientX - d.sx) / m.scale
   334→    const dy = (e.clientY - d.sy) / m.scale
   335→    movedRef.current = Math.max(movedRef.current, Math.abs(dx) + Math.abs(dy))
   336→    if (movedRef.current > 3) {
   337→      setView(v => ({ ...v, x: d.vx + dx, y: d.vy + dy }))
   338→    }
   339→  }
   340→  const onPointerUp = () => { dragRef.current = null }
   341→
   342→  // ---------- Clic fond de carte ----------
   343→  const onBackgroundClick = (e: React.MouseEvent) => {
   344→    if (movedRef.current > 3) return
   345→    const pt = svgPointFromEvent(e)
   346→    if (!pt) return
   347→    if (tool === "measure") { setMeasurePts(p => [...p, pt]); return }
   348→    if (tool === "radius") { setRadiusCenter(pt); return }
   349→    if (tool === "polygon" && !polygonClosed) { setPolygonPts(p => [...p, pt]); return }
   350→    selectStation(null)
   351→  }
   352→
   353→  const onDoubleClick = (e: React.MouseEvent) => {
   354→    if (tool === "polygon" && polygonPts.length >= 3) { setPolygonClosed(true); return }
   355→    const pt = svgPointFromEvent(e)
   356→    if (!pt) return
   357→    setView(v => {
   358→      const k = clamp(v.k * 1.7, 1, 14)
   359→      const mx = (pt[0] - v.x) / v.k, my = (pt[1] - v.y) / v.k
   360→      return { k, x: pt[0] - mx * k, y: pt[1] - my * k }
   361→    })
   362→  }
   363→
   364→  // ---------- Sélections par outil ----------
   365→  const insideRadius = useMemo(() => {
   366→    if (!radiusCenter || tool !== "radius") return null
   367→    const c = unproject(radiusCenter[0], radiusCenter[1])
   368→    const set = new Set<string>()
   369→    for (const s of stations) {
   370→      if (haversineKm(c.lat, c.lng, s.lat, s.lng) <= radiusKm) set.add(s.code)
   371→    }
   372→    return set
   373→  }, [radiusCenter, radiusKm, stations, tool])
   374→
   375→  const insidePolygon = useMemo(() => {
   376→    if (tool !== "polygon" || polygonPts.length < 3 || !polygonClosed) return null
   377→    const poly = polygonPts.map(p => { const u = unproject(p[0], p[1]); return [u.lng, u.lat] as [number, number] })
   378→    const set = new Set<string>()
   379→    for (const s of stations) if (pointInPolygon(s.lng, s.lat, poly)) set.add(s.code)
   380→    return set
   381→  }, [polygonPts, polygonClosed, stations, tool])
   382→
   383→  const exportSelection = (format: "csv" | "geojson") => {
   384→    const set = tool === "polygon" ? insidePolygon : insideRadius
   385→    if (!set) return
   386→    const selected = stations.filter(s => set.has(s.code))
   387→    if (!selected.length) { toast.info("Aucune station dans la sélection."); return }
   388→    let blob: Blob
   389→    let ext: string
   390→    if (format === "csv") {
   391→      const header = "code;nom;enseigne;gouvernorat;delegation;statut;trafic;lat;lng;services\n"
   392→      const rows = selected.map(s =>
   393→        [s.code, s.name, s.brandName, s.gouvernorat, s.delegation, s.status, s.traffic, s.lat, s.lng, s.services.join("|")].join(";")
   394→      ).join("\n")
   395→      blob = new Blob(["\ufeff" + header + rows], { type: "text/csv;charset=utf-8" })
   396→      ext = "csv"
   397→    } else {
   398→      const fc = {
   399→        type: "FeatureCollection",
   400→        features: selected.map(s => ({
   401→          type: "Feature",
   402→          geometry: { type: "Point", coordinates: [s.lng, s.lat] },
   403→          properties: { code: s.code, nom: s.name, enseigne: s.brandName, gouvernorat: s.gouvernorat, statut: s.status, trafic: s.traffic, services: s.services },
   404→        })),
   405→      }
   406→      blob = new Blob([JSON.stringify(fc, null, 2)], { type: "application/geo+json" })
   407→      ext = "geojson"
   408→    }
   409→    const a = document.createElement("a")
   410→    a.href = URL.createObjectURL(blob)
   411→    a.download = `carbuscope-selection.${ext}`
   412→    a.click()
   413→    URL.revokeObjectURL(a.href)
   414→    toast.success(`${selected.length} stations exportées (${ext.toUpperCase()})`)
   415→  }
   416→
   417→  const clearTools = () => {
   418→    setTool(null); setMeasurePts([]); setRadiusCenter(null); setPolygonPts([]); setPolygonClosed(false)
   419→  }
   420→
   421→  const measureTotal = useMemo(() => {
   422→    if (measurePts.length < 2) return 0
   423→    let t = 0
   424→    for (let i = 1; i < measurePts.length; i++) {
   425→      const a = unproject(measurePts[i - 1][0], measurePts[i - 1][1])
   426→      const b = unproject(measurePts[i][0], measurePts[i][1])
   427→      t += haversineKm(a.lat, a.lng, b.lat, b.lng)
   428→    }
   429→    return t
   430→  }, [measurePts])
   431→
   432→  // ---------- Rendu stations (mémoïsé) ----------
   433→  const stationNodes = useMemo(() => {
   434→    return stations.map(s => {
   435→      const p = project(s.lng, s.lat)
   436→      const isOwn = ownBrandSlug && s.brandSlug === ownBrandSlug
   437→      const isSel = s.code === selCode
   438→      const isHover = s.code === hoverCode
   439→      const inRad = insideRadius?.has(s.code)
   440→      const inPoly = insidePolygon?.has(s.code)
   441→      const highlighted = isOwn || isSel || isHover || inRad || inPoly
   442→      const r = isOwn ? 3.6 : 2.7
   443→      if (s.status === "CLOSED") {
   444→        return (
   445→          <g key={s.id} onClick={(e) => { e.stopPropagation(); if (movedRef.current <= 3) selectStation(s) }}
   446→             onMouseEnter={() => setHoverCode(s.code)} onMouseLeave={() => setHoverCode(null)} className="cursor-pointer">

   832→            ))}
   833→          </div>
   834→          <div className="mt-1.5 space-y-0.5 border-t pt-1 text-muted-foreground">
   835→            <span className="flex items-center gap-1"><span className="inline-block h-2 w-2 rounded-full border border-amber-500 bg-transparent" /> Rénovation</span>
   836→            <span className="flex items-center gap-1"><span className="inline-block h-2 w-2 rounded-full border border-dashed" /> Projet</span>
   837→            <span className="flex items-center gap-1"><span className="inline-block h-2 w-2 rounded-full bg-gray-400" /> Fermée</span>
   838→          </div>

   845→function brandLegend(stations: StationDTO[], ownSlug?: string | null): [string, string, number][] {
   846→  const m = new Map<string, [string, number]>()
   847→  for (const s of stations) {
   848→    const cur = m.get(s.brandName)
   849→    m.set(s.brandName, [s.brandColor, (cur?.[1] ?? 0) + 1])
   850→  }
   851→  return [...m.entries()]
   852→    .sort((a, b) => b[1][1] - a[1][1])
   853→    .slice(0, 8)
   854→    .map(([name, [color, count]]) => [ownSlug && stations.find(s => s.brandName === name)?.brandSlug === ownSlug ? `${name} (moi)` : name, color, count])
   855→}
