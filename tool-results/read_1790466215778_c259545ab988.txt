   944→   944→/** Résumé FR du patch pour le toast : « population : 52 400 · taux urbain : 78 % ». */
   945→   945→function describePatch(patch: Record<string, number | null>): string {
   946→   946→  return Object.entries(patch)
   947→   947→    .map(([k, v]) => {
   948→   948→      const label = TOAST_LABELS[k] ?? k
   949→   949→      return v === null ? `${label} effacé` : `${label} : ${k.endsWith("Pct") ? fmtPct(v) : fmtInt(v)}`
   950→   950→    })
   951→   951→    .join(" · ")
   952→   952→}

   954→   954→/** Résumé FR du patch gouvernorat : « population : 1 056 300 · urbain : 99 % ». */
   955→   955→function describeGouvPatch(patch: Record<string, number | null>): string {
   956→   956→  return Object.entries(patch)
   957→   957→    .map(([k, v]) => {
   958→   958→      const label = GOV_TOAST_LABELS[k] ?? k
   959→   959→      if (v === null) return `${label} effacé`
   960→   960→      if (k === "areaKm2") return `${label} : ${fmtInt(v)} km²`
   961→   961→      if (k === "avgFleetAge") return `${label} : ${v.toLocaleString("fr-FR", { maximumFractionDigits: 1 })} ans`
   962→   962→      if (k.endsWith("Pct") || k === "motorizationRate") return `${label} : ${fmtPct(v)}`
   963→   963→      return `${label} : ${fmtInt(v)}`
   964→   964→    })
   965→   965→    .join(" · ")
   966→   966→}
