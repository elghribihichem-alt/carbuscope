    78→/** Ligne enrichie du registre : prix précédent + Δ vs décret précédent du carburant. */
    79→type DecreeRow = FuelPriceDTO & { prevPrice: number | null; deltaPrev: number | null }
    80→
    81→/** Variation vs décret précédent : rouge (hausse) / émeraude (baisse) / ardoise (stable). */
    82→function DecreeDelta({ delta }: { delta: number | null }) {
    83→  if (delta === null) return <span className="text-xs text-muted-foreground">—</span>
    84→  return (
    85→    <span className={cn(
    86→      "text-xs font-semibold tabular-nums",
    87→      delta > 0
    88→        ? "text-red-600 dark:text-red-400"
    89→        : delta < 0
    90→          ? "text-emerald-600 dark:text-emerald-400"
    91→          : "text-muted-foreground",
    92→    )}>
    93→      {delta > 0 ? "+" : ""}{delta} m
    94→    </span>
    95→  )
    96→}

   100→export function BoDecrees() {
   101→  const [data, setData] = useState<FuelPricesResponse | null>(null)
   102→  const [error, setError] = useState<string | null>(null)
   103→  const [reloadKey, setReloadKey] = useState(0)
   104→
   105→  // Formulaire de publication
   106→  const [fuel, setFuel] = useState<FuelType>("ESSENCE")
   107→  const [price, setPrice] = useState("")
   108→  const [margin, setMargin] = useState("")
   109→  const [effectiveAt, setEffectiveAt] = useState(todayIso())
   110→  const [note, setNote] = useState("")
   111→  const [submitting, setSubmitting] = useState(false)
   112→  const [lastResult, setLastResult] = useState<DecreeResponse | null>(null)
   113→
   114→  useEffect(() => {
   115→    const ac = new AbortController()
   116→    fetch("/api/prices", { signal: ac.signal })
   117→      .then(r => (r.ok ? r.json() : Promise.reject(new Error(`HTTP ${r.status}`))))
   118→      .then((j: FuelPricesResponse) => { setData(j); setError(null) })
   119→      .catch(e => {
   120→        if (e instanceof DOMException && e.name === "AbortError") return
   121→        setError("Impossible de charger le registre des décrets.")
   122→      })
   123→    return () => ac.abort()
   124→  }, [reloadKey])
   125→
   126→  const currentByFuel = useMemo(() => {
   127→    const map: Partial<Record<FuelType, FuelPricesResponse["current"][number]>> = {}
   128→    for (const c of data?.current ?? []) map[c.fuelType] = c
   129→    return map
   130→  }, [data])
   131→
   132→  const current = currentByFuel[fuel]
   133→
   134→  // Suppression d'un décret erroné (T24) — confirmation obligatoire, refus
   135→  // serveur si dernier décret d'un carburant. Compte par carburant côté client
   136→  // pour désactiver visuellement les boutons concernés.
   137→  const countByFuel = useMemo(() => {
   138→    const m = new Map<string, number>()
   139→    for (const d of data?.history ?? []) m.set(d.fuelType, (m.get(d.fuelType) ?? 0) + 1)
   140→    return m
   141→  }, [data])
   142→
   143→  // Registre chronologique inversé (plus récent d'abord) + enrichissement :
   144→  // Δ vs décret précédent du même carburant, calculé UNE fois à partir de
   145→  // l'ordre chronologique (indépendant du tri courant de la vue) — même règle
   146→  // que l'onglet B2B « Prix & marges ». Badge « en vigueur » : le décret courant
   147→  // d'un carburant est le plus récent de son historique.
   148→  const decreeRows = useMemo<DecreeRow[]>(() => {
   149→    const desc = [...(data?.history ?? [])].sort(
   150→      (a, b) => new Date(b.effectiveAt).getTime() - new Date(a.effectiveAt).getTime()
   151→    )
   152→    const prevPrice = new Map<string, number>()
   153→    return desc.map(d => {
   154→      const p = prevPrice.get(d.fuelType) ?? null
   155→      prevPrice.set(d.fuelType, d.priceMillimes)
   156→      return { ...d, prevPrice: p, deltaPrev: p === null ? null : d.priceMillimes - p }
   157→    })
   158→  }, [data])
   159→
   160→  const [deletingId, setDeletingId] = useState<string | null>(null)
   161→  const deleteDecree = useCallback(async (d: FuelPriceDTO) => {
   162→    setDeletingId(d.id)
   163→    try {
   164→      const r = await fetch(`/api/prices/${d.id}`, { method: "DELETE" })
   165→      const j = await r.json().catch(() => ({}))
   166→      if (!r.ok) throw new Error(j.error ?? `HTTP ${r.status}`)
   167→      toast.success("Décret retiré du registre", {
   168→        description: j.fallbackPriceMillimes != null
   169→          ? `Prix en vigueur recalculé sur le décret précédent : ${(j.fallbackPriceMillimes / 1000).toFixed(3).replace(".", ",")} DT.`
   170→          : "Le registre chronologique a été mis à jour.",
   171→      })
   172→      setLastResult(null)
   173→      setReloadKey(k => k + 1)
   174→    } catch (e) {
   175→      toast.error("Suppression refusée", { description: e instanceof Error ? e.message : "Erreur inattendue." })
   176→    } finally {
   177→      setDeletingId(null)
   178→    }
   179→  }, [])
   180→
   181→  // Pré-remplissage intelligent au changement de carburant : prix proposé =
   182→  // décret en vigueur + 25 millimes (révision typique), marge inchangée.
   183→  const prefillFor = (f: FuelType) => {
   184→    const c = currentByFuel[f]
   185→    setFuel(f)
   186→    setPrice(c ? String(c.priceMillimes + 25) : "")
   187→    setMargin(c ? String(c.marginMillimes) : "")
   188→  }
   189→
   190→  // Pré-remplissage initial dès que les données arrivent.
   191→  useEffect(() => {
   192→    if (data && price === "" && margin === "") {
   193→      const c = currentByFuel.ESSENCE
   194→      if (c) { setPrice(String(c.priceMillimes + 25)); setMargin(String(c.marginMillimes)) }
   195→    }
   196→  }, [data, price, margin, currentByFuel])
   197→
   198→  const priceNum = Number(price)
   199→  const marginNum = Number(margin)
   200→  const priceValid = Number.isInteger(priceNum) && priceNum >= 300 && priceNum <= 10_000
   201→  const marginValid = Number.isInteger(marginNum) && marginNum >= 0 && marginNum <= 1_000
   202→  const dateValid = /^\d{4}-\d{2}-\d{2}$/.test(effectiveAt) && !Number.isNaN(new Date(effectiveAt).getTime())
   203→  const formValid = priceValid && marginValid && dateValid
   204→
   205→  const delta = current ? priceNum - current.priceMillimes : null
   206→  const marginDelta = current ? marginNum - current.marginMillimes : null
   207→  const deltaPct = current && delta !== null && current.priceMillimes > 0
   208→    ? (delta / current.priceMillimes) * 100
   209→    : null
   210→
   211→  const publish = useCallback(async () => {
   212→    if (!formValid || submitting) return
   213→    setSubmitting(true)
   214→    try {
   215→      const r = await fetch("/api/prices", {
   216→        method: "POST",
   217→        headers: { "Content-Type": "application/json" },
   218→        body: JSON.stringify({
   219→          fuelType: fuel,
   220→          priceMillimes: priceNum,
   221→          marginMillimes: marginNum,
   222→          effectiveAt,
   223→          decreeNote: note.trim() || undefined,
   224→        }),
   225→      })
   226→      const j = (await r.json()) as DecreeResponse & { error?: string }
   227→      if (!r.ok) throw new Error(j.error ?? `HTTP ${r.status}`)
   228→      setLastResult(j)
   229→      toast.success("Décret publié et enseignes notifiées", {
   230→        description: `${FUEL_META[fuel].label} : ${fmtPrice(j.decree.priceMillimes)} ${FUEL_META[fuel].unit} — ${j.notifiedBrands} enseigne(s) notifiée(s) via la cloche du portail.`,
   231→        duration: 6000,
   232→      })
   233→      setNote("")
   234→      setReloadKey(k => k + 1)
   235→    } catch (e) {
   236→      toast.error("Publication refusée", {
   237→        description: e instanceof Error ? e.message : "Erreur inconnue.",
   238→      })
   239→    } finally {
   240→      setSubmitting(false)
   241→    }
   242→  }, [formValid, submitting, fuel, priceNum, marginNum, effectiveAt, note])
   243→
   244→  const resetForm = () => {
   245→    const c = currentByFuel[fuel]
   246→    setPrice(c ? String(c.priceMillimes + 25) : "")
   247→    setMargin(c ? String(c.marginMillimes) : "")
   248→    setEffectiveAt(todayIso())
   249→    setNote("")
   250→  }
   251→
   252→  if (error && !data) {
   253→    return (
   254→      <div className="p-1">
   255→        <ErrorState message={error} retry={() => setReloadKey(k => k + 1)} />
   256→      </div>
   257→    )
   258→  }
   259→  if (!data) return <TabSkeleton title="Chargement du registre des décrets…" />
   260→
   261→  const currentKeys = new Set(
   262→    data.current.map(c => `${c.fuelType}|${c.effectiveAt}`)
   263→  )
   264→
   265→  // Cellule « Purger » : AlertDialog de confirmation intact — désactivée
   266→  // visuellement pour le dernier décret d'un carburant (référence obligatoire).
   267→  const renderPurge = (d: DecreeRow) => {
   268→    const Meta = FUEL_META[d.fuelType as FuelType] ?? FUEL_META.ESSENCE
   269→    if ((countByFuel.get(d.fuelType) ?? 0) <= 1) {
   270→      return (
   271→        <span
   272→          className="inline-block h-3.5 w-3.5 text-muted-foreground/30"
   273→          title="Dernier décret de ce carburant — suppression interdite (référence de prix obligatoire)"
   274→        >
   275→          <Trash2 className="h-3.5 w-3.5" aria-hidden />
   276→        </span>
   277→      )
   278→    }
   279→    return (
   280→      <AlertDialog>
   281→        <AlertDialogTrigger asChild>
   282→          <Button
   283→            variant="ghost"
   284→            size="icon"
   285→            disabled={deletingId === d.id}
   286→            aria-label={`Retirer le décret ${Meta.label} du ${fmtDateFr(d.effectiveAt)} (${fmtPrice(d.priceMillimes)} DT)`}
   287→            title="Retirer ce décret du registre (décret erroné)"
   288→            className="h-7 w-7 text-muted-foreground/70 transition-colors hover:bg-red-50 hover:text-red-600 focus-visible:ring-2 focus-visible:ring-red-500/40 dark:hover:bg-red-950 dark:hover:text-red-400"
   289→          >
   290→            {deletingId === d.id ? <Loader2 className="h-3.5 w-3.5 animate-spin" aria-hidden /> : <Trash2 className="h-3.5 w-3.5" aria-hidden />}
   291→          </Button>
   292→        </AlertDialogTrigger>
   293→        <AlertDialogContent>
   294→          <AlertDialogHeader>
   295→            <AlertDialogTitle>Retirer ce décret du registre ?</AlertDialogTitle>
   296→            <AlertDialogDescription asChild>
   297→              <div className="space-y-2 text-sm text-muted-foreground">
   298→                <p>
   299→                  Décret <span className="font-semibold text-foreground">{Meta.label}</span> du{" "}
   300→                  <span className="font-semibold text-foreground">{fmtDateFr(d.effectiveAt)}</span> — prix{" "}
   301→                  <span className="font-semibold tabular-nums text-foreground">{fmtPrice(d.priceMillimes)} DT</span>, marge{" "}
   302→                  <span className="font-semibold tabular-nums text-foreground">{fmtPrice(d.marginMillimes)} DT/L</span>.
   303→                </p>
   304→                <p>
   305→                  Cette purge est irréversible. Si ce décret était le plus récent de son carburant,
   306→                  le prix en vigueur retombera sur le décret précédent — les enseignes ne sont PAS
   307→                  renotifiées (une publication de correction le fera proprement).
   308→                </p>
   309→              </div>
   310→            </AlertDialogDescription>
   311→          </AlertDialogHeader>
   312→          <AlertDialogFooter>
   313→            <AlertDialogCancel>Annuler</AlertDialogCancel>
   314→            <AlertDialogAction
   315→              className="bg-red-600 text-white hover:bg-red-700 focus-visible:ring-red-500/50"
   316→              onClick={() => void deleteDecree(d)}
   317→            >
   318→              <Trash2 className="mr-1 h-4 w-4" aria-hidden />
   319→              Retirer du registre
   320→            </AlertDialogAction>
   321→          </AlertDialogFooter>
   322→        </AlertDialogContent>
   323→      </AlertDialog>
   324→    )
   325→  }
   326→
   327→  // Colonnes — la colonne « Purger » (actions) reste présente dans les 3 vues.
   328→  const columns: DataViewColumn<DecreeRow>[] = [
   329→    {
   330→      key: "date", header: "Date", className: "w-[7rem] whitespace-nowrap",
   331→      sortValue: (d) => new Date(d.effectiveAt).getTime(),
   332→      render: (d) => <span className="text-xs tabular-nums">{fmtDateFr(d.effectiveAt)}</span>,
   333→    },
   334→    {
   335→      key: "fuelType", header: "Carburant", className: "w-[10rem]",
   336→      sortValue: (d) => (FUEL_META[d.fuelType as FuelType] ?? FUEL_META.ESSENCE).label,
   337→      render: (d) => {
   338→        const Meta = FUEL_META[d.fuelType as FuelType] ?? FUEL_META.ESSENCE
   339→        return (
   340→          <Badge variant="outline" className={cn("gap-1 text-[11px]", Meta.chip)}>
   341→            {Meta.label}
   342→            {currentKeys.has(`${d.fuelType}|${d.effectiveAt}`) && (
   343→              <span className="font-semibold text-emerald-700 dark:text-emerald-400">· en vigueur</span>
   344→            )}
   345→          </Badge>
   346→        )
   347→      },
   348→    },
   349→    {
   350→      key: "priceMillimes", header: "Prix", align: "right",
   351→      sortValue: (d) => d.priceMillimes,
   352→      render: (d) => <span className="text-sm font-semibold tabular-nums">{fmtPrice(d.priceMillimes)}</span>,
   353→    },
   354→    {
   355→      key: "delta", header: "Δ", align: "right",
   356→      sortValue: (d) => d.deltaPrev ?? 0,
   357→      render: (d) => <DecreeDelta delta={d.deltaPrev} />,
   358→    },
   359→    {
   360→      key: "marginMillimes", header: "Marge", align: "right",
   361→      sortValue: (d) => d.marginMillimes,
   362→      render: (d) => <span className="text-xs tabular-nums text-muted-foreground">{fmtPrice(d.marginMillimes)} DT</span>,
   363→    },
   364→    {
   365→      key: "decreeNote", header: "Contexte", className: "max-w-[16rem]",
   366→      sortValue: (d) => d.decreeNote ?? "",
   367→      render: (d) => (
   368→        <span className="block truncate text-xs text-muted-foreground" title={d.decreeNote ?? undefined}>
   369→          {d.decreeNote ?? "—"}
   370→        </span>
   371→      ),
   372→    },
   373→    {
   374→      key: "purge", header: "Purger", align: "center", className: "w-[3.5rem]",
   375→      render: (d) => renderPurge(d),
   376→    },
   377→  ]
   378→
   379→  const DeltaIcon = delta === null || delta === 0 ? Minus : delta > 0 ? TrendingUp : TrendingDown
   380→
   381→  return (
   382→    <div className="space-y-4">
   383→      {/* ---------- KPIs ---------- */}
   384→      <div className="grid grid-cols-2 gap-3 md:grid-cols-4">
   385→        <KpiCard
   386→          label="Décrets au registre"
   387→          value={String(data.totals.decrees)}
   388→          sub="Depuis 2021 — tous carburants"
   389→          icon={ScrollText}
   390→          tone="info"
   391→        />
   392→        <KpiCard
   393→          label="Dernière publication"
   394→          value={data.totals.lastChangeAt ? fmtDateFr(data.totals.lastChangeAt) : "—"}
   395→          sub={data.totals.nextReviewLabel}
   396→          icon={CalendarClock}
   397→          tone="success"
   398→        />
   399→        <KpiCard
   400→          label="Enseignes notifiées"
   401→          value={lastResult ? String(lastResult.notifiedBrands) : "6"}
   402→          sub={lastResult ? "Lors de la dernière publication" : "Par décret publié (cloche portail)"}
   403→          icon={BellRing}
   404→          tone="warning"
   405→        />
   406→        <KpiCard
   407→          label="Carburants suivis"
   408→          value="3"
   409→          sub="Essence · Gazole · GPL"
   410→          icon={Banknote}
   411→          tone="danger"
   412→        />
   413→      </div>
   414→
   415→      {/* ---------- Décrets en vigueur ---------- */}
   416→      <Card>
   417→        <CardHeader>
   418→          <CardTitle className="flex flex-wrap items-center gap-2">
   419→            <Scale className="h-4.5 w-4.5 text-emerald-600 dark:text-emerald-400" aria-hidden />
   420→            Décrets en vigueur
   421→          </CardTitle>
   422→          <CardDescription>
   423→            Prix pompe identiques sur tout le territoire — dernière révision en vigueur par carburant.
   424→          </CardDescription>
   425→        </CardHeader>
   426→        <CardContent className="grid grid-cols-1 gap-3 sm:grid-cols-3">
   427→          {(Object.keys(FUEL_META) as FuelType[]).map(f => {
   428→            const c = currentByFuel[f]
   429→            const Meta = FUEL_META[f]
   430→            const Icon = Meta.icon
   431→            const up = (c?.deltaMillimes ?? 0) > 0
   432→            return (
   433→              <button
   434→                key={f}
   435→                type="button"
   436→                onClick={() => prefillFor(f)}
   437→                aria-pressed={fuel === f}
   438→                className={cn(
   439→                  "group relative overflow-hidden rounded-xl border bg-card p-4 text-left transition-all hover:-translate-y-0.5 hover:shadow-md focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-emerald-500/60",
   440→                  fuel === f && `border-transparent ring-2 ${Meta.ring}`
   441→                )}
   442→                aria-label={`Préparer un décret ${Meta.label} — prix actuel ${c ? fmtPrice(c.priceMillimes) : "inconnu"}`}
   443→              >
   444→                <div className="flex items-center justify-between gap-2">
   445→                  <Badge variant="outline" className={cn("gap-1", Meta.chip)}>
   446→                    <Icon className="h-3 w-3" aria-hidden /> {Meta.label}
   447→                  </Badge>
   448→                  {fuel === f && (
   449→                    <span className="text-[10px] font-semibold uppercase tracking-wide text-emerald-600 dark:text-emerald-400">
   450→                      Sélection
   451→                    </span>
   452→                  )}
   453→                </div>
   454→                <p className="mt-2 font-bold tabular-nums leading-none">
   455→                  <span className="text-3xl">{c ? fmtPrice(c.priceMillimes) : "—"}</span>
   456→                  <span className="ml-1.5 text-sm font-medium text-muted-foreground">{Meta.unit}</span>
   457→                </p>
   458→                <div className="mt-2 flex flex-wrap items-center gap-x-3 gap-y-1 text-xs text-muted-foreground">
   459→                  {c && (
   460→                    <>
   461→                      <span className={cn(
   462→                        "inline-flex items-center gap-1 font-semibold",
   463→                        up ? "text-red-600 dark:text-red-400" : (c.deltaMillimes ?? 0) < 0 ? "text-emerald-600 dark:text-emerald-400" : ""
