  1022→function fmtNextPass(iso: string): string {
  1023→  const d = new Date(iso)
  1024→  const today = new Date()
  1025→  const isTomorrow = d.getDate() === today.getDate() + 1
  1026→    || (today.getDate() === new Date(today.getFullYear(), today.getMonth() + 1, 0).getDate() && d.getDate() === 1)
  1027→  return `${isTomorrow ? "demain" : "aujourd'hui"} 08:00`
  1028→}
