   281→function PriceHistoryChart({ data }: { data: FuelPricesResponse }) {
   282→  const [visible, setVisible] = useState<Record<FuelType, boolean>>({
   283→    ESSENCE: true, GAZOLE: true, GPL: true,
   284→  })
   285→
   286→  // Série par date de décret : une ligne par carburant, valeurs en DT.
   287→  const series = useMemo(() => {
   288→    const byDate = new Map<string, Record<string, number | string>>()
   289→    for (const h of data.history) {
   290→      const key = h.effectiveAt.slice(0, 10)
   291→      const row = byDate.get(key) ?? { date: key, dateLabel: fmtDateFr(h.effectiveAt) }
   292→      row[h.fuelType] = h.priceMillimes / 1000
   293→      byDate.set(key, row)
   294→    }
   295→    return [...byDate.values()].sort((a, b) => String(a.date).localeCompare(String(b.date)))
   296→  }, [data.history])
   297→
   298→  // Points de changement (pour pastilles) : dates où un carburant visible a bougé.
   299→  const changeDots = useMemo(() => {
   300→    const dots: { date: string; fuel: FuelType; value: number }[] = []
   301→    const prev = new Map<FuelType, number>()
   302→    for (const h of data.history) {
   303→      const p = prev.get(h.fuelType)
   304→      if (p !== undefined && p !== h.priceMillimes && visible[h.fuelType]) {
   305→        dots.push({ date: h.effectiveAt.slice(0, 10), fuel: h.fuelType, value: h.priceMillimes / 1000 })
   306→      }
   307→      prev.set(h.fuelType, h.priceMillimes)
   308→    }
   309→    return dots
   310→  }, [data.history, visible])
   311→
   312→  const anyVisible = Object.values(visible).some(Boolean)
   313→
   314→  return (
   315→    <Card className="h-full">
   316→      <CardHeader className="pb-2">
   317→        <div className="flex flex-wrap items-start justify-between gap-2">
   318→          <div>
   319→            <CardTitle className="text-base leading-tight">Historique des décrets — 2021 → aujourd&apos;hui</CardTitle>
   320→            <CardDescription className="mt-0.5 text-xs">
   321→              Prix pompe par carburant (DT/{`L`}) — chaque marche = un décret ministériel
   322→            </CardDescription>
   323→          </div>
   324→          <div className="flex flex-wrap gap-1.5">
   325→            {(Object.keys(FUEL_META) as FuelType[]).map(f => (
   326→              <button
   327→                key={f}
   328→                type="button"
   329→                onClick={() => setVisible(v => ({ ...v, [f]: !v[f] }))}
   330→                aria-pressed={visible[f]}
   331→                className={cn(
   332→                  "inline-flex items-center gap-1.5 rounded-full border px-2.5 py-1 text-[11px] font-medium transition-all focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-emerald-500",
   333→                  visible[f]
   334→                    ? "border-border bg-background shadow-sm hover:bg-muted/60"
   335→                    : "border-transparent bg-muted/40 text-muted-foreground/60 hover:bg-muted",
   336→                )}
   337→              >
   338→                <span
   339→                  className="h-2 w-2 rounded-full"
   340→                  style={{ background: visible[f] ? FUEL_META[f].color : "#94a3b8" }}
   341→                  aria-hidden
   342→                />
   343→                {FUEL_META[f].label.split(" ")[0]}
   344→              </button>
   345→            ))}
   346→          </div>
   347→        </div>
   348→      </CardHeader>
   349→      <CardContent className="h-[340px] pl-0">
   350→        {anyVisible ? (
   351→          <ResponsiveContainer width="100%" height="100%">
   352→            <ComposedChart data={series} margin={{ top: 10, right: 16, left: 0, bottom: 0 }}>
   353→              <CartesianGrid stroke={GRID_STROKE} strokeDasharray="3 3" vertical={false} />
   354→              <XAxis
   355→                dataKey="dateLabel"
   356→                tick={{ ...AXIS_STYLE, fontSize: 10 }}
   357→                tickLine={false}
   358→                axisLine={false}
   359→                interval="preserveStartEnd"
   360→                minTickGap={24}
   361→              />
   362→              <YAxis
   363→                tick={{ ...AXIS_STYLE, fontSize: 10 }}
   364→                tickLine={false}
   365→                axisLine={false}
   366→                width={44}
   367→                tickFormatter={(v: number) => v.toFixed(2)}
   368→                domain={["auto", "auto"]}
   369→              />
   370→              <Tooltip
   371→                contentStyle={tooltipStyle()}
   372→                formatter={(value: number | string, name: string) => [
   373→                  `${Number(value).toFixed(3).replace(".", ",")} DT`,
   374→                  FUEL_META[name as FuelType]?.label ?? name,
   375→                ]}
   376→              />
   377→              {(Object.keys(FUEL_META) as FuelType[]).filter(f => visible[f]).map(f => (
   378→                <Line
   379→                  key={f}
   380→                  type="stepAfter"
   381→                  dataKey={f}
   382→                  stroke={FUEL_META[f].color}
   383→                  strokeWidth={2.2}
   384→                  dot={false}
   385→                  activeDot={{ r: 4, strokeWidth: 2 }}
   386→                />
   387→              ))}
   388→              {changeDots.map((d, i) => (
   389→                <ReferenceDot
   390→                  key={`${d.date}-${d.fuel}-${i}`}
   391→                  x={d.date}
   392→                  y={d.value}
   393→                  r={3.5}
   394→                  fill={FUEL_META[d.fuel].color}
   395→                  stroke="var(--background)"
   396→                  strokeWidth={1.5}
   397→                  // dataKey inutile ici : les points s'affichent sur leur série
   398→                  ifOverflow="extendDomain"
   399→                />
   400→              ))}
   401→            </ComposedChart>
   402→          </ResponsiveContainer>
   403→        ) : (
   404→          <div className="flex h-full items-center justify-center text-sm text-muted-foreground">
   405→            Sélectionnez au moins un carburant.
   406→          </div>
   407→        )}
   408→      </CardContent>
   409→    </Card>
   410→  )
   411→}

   413→// ---------- Simulateur d'impact marge (A.6.4 — « marge par litre ») ----------
   414→
   415→function MarginSimulator({ current }: { current: FuelPriceCurrent[] }) {
   416→  // Paramètres (défauts réalistes station tunisienne).
   417→  const [mainFuel, setMainFuel] = useState<FuelType>("GAZOLE")
   418→  const [traffic, setTraffic] = useState(900) // véh/j captifs
   419→  const [litresPerVeh, setLitresPerVeh] = useState(32) // L pleins moyens
   420→  const [fillRate, setFillRate] = useState(70) // % des véhicules qui font le plein
   421→  const [shareMain, setShareMain] = useState(65) // % essence vs gazole du volume
   422→  const [includeSecondary, setIncludeSecondary] = useState(true)
   423→
   424→  const priceByFuel = useMemo(
   425→    () => Object.fromEntries(current.map(c => [c.fuelType, c])) as Record<FuelType, FuelPriceCurrent>,
   426→    [current],
   427→  )
   428→
   429→  const main = priceByFuel[mainFuel]
   430→  const secondaryFuel: FuelType = mainFuel === "ESSENCE" ? "GAZOLE" : "ESSENCE"
   431→  const secondary = priceByFuel[secondaryFuel]
   432→
   433→  // Volumes annuels.
   434→  const fuelingsPerDay = (traffic * fillRate) / 100
   435→  const litresMainPerYear = fuelingsPerDay * litresPerVeh * (shareMain / 100) * 365
   436→  const litresSecondaryPerYear = includeSecondary
   437→    ? fuelingsPerDay * litresPerVeh * ((100 - shareMain) / 100) * 365
   438→    : 0
   439→
   440→  const marginMainPerYear = litresMainPerYear * (main?.marginMillimes ?? 0) / 1000 // DT
   441→  const marginSecondaryPerYear = litresSecondaryPerYear * (secondary?.marginMillimes ?? 0) / 1000
   442→  const marginTotal = marginMainPerYear + marginSecondaryPerYear
   443→
   444→  // Sensibilités.
   445→  const plus5Volume = marginTotal * 0.05
   446→  const oneMoreVehiclePerDay = (litresPerVeh * (fillRate / 100) * 365) * (
   447→    (shareMain / 100) * (main?.marginMillimes ?? 0) / 1000
   448→    + (includeSecondary ? ((100 - shareMain) / 100) * (secondary?.marginMillimes ?? 0) / 1000 : 0)
   449→  ) // DT/an par véhicule/jour supplémentaire
   450→
   451→  const fmtDT = (v: number) => `${NUM.format(Math.round(v))} DT`
   452→
   453→  const savePreset = () => {
   454→    toast.success("Scénario marge copié", {
   455→      description: `Marge annuelle : ${fmtDT(marginTotal)} — ${NUM.format(Math.round(litresMainPerYear + litresSecondaryPerYear))} L/an.`,
   456→    })
   457→  }
   458→
   459→  return (
   460→    <Card className="h-full border-emerald-200/70 dark:border-emerald-900/60">
   461→      <CardHeader className="bg-gradient-to-br from-emerald-50/70 via-transparent to-transparent pb-3 dark:from-emerald-950/40">
   462→        <div className="flex items-start justify-between gap-2">
   463→          <div className="flex items-center gap-2.5">
   464→            <span className="flex h-8 w-8 items-center justify-center rounded-lg bg-emerald-600 text-white shadow-sm">
   465→              <Gauge className="h-4 w-4" aria-hidden />
   466→            </span>
   467→            <div>
   468→              <CardTitle className="text-base leading-tight">Simulateur d&apos;impact marge</CardTitle>
   469→              <CardDescription className="mt-0.5 text-xs">
   470→                La « marge par litre » du calculateur de rentabilité (A.6.4), chiffrée pour une station
   471→              </CardDescription>

   632→function DecreesTable({ data }: { data: FuelPricesResponse }) {
   633→  // Lignes triées par date décroissante, ancien → nouveau prix par carburant.
   634→  const rows = useMemo(() => {
   635→    const prev = new Map<string, { price: number; note: string | null; at: string }>()
   636→    return [...data.history]
   637→      .sort((a, b) => b.effectiveAt.localeCompare(a.effectiveAt))
   638→      .map(h => {
   639→        const p = prev.get(h.fuelType)
   640→        prev.set(h.fuelType, { price: h.priceMillimes, note: h.decreeNote, at: h.effectiveAt })
   641→        return {
   642→          ...h,
   643→          prevPrice: p?.price ?? null,
   644→          delta: p ? h.priceMillimes - p.price : null,
   645→        }
   646→      })
   647→  }, [data.history])
   648→
   649→  return (
   650→    <Card>
   651→      <CardHeader className="pb-3">
   652→        <CardTitle className="text-base">Registre des décrets</CardTitle>
   653→        <CardDescription className="text-xs">
   654→          {data.totals.decrees} décrets référencés depuis 2021 — {data.totals.nextReviewLabel}
   655→        </CardDescription>
   656→      </CardHeader>
   657→      <CardContent className="p-0">
   658→        <ScrollArea className="max-h-96">
   659→          <table className="w-full text-sm">
   660→            <thead className="sticky top-0 z-10 bg-muted/80 backdrop-blur">
   661→              <tr className="border-b text-left">
   662→                <th className="px-4 py-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground">Date</th>
   663→                <th className="px-2 py-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground">Carburant</th>
   664→                <th className="px-2 py-2 text-right text-xs font-semibold uppercase tracking-wide text-muted-foreground">Ancien</th>
   665→                <th className="px-2 py-2 text-right text-xs font-semibold uppercase tracking-wide text-muted-foreground">Nouveau</th>
   666→                <th className="px-2 py-2 text-right text-xs font-semibold uppercase tracking-wide text-muted-foreground">Δ</th>
   667→                <th className="px-2 py-2 text-right text-xs font-semibold uppercase tracking-wide text-muted-foreground">Marge</th>
   668→                <th className="px-4 py-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground">Contexte</th>
   669→              </tr>
   670→            </thead>
   671→            <tbody>
   672→              {rows.map((r, i) => {
   673→                const meta = FUEL_META[r.fuelType]
   674→                const delta = r.delta
   675→                return (
   676→                  <tr
   677→                    key={r.id}
   678→                    className={cn(
   679→                      "border-b/60 transition-colors hover:bg-muted/50",
   680→                      i % 2 === 1 && "bg-muted/20",
   681→                    )}
   682→                  >
   683→                    <td className="whitespace-nowrap px-4 py-2 text-xs tabular-nums">{fmtDateFr(r.effectiveAt)}</td>
   684→                    <td className="px-2 py-2">
   685→                      <span className="inline-flex items-center gap-1.5 text-xs font-medium">
   686→                        <span className="h-2 w-2 rounded-full" style={{ background: meta.color }} aria-hidden />
   687→                        {meta.label}
   688→                      </span>
   689→                    </td>
   690→                    <td className="px-2 py-2 text-right text-xs tabular-nums text-muted-foreground">
   691→                      {r.prevPrice !== null ? fmtPrice(r.prevPrice) : "—"}
   692→                    </td>
   693→                    <td className="px-2 py-2 text-right text-xs font-bold tabular-nums">{fmtPrice(r.priceMillimes)}</td>
   694→                    <td className="px-2 py-2 text-right text-xs tabular-nums">
   695→                      {delta === null ? (
   696→                        <span className="text-muted-foreground">—</span>
   697→                      ) : delta === 0 ? (
   698→                        <span className="rounded bg-slate-100 px-1.5 py-0.5 font-semibold text-slate-600 dark:bg-slate-800 dark:text-slate-400">
   699→                          0
   700→                        </span>
   701→                      ) : (
   702→                        <span
   703→                          className={cn(
   704→                            "rounded px-1.5 py-0.5 font-semibold",
   705→                            delta > 0
   706→                              ? "bg-red-100 text-red-700 dark:bg-red-950 dark:text-red-400"
   707→                              : "bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-400",
   708→                          )}
   709→                        >
   710→                          {delta > 0 ? "+" : ""}{fmtMillimes(delta)}
   711→                        </span>
   712→                      )}
   713→                    </td>
   714→                    <td className="px-2 py-2 text-right text-xs tabular-nums text-muted-foreground">
   715→                      {fmtMillimes(r.marginMillimes)}/L
   716→                    </td>

   723→            </tbody>
   724→          </table>
   725→        </ScrollArea>
   726→      </CardContent>
   727→    </Card>
   728→  )
