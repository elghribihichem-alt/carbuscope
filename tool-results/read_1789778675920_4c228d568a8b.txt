    77→// ---------- Onglet ----------
    78→
    79→export function BoDecrees() {
    80→  const [data, setData] = useState<FuelPricesResponse | null>(null)
    81→  const [error, setError] = useState<string | null>(null)
    82→  const [reloadKey, setReloadKey] = useState(0)
    83→
    84→  // Formulaire de publication
    85→  const [fuel, setFuel] = useState<FuelType>("ESSENCE")
    86→  const [price, setPrice] = useState("")
    87→  const [margin, setMargin] = useState("")
    88→  const [effectiveAt, setEffectiveAt] = useState(todayIso())
    89→  const [note, setNote] = useState("")
    90→  const [submitting, setSubmitting] = useState(false)
    91→  const [lastResult, setLastResult] = useState<DecreeResponse | null>(null)
    92→
    93→  useEffect(() => {
    94→    const ac = new AbortController()
    95→    fetch("/api/prices", { signal: ac.signal })
    96→      .then(r => (r.ok ? r.json() : Promise.reject(new Error(`HTTP ${r.status}`))))
    97→      .then((j: FuelPricesResponse) => { setData(j); setError(null) })
    98→      .catch(e => {
    99→        if (e instanceof DOMException && e.name === "AbortError") return
   100→        setError("Impossible de charger le registre des décrets.")
   101→      })
   102→    return () => ac.abort()
   103→  }, [reloadKey])
   104→
   105→  const currentByFuel = useMemo(() => {
   106→    const map: Partial<Record<FuelType, FuelPricesResponse["current"][number]>> = {}
   107→    for (const c of data?.current ?? []) map[c.fuelType] = c
   108→    return map
   109→  }, [data])
   110→
   111→  const current = currentByFuel[fuel]
   112→
   113→  // Suppression d'un décret erroné (T24) — confirmation obligatoire, refus
   114→  // serveur si dernier décret d'un carburant. Compte par carburant côté client
   115→  // pour désactiver visuellement les boutons concernés.
   116→  const countByFuel = useMemo(() => {
   117→    const m = new Map<string, number>()
   118→    for (const d of data?.history ?? []) m.set(d.fuelType, (m.get(d.fuelType) ?? 0) + 1)
   119→    return m
   120→  }, [data])
   121→
   122→  const [deletingId, setDeletingId] = useState<string | null>(null)
   123→  const deleteDecree = useCallback(async (d: FuelPriceDTO) => {
   124→    setDeletingId(d.id)
   125→    try {
   126→      const r = await fetch(`/api/prices/${d.id}`, { method: "DELETE" })
   127→      const j = await r.json().catch(() => ({}))
   128→      if (!r.ok) throw new Error(j.error ?? `HTTP ${r.status}`)
   129→      toast.success("Décret retiré du registre", {
   130→        description: j.fallbackPriceMillimes != null
   131→          ? `Prix en vigueur recalculé sur le décret précédent : ${(j.fallbackPriceMillimes / 1000).toFixed(3).replace(".", ",")} DT.`
   132→          : "Le registre chronologique a été mis à jour.",
   133→      })
   134→      setLastResult(null)
   135→      setReloadKey(k => k + 1)
   136→    } catch (e) {
   137→      toast.error("Suppression refusée", { description: e instanceof Error ? e.message : "Erreur inattendue." })
   138→    } finally {
   139→      setDeletingId(null)
   140→    }
   141→  }, [])
   142→
   143→  // Pré-remplissage intelligent au changement de carburant : prix proposé =
   144→  // décret en vigueur + 25 millimes (révision typique), marge inchangée.
   145→  const prefillFor = (f: FuelType) => {
   146→    const c = currentByFuel[f]
   147→    setFuel(f)
   148→    setPrice(c ? String(c.priceMillimes + 25) : "")
   149→    setMargin(c ? String(c.marginMillimes) : "")
   150→  }
   151→
   152→  // Pré-remplissage initial dès que les données arrivent.
   153→  useEffect(() => {
   154→    if (data && price === "" && margin === "") {
   155→      const c = currentByFuel.ESSENCE
   156→      if (c) { setPrice(String(c.priceMillimes + 25)); setMargin(String(c.marginMillimes)) }
   157→    }
   158→  }, [data, price, margin, currentByFuel])
   159→
   160→  const priceNum = Number(price)
   161→  const marginNum = Number(margin)
   162→  const priceValid = Number.isInteger(priceNum) && priceNum >= 300 && priceNum <= 10_000
   163→  const marginValid = Number.isInteger(marginNum) && marginNum >= 0 && marginNum <= 1_000
   164→  const dateValid = /^\d{4}-\d{2}-\d{2}$/.test(effectiveAt) && !Number.isNaN(new Date(effectiveAt).getTime())
   165→  const formValid = priceValid && marginValid && dateValid
   166→
   167→  const delta = current ? priceNum - current.priceMillimes : null
   168→  const marginDelta = current ? marginNum - current.marginMillimes : null
   169→  const deltaPct = current && delta !== null && current.priceMillimes > 0
   170→    ? (delta / current.priceMillimes) * 100
   171→    : null
   172→
   173→  const publish = useCallback(async () => {
   174→    if (!formValid || submitting) return
   175→    setSubmitting(true)
   176→    try {
   177→      const r = await fetch("/api/prices", {
   178→        method: "POST",
   179→        headers: { "Content-Type": "application/json" },
   180→        body: JSON.stringify({
   181→          fuelType: fuel,
   182→          priceMillimes: priceNum,
   183→          marginMillimes: marginNum,
   184→          effectiveAt,
   185→          decreeNote: note.trim() || undefined,
   186→        }),
   187→      })
   188→      const j = (await r.json()) as DecreeResponse & { error?: string }
   189→      if (!r.ok) throw new Error(j.error ?? `HTTP ${r.status}`)
   190→      setLastResult(j)
   191→      toast.success("Décret publié et enseignes notifiées", {
   192→        description: `${FUEL_META[fuel].label} : ${fmtPrice(j.decree.priceMillimes)} ${FUEL_META[fuel].unit} — ${j.notifiedBrands} enseigne(s) notifiée(s) via la cloche du portail.`,
   193→        duration: 6000,
   194→      })
   195→      setNote("")
   196→      setReloadKey(k => k + 1)
   197→    } catch (e) {
   198→      toast.error("Publication refusée", {
   199→        description: e instanceof Error ? e.message : "Erreur inconnue.",
   200→      })
   201→    } finally {
   202→      setSubmitting(false)
   203→    }
   204→  }, [formValid, submitting, fuel, priceNum, marginNum, effectiveAt, note])
   205→
   206→  const resetForm = () => {
   207→    const c = currentByFuel[fuel]
   208→    setPrice(c ? String(c.priceMillimes + 25) : "")
   209→    setMargin(c ? String(c.marginMillimes) : "")
   210→    setEffectiveAt(todayIso())
   211→    setNote("")
   212→  }
   213→
   214→  if (error && !data) {
   215→    return (
   216→      <div className="p-1">
   217→        <ErrorState message={error} retry={() => setReloadKey(k => k + 1)} />
   218→      </div>
   219→    )
   220→  }
   221→  if (!data) return <TabSkeleton title="Chargement du registre des décrets…" />
   222→
   223→  // Registre chronologique inversé (plus récent d'abord) + badge « en vigueur » :
   224→  // le décret courant d'un carburant est le plus récent de son historique.
   225→  const historyDesc = [...data.history].sort(
   226→    (a, b) => new Date(b.effectiveAt).getTime() - new Date(a.effectiveAt).getTime()
   227→  )
   228→  const currentKeys = new Set(
   229→    data.current.map(c => `${c.fuelType}|${c.effectiveAt}`)
   230→  )
   231→
   232→  const DeltaIcon = delta === null || delta === 0 ? Minus : delta > 0 ? TrendingUp : TrendingDown
   233→
   234→  return (
   235→    <div className="space-y-4">
   236→      {/* ---------- KPIs ---------- */}
   237→      <div className="grid grid-cols-2 gap-3 md:grid-cols-4">
   238→        <KpiCard
   239→          label="Décrets au registre"
   240→          value={String(data.totals.decrees)}
   241→          sub="Depuis 2021 — tous carburants"
   242→          icon={ScrollText}
   243→          tone="info"
   244→        />
   245→        <KpiCard
   246→          label="Dernière publication"
   247→          value={data.totals.lastChangeAt ? fmtDateFr(data.totals.lastChangeAt) : "—"}
   248→          sub={data.totals.nextReviewLabel}
   249→          icon={CalendarClock}
   250→          tone="success"
   251→        />
   252→        <KpiCard
   253→          label="Enseignes notifiées"
   254→          value={lastResult ? String(lastResult.notifiedBrands) : "6"}
   255→          sub={lastResult ? "Lors de la dernière publication" : "Par décret publié (cloche portail)"}
   256→          icon={BellRing}
   257→          tone="warning"
   258→        />
   259→        <KpiCard
   260→          label="Carburants suivis"
   261→          value="3"
   262→          sub="Essence · Gazole · GPL"
   263→          icon={Banknote}
   264→          tone="danger"
   265→        />
   266→      </div>
   267→
   268→      {/* ---------- Décrets en vigueur ---------- */}
   269→      <Card>
   270→        <CardHeader>
   271→          <CardTitle className="flex flex-wrap items-center gap-2">
   272→            <Scale className="h-4.5 w-4.5 text-emerald-600 dark:text-emerald-400" aria-hidden />
   273→            Décrets en vigueur
   274→          </CardTitle>
   275→          <CardDescription>
   276→            Prix pompe identiques sur tout le territoire — dernière révision en vigueur par carburant.
   277→          </CardDescription>
   278→        </CardHeader>
   279→        <CardContent className="grid grid-cols-1 gap-3 sm:grid-cols-3">
   280→          {(Object.keys(FUEL_META) as FuelType[]).map(f => {
   281→            const c = currentByFuel[f]
   282→            const Meta = FUEL_META[f]
   283→            const Icon = Meta.icon
   284→            const up = (c?.deltaMillimes ?? 0) > 0
   285→            return (
   286→              <button
   287→                key={f}
   288→                type="button"
   289→                onClick={() => prefillFor(f)}
   290→                aria-pressed={fuel === f}
   291→                className={cn(
   292→                  "group relative overflow-hidden rounded-xl border bg-card p-4 text-left transition-all hover:-translate-y-0.5 hover:shadow-md focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-emerald-500/60",
   293→                  fuel === f && `border-transparent ring-2 ${Meta.ring}`
   294→                )}
   295→                aria-label={`Préparer un décret ${Meta.label} — prix actuel ${c ? fmtPrice(c.priceMillimes) : "inconnu"}`}
   296→              >
   297→                <div className="flex items-center justify-between gap-2">
   298→                  <Badge variant="outline" className={cn("gap-1", Meta.chip)}>
   299→                    <Icon className="h-3 w-3" aria-hidden /> {Meta.label}
   300→                  </Badge>
   301→                  {fuel === f && (
   302→                    <span className="text-[10px] font-semibold uppercase tracking-wide text-emerald-600 dark:text-emerald-400">
   303→                      Sélection
   304→                    </span>
   305→                  )}
   306→                </div>
   307→                <p className="mt-2 font-bold tabular-nums leading-none">
   308→                  <span className="text-3xl">{c ? fmtPrice(c.priceMillimes) : "—"}</span>
   309→                  <span className="ml-1.5 text-sm font-medium text-muted-foreground">{Meta.unit}</span>
   310→                </p>
   311→                <div className="mt-2 flex flex-wrap items-center gap-x-3 gap-y-1 text-xs text-muted-foreground">
   312→                  {c && (
   313→                    <>
   314→                      <span className={cn(
   315→                        "inline-flex items-center gap-1 font-semibold",
   316→                        up ? "text-red-600 dark:text-red-400" : (c.deltaMillimes ?? 0) < 0 ? "text-emerald-600 dark:text-emerald-400" : ""
   317→                      )}>
   318→                        {up ? "▲" : (c.deltaMillimes ?? 0) < 0 ? "▼" : "＝"}
   319→                        {c.deltaMillimes !== null ? `${c.deltaMillimes > 0 ? "+" : ""}${c.deltaMillimes} m` : "premier"}
   320→                      </span>
   321→                      <span>depuis le {fmtDateFr(c.effectiveAt)}</span>
   322→                      <span>· marge {fmtPrice(c.marginMillimes)} DT</span>
   323→                    </>
   324→                  )}
   325→                </div>
   326→                <p className="mt-2 text-[11px] leading-snug text-muted-foreground/80">
   327→                  Cliquer pour préparer le prochain décret {Meta.label.toLowerCase()}.
   328→                </p>
   329→              </button>
   330→            )
   331→          })}
   332→        </CardContent>
   333→      </Card>
   334→
   335→      {/* ---------- Publication d'un décret ---------- */}
   336→      <Card className="border-emerald-200/70 dark:border-emerald-900/50">
   337→        <CardHeader>
   338→          <CardTitle className="flex flex-wrap items-center gap-2">
   339→            <Send className="h-4.5 w-4.5 text-emerald-600 dark:text-emerald-400" aria-hidden />
   340→            Publier un nouveau décret
   341→          </CardTitle>
   342→          <CardDescription>
   343→            La publication met à jour l&apos;observatoire B2B, le mobile B2C et notifie chaque enseigne
   344→            via la cloche de notification du portail.
   345→          </CardDescription>
   346→        </CardHeader>
   347→        <CardContent className="grid grid-cols-1 gap-6 lg:grid-cols-[1fr_20rem]">
   348→          {/* Formulaire */}
   349→          <div className="space-y-4">
   350→            <div className="space-y-2">
   351→              <Label>Carburant concerné</Label>
   352→              <div className="grid grid-cols-3 gap-2">
   353→                {(Object.keys(FUEL_META) as FuelType[]).map(f => {
   354→                  const Meta = FUEL_META[f]
   355→                  const Icon = Meta.icon
   356→                  const active = fuel === f
   357→                  return (
   358→                    <button
   359→                      key={f}
   360→                      type="button"
   361→                      onClick={() => prefillFor(f)}
   362→                      aria-pressed={active}
   363→                      className={cn(
   364→                        "flex items-center justify-center gap-2 rounded-lg border px-3 py-2.5 text-sm font-medium transition-all focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-emerald-500/60",
   365→                        active
   366→                          ? cn("border-transparent ring-2", Meta.ring, "bg-muted/60")
   367→                          : "border-border bg-background hover:bg-muted/50"
   368→                      )}
   369→                    >
   370→                      <Icon className="h-4 w-4" aria-hidden />
   371→                      {Meta.label}
   372→                    </button>
   373→                  )
   374→                })}
   375→              </div>
   376→            </div>
   377→
   378→            <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   379→              <div className="space-y-2">
   380→                <Label htmlFor="decree-price">
   381→                  Nouveau prix pompe <span className="text-muted-foreground">(millimes)</span>
   382→                </Label>
   383→                <Input
   384→                  id="decree-price"
   385→                  inputMode="numeric"
   386→                  value={price}
   387→                  onChange={e => setPrice(e.target.value.replace(/[^\d]/g, ""))}
   388→                  placeholder="2675"
   389→                  aria-invalid={price !== "" && !priceValid}
   390→                  aria-describedby="decree-price-hint"
   391→                  className="tabular-nums"
   392→                />
   393→                <p id="decree-price-hint" className={cn(
   394→                  "text-xs",
   395→                  priceValid ? "text-muted-foreground" : "text-red-600 dark:text-red-400"
   396→                )}>
   397→                  {priceValid
   398→                    ? `= ${fmtPrice(priceNum)} ${FUEL_META[fuel].unit} · entier entre 300 et 10 000`
   399→                    : "Entier en millimes requis (300 – 10 000)."}
   400→                </p>
   401→              </div>
   402→              <div className="space-y-2">
   403→                <Label htmlFor="decree-margin">
   404→                  Marge détaillant <span className="text-muted-foreground">(millimes/L)</span>
   405→                </Label>
   406→                <Input
   407→                  id="decree-margin"
   408→                  inputMode="numeric"
   409→                  value={margin}
   410→                  onChange={e => setMargin(e.target.value.replace(/[^\d]/g, ""))}
   411→                  placeholder="145"
   412→                  aria-invalid={margin !== "" && !marginValid}
   413→                  aria-describedby="decree-margin-hint"
   414→                  className="tabular-nums"
   415→                />
   416→                <p id="decree-margin-hint" className={cn(
   417→                  "text-xs",
   418→                  marginValid ? "text-muted-foreground" : "text-red-600 dark:text-red-400"
   419→                )}>
   420→                  {marginValid
   421→                    ? `= ${fmtPrice(marginNum)} DT par litre vendu`
   422→                    : "Entier en millimes requis (0 – 1 000)."}
   423→                </p>
   424→              </div>
   425→            </div>
   426→
   427→            <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   428→              <div className="space-y-2">
   429→                <Label htmlFor="decree-date">Date d&apos;application</Label>
   430→                <Input
   431→                  id="decree-date"
   432→                  type="date"
   433→                  value={effectiveAt}
   434→                  onChange={e => setEffectiveAt(e.target.value)}
   435→                  aria-invalid={!dateValid}
   436→                />
   437→                {!dateValid && (
   438→                  <p className="text-xs text-red-600 dark:text-red-400">Date invalide.</p>
   439→                )}
   440→              </div>
   441→              <div className="space-y-2">
   442→                <Label htmlFor="decree-note">Référence / contexte <span className="text-muted-foreground">(optionnel)</span></Label>
   443→                <Input
   444→                  id="decree-note"
   445→                  value={note}
   446→                  maxLength={200}
   447→                  onChange={e => setNote(e.target.value)}
   448→                  placeholder="Révision trimestrielle — décret n° 2026-XX"
   449→                  aria-describedby="decree-note-hint"
   450→                />
   451→                <p id="decree-note-hint" className="text-xs text-muted-foreground">
   452→                  {200 - note.length} caractères restants.
   453→                </p>
   454→              </div>
   455→            </div>
   456→
   457→            <div className="flex flex-wrap items-center gap-2 pt-1">
   458→              <Button
   459→                onClick={() => void publish()}
   460→                disabled={!formValid || submitting}
   461→                className="gap-2"
   462→                size="lg"
   463→              >
   464→                {submitting ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <Send className="h-4 w-4" aria-hidden />}
   465→                {submitting ? "Publication…" : "Publier & notifier les enseignes"}
   466→              </Button>
   467→              <Button variant="outline" size="lg" onClick={resetForm} className="gap-2">
   468→                <RotateCcw className="h-3.5 w-3.5" aria-hidden />
   469→                Réinitialiser
   470→              </Button>
   471→            </div>
   472→          </div>
   473→
   474→          {/* Aperçu de contrôle (vérification avant publication) */}
   475→          <div className="space-y-3 rounded-xl border bg-muted/30 p-4">
   476→            <p className="flex items-center gap-1.5 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   477→              <ShieldAlert className="h-3.5 w-3.5" aria-hidden />
   478→              Contrôle avant publication
   479→            </p>
   480→
   481→            <div className="rounded-lg border bg-background p-3">
   482→              <p className="text-xs text-muted-foreground">Décret en vigueur ({FUEL_META[fuel].label})</p>
   483→              <p className="mt-1 text-lg font-bold tabular-nums leading-tight">
   484→                {current ? fmtPrice(current.priceMillimes) : "—"}
