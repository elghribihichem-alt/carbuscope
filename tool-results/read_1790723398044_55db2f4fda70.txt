    81→    81→function sortField(key: SortKey): string {
    82→    82→  return key.slice(0, key.lastIndexOf("-"))
    83→    83→}

    85→    85→function toggleSort(cur: SortKey, target: SortKey): SortKey {
    86→    86→  if (sortField(cur) !== sortField(target)) return target
    87→    87→  const field = sortField(cur)
    88→    88→  return cur.endsWith("-asc") ? (`${field}-desc` as SortKey) : (`${field}-asc` as SortKey)
    89→    89→}

    91→    91→function compareStations(a: StationDTO, b: StationDTO, sort: SortKey): number {
    92→    92→  const field = sortField(sort)
    93→    93→  const mul = sort.endsWith("-asc") ? 1 : -1
    94→    94→  let r = 0
    95→    95→  if (field === "code") r = a.code.localeCompare(b.code, "fr")
    96→    96→  else if (field === "name") r = a.name.localeCompare(b.name, "fr")
    97→    97→  else if (field === "gov") r = a.gouvernorat.localeCompare(b.gouvernorat, "fr") || a.delegation.localeCompare(b.delegation, "fr")
    98→    98→  else if (field === "traffic") r = a.traffic - b.traffic
    99→    99→  else if (field === "verified") r = new Date(a.lastVerified).getTime() - new Date(b.lastVerified).getTime()
   100→   100→  else if (field === "rating") r = a.rating - b.rating
   101→   101→  else if (field === "status") r = (STATUS_ORDER[a.status] ?? 9) - (STATUS_ORDER[b.status] ?? 9)
   102→   102→  if (r === 0) return a.code.localeCompare(b.code, "fr")
   103→   103→  return r * mul
   104→   104→}

   106→   106→function SortTh({
   107→   107→  label, target, sort, onSort, className,
   108→   108→}: {
   109→   109→  label: string
   110→   110→  target: SortKey
   111→   111→  sort: SortKey
   112→   112→  onSort: (k: SortKey) => void
   113→   113→  className?: string
   114→   114→}) {
   115→   115→  const isActive = sortField(sort) === sortField(target)
   116→   116→  const asc = sort.endsWith("-asc")
   117→   117→  return (
   118→   118→    <th scope="col" className={cn("h-10 whitespace-nowrap px-2 text-left align-middle font-medium", className)}>
   119→   119→      <button
   120→   120→        type="button"
   121→   121→        onClick={() => onSort(target)}
   122→   122→        className="inline-flex items-center gap-1 text-[11px] font-medium uppercase tracking-wide text-muted-foreground transition-colors hover:text-foreground focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-ring"
   123→   123→        aria-label={`Trier par ${label}${isActive ? (asc ? " (croissant)" : " (décroissant)") : ""}`}
   124→   124→      >
   125→   125→        {label}
   126→   126→        {isActive
   127→   127→          ? (asc ? <ArrowUp className="h-3 w-3" aria-hidden /> : <ArrowDown className="h-3 w-3" aria-hidden />)
   128→   128→          : <ArrowUpDown className="h-3 w-3 opacity-40" aria-hidden />}
   129→   129→      </button>
   130→   130→    </th>
   131→   131→  )
   132→   132→}
