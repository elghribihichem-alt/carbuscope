    66→let staticPromise: Promise<StaticLayers> | null = null
    67→function loadStatic(): Promise<StaticLayers> {
    68→  if (!staticPromise) {
    69→    staticPromise = (async () => {
    70→      const safe = async (url: string, key: string) => {
    71→        try {
    72→          const r = await fetch(url)
    73→          if (!r.ok) return []
    74→          const j = await r.json()
    75→          return j[key] ?? []
    76→        } catch { return [] }
    77→      }
    78→      const [roads, projects, governorats, services] = await Promise.all([
    79→        safe("/api/roads", "roads"),
    80→        safe("/api/projects", "projects"),
    81→        safe("/api/governorats", "governorats"),
    82→        safe("/api/services", "services"),
    83→      ])
    84→      return { roads, projects, governorats, services }
    85→    })()
    86→  }
    87→  return staticPromise
    88→}

    94→}

    99→export function TunisiaMap({
   100→  stations, ownBrandSlug, selectedCode, onSelect, onViewDetail,
   101→  layers, showLayerControl = true, tools = false, showLegend = true,
   102→  className = "h-[560px]", focusCode, fitTo, autoFitOnLoad = false, govStationCounts,
   103→}: TunisiaMapProps) {
   104→  const wrapRef = useRef<HTMLDivElement>(null)
   105→  const svgRef = useRef<SVGSVGElement>(null)
   106→  const movedRef = useRef(0)
   107→
   108→  const [view, setView] = useState({ k: 1, x: 0, y: 0 })
   109→  const [hoverCode, setHoverCode] = useState<string | null>(null)
   110→  const [selCode, setSelCode] = useState<string | null>(selectedCode ?? null)
   111→  const [layerState, setLayerState] = useState<MapLayers>({
   112→    stations: true, roads: true, traffic: false, zones: false, projects: false, fleet: false,
   113→    ...(layers ?? {}),
   114→  })
   115→  const [staticData, setStaticData] = useState<StaticLayers | null>(null)
   116→  const [detail, setDetail] = useState<StationDetail | null>(null)
   117→  const [detailCode, setDetailCode] = useState<string | null>(null)
   118→  const [showHistory, setShowHistory] = useState(false)
   119→
   120→  // outils
   121→  const [tool, setTool] = useState<Tool>(null)
   122→  const [measurePts, setMeasurePts] = useState<[number, number][]>([])
   123→  const [radiusCenter, setRadiusCenter] = useState<[number, number] | null>(null)
   124→  const [radiusKm, setRadiusKm] = useState(10)
   125→  const [polygonPts, setPolygonPts] = useState<[number, number][]>([])
   126→  const [polygonClosed, setPolygonClosed] = useState(false)
   127→
   128→  useEffect(() => { loadStatic().then(setStaticData).catch(() => {}) }, [])
   129→
   130→  // ---------- Conversions écran ↔ SVG (letterbox-safe) ----------
   131→  // métriques de conteneur suivies par ResizeObserver (accessibles au rendu)
   132→  const [mstate, setMstate] = useState<{ scale: number; offX: number; offY: number; w: number; h: number }>({
   133→    scale: 0.5, offX: 0, offY: 0, w: 520, h: 1000,
   134→  })
   135→
   136→  useEffect(() => {
   137→    const el = svgRef.current
   138→    if (!el) return
   139→    const update = () => {
   140→      const r = el.getBoundingClientRect()
   141→      const scale = Math.min(r.width / MAP_W, r.height / MAP_H) || 0.5
   142→      setMstate({
   143→        scale,
   144→        offX: (r.width - MAP_W * scale) / 2,
   145→        offY: (r.height - MAP_H * scale) / 2,
   146→        w: r.width,
   147→        h: r.height,
   148→      })
   149→    }
   150→    const raf = requestAnimationFrame(update)
   151→    const ro = new ResizeObserver(update)
   152→    ro.observe(el)
   153→    return () => { cancelAnimationFrame(raf); ro.disconnect() }
   154→  }, [])
   155→
   156→  const metrics = useCallback(() => {
   157→    const el = svgRef.current
   158→    if (!el) return null
   159→    const r = el.getBoundingClientRect()
   160→    const scale = Math.min(r.width / MAP_W, r.height / MAP_H)
   161→    const offX = (r.width - MAP_W * scale) / 2
   162→    const offY = (r.height - MAP_H * scale) / 2
   163→    return { scale, offX, offY, rect: r }
   164→  }, [])
   165→
   166→  const svgPointFromEvent = useCallback((e: { clientX: number; clientY: number }): [number, number] | null => {
   167→    const m = metrics()
   168→    if (!m) return null
   169→    const sx = (e.clientX - m.rect.left - m.offX) / m.scale
   170→    const sy = (e.clientY - m.rect.top - m.offY) / m.scale
   171→    return [sx, sy]
   172→  }, [metrics])
   173→
   174→  const toScreen = useCallback((p: { x: number; y: number }) => ({
   175→    left: mstate.offX + (p.x * view.k + view.x) * mstate.scale,
   176→    top: mstate.offY + (p.y * view.k + view.y) * mstate.scale,
   177→  }), [mstate, view])
   178→
   179→  // ---------- Zoom / pan ----------
   180→  useEffect(() => {
   181→    const el = svgRef.current
   182→    if (!el) return
   183→    const onWheel = (e: WheelEvent) => {
   184→      e.preventDefault()
   185→      const pt = svgPointFromEvent(e)
   186→      if (!pt) return
   187→      setView(v => {
   188→        const factor = e.deltaY < 0 ? 1.18 : 1 / 1.18
   189→        const k = clamp(v.k * factor, 1, 14)
   190→        const mx = (pt[0] - v.x) / v.k
   191→        const my = (pt[1] - v.y) / v.k
   192→        return { k, x: pt[0] - mx * k, y: pt[1] - my * k }
   193→      })
   194→    }
   195→    el.addEventListener("wheel", onWheel, { passive: false })
   196→    return () => el.removeEventListener("wheel", onWheel)
   197→  }, [svgPointFromEvent])
   198→
   199→  const zoomBy = (factor: number) => setView(v => {
   200→    const k = clamp(v.k * factor, 1, 14)
   201→    const cx = MAP_W / 2, cy = MAP_H / 2
   202→    const mx = (cx - v.x) / v.k, my = (cy - v.y) / v.k
   203→    return { k, x: cx - mx * k, y: cy - my * k }
   204→  })
   205→
   206→  const resetView = () => setView({ k: 1, x: 0, y: 0 })
   207→
   208→  const fitBounds = useCallback((sts: StationDTO[]) => {
   209→    if (!sts.length) { resetView(); return }
   210→    const pts = sts.map(s => project(s.lng, s.lat))
   211→    const minX = Math.min(...pts.map(p => p.x)), maxX = Math.max(...pts.map(p => p.x))
   212→    const minY = Math.min(...pts.map(p => p.y)), maxY = Math.max(...pts.map(p => p.y))
   213→    const bw = Math.max(maxX - minX, 10), bh = Math.max(maxY - minY, 10)
   214→    const k = clamp(Math.min((MAP_W - 80) / bw, (MAP_H - 80) / bh), 1, 6)
   215→    const cx = (minX + maxX) / 2, cy = (minY + maxY) / 2
   216→    setView({ k, x: MAP_W / 2 - cx * k, y: MAP_H / 2 - cy * k })
   217→  }, [])
   218→
   219→  useEffect(() => {
   220→    if (!(autoFitOnLoad && stations.length)) return
   221→    const raf = requestAnimationFrame(() => fitBounds(stations))
   222→    return () => cancelAnimationFrame(raf)
   223→  }, [autoFitOnLoad, stations, fitBounds])
   224→
   225→  useEffect(() => {
   226→    if (!fitTo) return
   227→    const raf = requestAnimationFrame(() => fitBounds(fitTo))
   228→    return () => cancelAnimationFrame(raf)
   229→  }, [fitTo, fitBounds])
   230→
   231→  const selectStation = useCallback((s: StationDTO | null) => {
   232→    setSelCode(s ? s.code : null)
   233→    setShowHistory(false)
   234→    onSelect?.(s)
   235→  }, [onSelect])
   236→
   237→  // ---------- Focus « voir sur la carte » ----------
   238→  useEffect(() => {
   239→    if (!focusCode) return
   240→    const raf = requestAnimationFrame(() => {
   241→      const st = stations.find(s => s.code === focusCode)
   242→      if (st) {
   243→        selectStation(st)
   244→        const p = project(st.lng, st.lat)
   245→        setView({ k: 6, x: MAP_W / 2 - p.x * 6, y: MAP_H / 2 - p.y * 6 })
   246→      }
   247→    })
   248→    return () => cancelAnimationFrame(raf)
   249→  }, [focusCode, stations, selectStation])
   250→
   251→  // ---------- Détail fiche (dérivé : actif seulement si cohérent avec la sélection) ----------
   252→  const activeDetail = selCode && detailCode === selCode ? detail : null
   253→  useEffect(() => {
   254→    if (!selCode) return
   255→    const st = stations.find(s => s.code === selCode)
   256→    if (!st) return
   257→    if (detailCode === selCode) return
   258→    const ctrl = new AbortController()
   259→    fetch(`/api/stations/${st.id}`, { signal: ctrl.signal })
   260→      .then(r => (r.ok ? r.json() : null))
   261→      .then((d: StationDetail | null) => { if (d) { setDetail(d); setDetailCode(selCode) } })
   262→      .catch(() => {})
   263→    return () => ctrl.abort()
   264→  }, [selCode, stations, detailCode])
   265→
   266→  // ---------- Drag pan ----------
   267→  const dragRef = useRef<{ sx: number; sy: number; vx: number; vy: number } | null>(null)
   268→  const onPointerDown = (e: React.PointerEvent) => {
   269→    if (e.button !== 0) return
   270→    movedRef.current = 0
   271→    dragRef.current = { sx: e.clientX, sy: e.clientY, vx: view.x, vy: view.y }
   272→    ;(e.target as Element).setPointerCapture?.(e.pointerId)
   273→  }
   274→  const onPointerMove = (e: React.PointerEvent) => {
   275→    const d = dragRef.current
   276→    if (!d) return
   277→    const m = metrics()
   278→    if (!m) return
   279→    const dx = (e.clientX - d.sx) / m.scale
   280→    const dy = (e.clientY - d.sy) / m.scale
   281→    movedRef.current = Math.max(movedRef.current, Math.abs(dx) + Math.abs(dy))
   282→    if (movedRef.current > 3) {
   283→      setView(v => ({ ...v, x: d.vx + dx, y: d.vy + dy }))
   284→    }
   285→  }
   286→  const onPointerUp = () => { dragRef.current = null }
   287→
   288→  // ---------- Clic fond de carte ----------
   289→  const onBackgroundClick = (e: React.MouseEvent) => {
   290→    if (movedRef.current > 3) return
   291→    const pt = svgPointFromEvent(e)
   292→    if (!pt) return
   293→    if (tool === "measure") { setMeasurePts(p => [...p, pt]); return }
   294→    if (tool === "radius") { setRadiusCenter(pt); return }
   295→    if (tool === "polygon" && !polygonClosed) { setPolygonPts(p => [...p, pt]); return }
   296→    selectStation(null)
   297→  }
   298→
   299→  const onDoubleClick = (e: React.MouseEvent) => {
   300→    if (tool === "polygon" && polygonPts.length >= 3) { setPolygonClosed(true); return }
   301→    const pt = svgPointFromEvent(e)
   302→    if (!pt) return
   303→    setView(v => {
   304→      const k = clamp(v.k * 1.7, 1, 14)
   305→      const mx = (pt[0] - v.x) / v.k, my = (pt[1] - v.y) / v.k
   306→      return { k, x: pt[0] - mx * k, y: pt[1] - my * k }
   307→    })
   308→  }
   309→
   310→  // ---------- Sélections par outil ----------
   311→  const insideRadius = useMemo(() => {
   312→    if (!radiusCenter || tool !== "radius") return null
   313→    const c = unproject(radiusCenter[0], radiusCenter[1])
   314→    const set = new Set<string>()
   315→    for (const s of stations) {
   316→      if (haversineKm(c.lat, c.lng, s.lat, s.lng) <= radiusKm) set.add(s.code)
   317→    }
   318→    return set
   319→  }, [radiusCenter, radiusKm, stations, tool])
   320→
   321→  const insidePolygon = useMemo(() => {
   322→    if (tool !== "polygon" || polygonPts.length < 3 || !polygonClosed) return null
   323→    const poly = polygonPts.map(p => { const u = unproject(p[0], p[1]); return [u.lng, u.lat] as [number, number] })
   324→    const set = new Set<string>()
   325→    for (const s of stations) if (pointInPolygon(s.lng, s.lat, poly)) set.add(s.code)
   326→    return set
   327→  }, [polygonPts, polygonClosed, stations, tool])
   328→
   329→  const exportSelection = (format: "csv" | "geojson") => {
   330→    const set = tool === "polygon" ? insidePolygon : insideRadius
   331→    if (!set) return
   332→    const selected = stations.filter(s => set.has(s.code))
   333→    if (!selected.length) { toast.info("Aucune station dans la sélection."); return }
   334→    let blob: Blob
   335→    let ext: string
   336→    if (format === "csv") {
   337→      const header = "code;nom;enseigne;gouvernorat;delegation;statut;trafic;lat;lng;services\n"
   338→      const rows = selected.map(s =>
   339→        [s.code, s.name, s.brandName, s.gouvernorat, s.delegation, s.status, s.traffic, s.lat, s.lng, s.services.join("|")].join(";")
   340→      ).join("\n")
   341→      blob = new Blob(["\ufeff" + header + rows], { type: "text/csv;charset=utf-8" })
   342→      ext = "csv"
   343→    } else {
   344→      const fc = {
   345→        type: "FeatureCollection",
   346→        features: selected.map(s => ({
   347→          type: "Feature",
   348→          geometry: { type: "Point", coordinates: [s.lng, s.lat] },
   349→          properties: { code: s.code, nom: s.name, enseigne: s.brandName, gouvernorat: s.gouvernorat, statut: s.status, trafic: s.traffic, services: s.services },
   350→        })),
   351→      }
   352→      blob = new Blob([JSON.stringify(fc, null, 2)], { type: "application/geo+json" })
   353→      ext = "geojson"
   354→    }
   355→    const a = document.createElement("a")
   356→    a.href = URL.createObjectURL(blob)
   357→    a.download = `carbuscope-selection.${ext}`
   358→    a.click()
   359→    URL.revokeObjectURL(a.href)
   360→    toast.success(`${selected.length} stations exportées (${ext.toUpperCase()})`)
   361→  }
   362→
   363→  const clearTools = () => {
   364→    setTool(null); setMeasurePts([]); setRadiusCenter(null); setPolygonPts([]); setPolygonClosed(false)
   365→  }
   366→
   367→  const measureTotal = useMemo(() => {
   368→    if (measurePts.length < 2) return 0
   369→    let t = 0
   370→    for (let i = 1; i < measurePts.length; i++) {
   371→      const a = unproject(measurePts[i - 1][0], measurePts[i - 1][1])
   372→      const b = unproject(measurePts[i][0], measurePts[i][1])
   373→      t += haversineKm(a.lat, a.lng, b.lat, b.lng)
   374→    }
   375→    return t
   376→  }, [measurePts])
   377→
   378→  // ---------- Rendu stations (mémoïsé) ----------
   379→  const stationNodes = useMemo(() => {
   380→    return stations.map(s => {
   381→      const p = project(s.lng, s.lat)
   382→      const isOwn = ownBrandSlug && s.brandSlug === ownBrandSlug
   383→      const isSel = s.code === selCode
   384→      const isHover = s.code === hoverCode
   385→      const inRad = insideRadius?.has(s.code)
   386→      const inPoly = insidePolygon?.has(s.code)
   387→      const highlighted = isOwn || isSel || isHover || inRad || inPoly
   388→      const r = isOwn ? 3.6 : 2.7
   389→      if (s.status === "CLOSED") {
   390→        return (
   391→          <g key={s.id} onClick={(e) => { e.stopPropagation(); if (movedRef.current <= 3) selectStation(s) }}
   392→             onMouseEnter={() => setHoverCode(s.code)} onMouseLeave={() => setHoverCode(null)} className="cursor-pointer">
   393→            <circle cx={p.x} cy={p.y} r={3.4} fill="#9ca3af" stroke="#6b7280" strokeWidth={1} />
   394→            <line x1={p.x - 2} y1={p.y - 2} x2={p.x + 2} y2={p.y + 2} stroke="#f8fafc" strokeWidth={0.9} />
   395→            <line x1={p.x - 2} y1={p.y + 2} x2={p.x + 2} y2={p.y - 2} stroke="#f8fafc" strokeWidth={0.9} />
   396→          </g>
   397→        )
   398→      }
   399→      return (
   400→        <g key={s.id} onClick={(e) => { e.stopPropagation(); if (movedRef.current <= 3) selectStation(s) }}
   401→           onMouseEnter={() => setHoverCode(s.code)} onMouseLeave={() => setHoverCode(null)} className="cursor-pointer">
   402→          {isSel && <circle cx={p.x} cy={p.y} r={8} fill={s.brandColor} opacity={0.25} className="animate-ping-slow" />}
   403→          {s.status === "PROJECT" ? (
   404→            <circle cx={p.x} cy={p.y} r={r} fill="none" stroke={s.brandColor} strokeWidth={1.6} strokeDasharray="2 1.6" />
   405→          ) : (
   406→            <circle cx={p.x} cy={p.y} r={r} fill={s.brandColor} opacity={highlighted ? 1 : 0.72}
   407→              stroke={s.status === "RENOVATION" ? "#d97706" : "white"} strokeWidth={s.status === "RENOVATION" ? 1.6 : isSel ? 1.8 : 0.7} />
   408→          )}
   409→        </g>
   410→      )
   411→    })
   412→  }, [stations, ownBrandSlug, selCode, hoverCode, insideRadius, insidePolygon, selectStation])
   413→
   414→  // ---------- Popup position ----------
   415→  const selectedStation = stations.find(s => s.code === selCode) ?? null
   416→  const popupPos = useMemo(() => {
   417→    if (!selectedStation) return null
   418→    const p = project(selectedStation.lng, selectedStation.lat)
   419→    return toScreen(p)
   420→  }, [selectedStation, toScreen])
   421→
   422→  const roads = staticData?.roads ?? []
   423→  const projects = staticData?.projects ?? []
   424→  const governorats = staticData?.governorats ?? []
   425→  const govCountFallback = useMemo(() => {
   426→    const m: Record<string, number> = {}
   427→    for (const s of stations) m[s.gouvernorat] = (m[s.gouvernorat] ?? 0) + 1
   428→    return m
   429→  }, [stations])
   430→  const govCountMap = govStationCounts ?? govCountFallback
   431→
   432→  const transform = `translate(${view.x} ${view.y}) scale(${view.k})`
   433→  const invK = 1 / view.k
   434→
   435→  return (
   436→    <div ref={wrapRef} className={`relative overflow-hidden rounded-xl border bg-[radial-gradient(ellipse_at_center,var(--secondary)_0%,var(--background)_75%)] ${className}`}>
   437→      <svg ref={svgRef} viewBox={`0 0 ${MAP_W} ${MAP_H}`} preserveAspectRatio="xMidYMid meet"
   438→        className="h-full w-full touch-none select-none cursor-grab active:cursor-grabbing"
   439→        onPointerDown={onPointerDown} onPointerMove={onPointerMove} onPointerUp={onPointerUp}
   440→        onClick={onBackgroundClick} onDoubleClick={onDoubleClick}
   441→        role="img" aria-label="Carte des stations-service de Tunisie">
   442→

   741→function brandLegend(stations: StationDTO[], ownSlug?: string | null): [string, string, number][] {
   742→  const m = new Map<string, [string, number]>()
   743→  for (const s of stations) {
   744→    const cur = m.get(s.brandName)
   745→    m.set(s.brandName, [s.brandColor, (cur?.[1] ?? 0) + 1])
   746→  }
   747→  return [...m.entries()]
   748→    .sort((a, b) => b[1][1] - a[1][1])
   749→    .slice(0, 8)
   750→    .map(([name, [color, count]]) => [ownSlug && stations.find(s => s.brandName === name)?.brandSlug === ownSlug ? `${name} (moi)` : name, color, count])
   751→}
