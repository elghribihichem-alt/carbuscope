    40→    40→function csvEscape(v: string | number): string {
    41→    41→  const s = String(v ?? "")
    42→    42→  return /[",;\n\r]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s
    43→    43→}
    44→    44→
    45→    45→function csvRow(cells: (string | number)[]): string {
    46→    46→  return cells.map(csvEscape).join(";")
    47→    47→}

    49→    49→function downloadCsv(filename: string, content: string): void {
    50→    50→  const blob = new Blob([`\ufeff${content}`], { type: "text/csv;charset=utf-8;" })
    51→    51→  const url = URL.createObjectURL(blob)
    52→    52→  const a = document.createElement("a")
    53→    53→  a.href = url
    54→    54→  a.download = filename
    55→    55→  document.body.appendChild(a)
    56→    56→  a.click()
    57→    57→  document.body.removeChild(a)
    58→    58→  URL.revokeObjectURL(url)
    59→    59→}

    95→    95→function isoToLocalDateInput(iso: string): string {
    96→    96→  const d = new Date(iso)
    97→    97→  const p = (n: number) => String(n).padStart(2, "0")

    99→    99→}

   102→   102→function parseDecimal(raw: string): number | null {
   103→   103→  const t = raw.trim().replace(",", ".")
   104→   104→  if (t === "") return null
   105→   105→  const v = Number(t)
   106→   106→  return Number.isFinite(v) ? v : null
   107→   107→}

   572→   572→  takenSlugs: Set<string>

   581→   581→
   582→   582→  // Chargement à l'ouverture (pattern reload : états gérés dans la fonction).
   583→   583→  const loadOptions = useCallback(async (open: boolean) => {
   584→   584→    if (!open) return
   585→   585→    setLoading(true)
