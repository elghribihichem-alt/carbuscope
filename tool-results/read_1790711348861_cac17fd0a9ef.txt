    58→function csvEscape(v: string | number): string {
    59→  const s = String(v ?? "")
    60→  return /[",;\n\r]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s
    61→}
    62→
    63→function csvRow(cells: (string | number)[]): string {
    64→  return cells.map(csvEscape).join(";")
    65→}

    67→function downloadCsv(filename: string, content: string): void {
    68→  const blob = new Blob([`\ufeff${content}`], { type: "text/csv;charset=utf-8;" })
    69→  const url = URL.createObjectURL(blob)
    70→  const a = document.createElement("a")
    71→  a.href = url
    72→  a.download = filename
    73→  document.body.appendChild(a)
    74→  a.click()
    75→  document.body.removeChild(a)
    76→  URL.revokeObjectURL(url)
    77→}

    86→function exportContractsCsv(rows: Row[], statusFilter: string, planFilter: string): string {
    87→  const lines: string[] = [
    88→    csvRow([
    89→      "Enseigne", "Plan", "Statut", "Tarif mensuel (DT)", "Sièges actifs",
    90→      "Quota sièges", "Stations", "Contact", "Renouvellement", "Impayé (DT)",
    91→    ]),
    92→  ]
    93→  for (const c of rows) {
    94→    lines.push(csvRow([
    95→      c.brandName,
    96→      PLAN_LABELS[c.plan] ?? c.plan,
    97→      CONTRACT_STATUS_META[c.status]?.label ?? c.status,
    98→      fmtCsvAmount(c.monthlyFee),
    99→      c.activeUsers,
   100→      c.seats,
   101→      `${c.stationCount} / ${c.stationQuota}`,
   102→      `${c.contactName} (${c.contactEmail})`,
   103→      new Date(c.renewsAt).toLocaleDateString("fr-FR"),
   104→      fmtCsvAmount(c.unpaidTotal),
   105→    ]))
   106→  }
   107→  // Ligne de totaux (T52-R2) — sommes des colonnes numériques du périmètre exporté.
   108→  lines.push(csvRow([
   109→    `Total — ${rows.length} contrat(s)`, "", "", "",
   110→    rows.reduce((s, c) => s + c.activeUsers, 0),
   111→    rows.reduce((s, c) => s + c.seats, 0),
   112→    "", "", "",
   113→    fmtCsvAmount(rows.reduce((s, c) => s + c.unpaidTotal, 0)),
   114→  ]))
   115→  const scope = `${STATUS_SLUGS[statusFilter] ?? "toutes"}${planFilter !== "TOUTES" ? `-${planFilter.toLowerCase()}` : ""}`
   116→  const filename = `carbuscope-contrats-${scope}-${todayStamp()}.csv`
   117→  downloadCsv(filename, lines.join("\r\n"))
   118→  return filename
   119→}

   123→/** ISO → valeur d'un input type=date (YYYY-MM-DD), fuseau local. */
   124→function isoToLocalDateInput(iso: string): string {
   125→  const d = new Date(iso)
   126→  const p = (n: number) => String(n).padStart(2, "0")
   127→  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())}`
   128→}

   130→/** Décimal français accepté : « 890 » ou « 890,5 ». null si vide/invalide. */
   131→function parseDecimal(raw: string): number | null {
   132→  const t = raw.trim().replace(",", ".")
   133→  if (t === "") return null
   134→  const v = Number(t)
   135→  return Number.isFinite(v) ? v : null
   136→}

   138→function ContractEditForm({
   139→  contract, actor, onClose, onSaved,
   140→}: {
   141→  contract: Row
   142→  actor: string
   143→  onClose: () => void
   144→  onSaved: (contract: Row, plan: string, fee: number) => Promise<void>
   145→}) {
   146→  const [plan, setPlan] = useState<string>(contract.plan)
   147→  const [fee, setFee] = useState(contract.monthlyFee > 0 ? String(contract.monthlyFee) : "")
   148→  const [seats, setSeats] = useState(String(contract.seats))
   149→  const [quota, setQuota] = useState(String(contract.stationQuota))
   150→  const [renews, setRenews] = useState(isoToLocalDateInput(contract.renewsAt))
   151→  const [saving, setSaving] = useState(false)
   152→  const [error, setError] = useState<string | null>(null)
   153→
   154→  async function submit(e: React.FormEvent<HTMLFormElement>) {
   155→    e.preventDefault()
   156→    if (saving) return
   157→    setError(null)
   158→
   159→    const feeNum = parseDecimal(fee)
   160→    if (feeNum === null || feeNum < 1 || feeNum > 100000) {
   161→      setError("Tarif mensuel invalide — saisissez un montant entre 1 et 100 000 DT.")
   162→      return
   163→    }
   164→    if (!/^\d+$/.test(seats.trim()) || Number(seats.trim()) < 1 || Number(seats.trim()) > 500) {
   165→      setError("Nombre de sièges invalide — entier entre 1 et 500.")
   166→      return
   167→    }
   168→    if (!/^\d+$/.test(quota.trim()) || Number(quota.trim()) < 1 || Number(quota.trim()) > 5000) {
   169→      setError("Quota de stations invalide — entier entre 1 et 5 000.")
   170→      return
   171→    }
   172→    if (!renews || Number.isNaN(Date.parse(renews))) {
   173→      setError("Date de renouvellement invalide.")
   174→      return
   175→    }
   176→    const seatsNum = Number(seats.trim())
   177→    const quotaNum = Number(quota.trim())
   178→
   179→    // N'envoyer que les champs réellement modifiés (pattern station-form-dialog).
   180→    const data: Record<string, unknown> = {}
   181→    if (plan !== contract.plan) data.plan = plan
   182→    if (feeNum !== contract.monthlyFee) data.monthlyFee = feeNum
   183→    if (seatsNum !== contract.seats) data.seats = seatsNum
   184→    if (quotaNum !== contract.stationQuota) data.stationQuota = quotaNum
   185→    if (renews !== isoToLocalDateInput(contract.renewsAt)) data.renewsAt = renews
   186→
   187→    if (Object.keys(data).length === 0) {
   188→      toast.info("Aucune modification à enregistrer.")
   189→      onClose()
   190→      return
   191→    }
   192→
   193→    setSaving(true)
   194→    try {
   195→      const r = await fetch(`/api/super/contracts/${contract.id}`, {
   196→        method: "PATCH",
   197→        headers: { "Content-Type": "application/json" },
   198→        body: JSON.stringify({ ...data, actor }),
   199→      })
   200→      if (!r.ok) {
   201→        const j = await r.json().catch(() => null)
   202→        throw new Error(j?.error ?? `Erreur ${r.status}`)
   203→      }
   204→      await onSaved(contract, plan, feeNum)
   205→      onClose()
   206→    } catch (e) {
   207→      setError(e instanceof Error ? e.message : "Échec de la modification du contrat.")
   208→    } finally {
   209→      setSaving(false)
   210→    }
   211→  }
   212→
   213→  return (
   214→    // noValidate : la validation française (role="alert") gère tous les cas,
   215→    // y compris les valeurs vides/négatives, sans tooltip natif du navigateur.
   216→    <form onSubmit={submit} noValidate className="space-y-4">
   217→      <div className="grid gap-3 sm:grid-cols-2">
   218→        <div className="space-y-1.5">
   219→          <Label htmlFor="sa-plan">Plan</Label>
   220→          <Select value={plan} onValueChange={setPlan}>
   221→            <SelectTrigger id="sa-plan" className="w-full">
   222→              <SelectValue placeholder="Sélectionner un plan" />
   223→            </SelectTrigger>
   224→            <SelectContent>
   225→              {PLAN_OPTIONS.map(p => (
   226→                <SelectItem key={p} value={p}>{PLAN_LABELS[p] ?? p}</SelectItem>
   227→              ))}
   228→            </SelectContent>
   229→          </Select>
   230→        </div>
   231→        <div className="space-y-1.5">
   232→          <Label htmlFor="sa-fee">Tarif mensuel</Label>
   233→          <div className="relative">
   234→            <Input
   235→              id="sa-fee"
   236→              type="number"
   237→              min={1}
   238→              max={100000}
   239→              step={0.001}
   240→              inputMode="decimal"
   241→              value={fee}
   242→              onChange={e => setFee(e.target.value)}
   243→              placeholder="Ex. 890"
   244→              className="pr-10 tabular-nums"
   245→              aria-describedby="sa-fee-hint"
   246→            />
   247→            <span className="pointer-events-none absolute inset-y-0 right-3 flex items-center text-xs font-medium text-muted-foreground" aria-hidden>DT</span>
   248→          </div>
   249→          <p id="sa-fee-hint" className="text-[11px] text-muted-foreground">DT par mois, jusqu&apos;à 3 décimales.</p>
   250→        </div>
   251→        <div className="space-y-1.5">
   252→          <Label htmlFor="sa-seats">Sièges (utilisateurs inclus)</Label>
   253→          <Input
   254→            id="sa-seats"
   255→            type="number"
   256→            min={1}
   257→            max={500}
   258→            step={1}
   259→            inputMode="numeric"
   260→            value={seats}
   261→            onChange={e => setSeats(e.target.value)}
   262→            className="tabular-nums"
   263→          />
   264→        </div>
   265→        <div className="space-y-1.5">
   266→          <Label htmlFor="sa-quota">Quota stations</Label>
   267→          <Input
   268→            id="sa-quota"
   269→            type="number"
   270→            min={1}
   271→            max={5000}
   272→            step={1}
   273→            inputMode="numeric"
   274→            value={quota}
   275→            onChange={e => setQuota(e.target.value)}
   276→            className="tabular-nums"
   277→          />
   278→        </div>
   279→        <div className="space-y-1.5 sm:col-span-2">
   280→          <Label htmlFor="sa-renews">Renouvellement</Label>
   281→          <Input
   282→            id="sa-renews"
   283→            type="date"
   284→            value={renews}
   285→            onChange={e => setRenews(e.target.value)}
   286→          />
   287→          <p className="text-[11px] text-muted-foreground">
   288→            Prochaine échéance annuelle du contrat — actuellement le {fmtDate(contract.renewsAt)}.
   289→          </p>
   290→        </div>
   291→      </div>
   292→
   293→      {error && (
   294→        <p role="alert" className="rounded-md border border-destructive/40 bg-destructive/10 px-3 py-2 text-xs font-medium text-destructive">
   295→          {error}
   296→        </p>
   297→      )}
   298→
   299→      <DialogFooter className="gap-2">
   300→        <Button type="button" variant="outline" size="sm" disabled={saving} onClick={onClose}>
   301→          Annuler
   302→        </Button>
   303→        <Button type="submit" size="sm" className="gap-1.5" disabled={saving}>
   304→          {saving ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <Save className="h-4 w-4" aria-hidden />}
   305→          Enregistrer les modifications
   306→        </Button>
   307→      </DialogFooter>
   308→    </form>
   309→  )
   310→}

   347→/** Valeur d'un input type=date pour aujourd'hui + n jours, fuseau local. */
   348→function localDateInDays(n: number): string {
   349→  const d = new Date(Date.now() + n * 86_400_000)
   350→  const p = (x: number) => String(x).padStart(2, "0")
   351→  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())}`
   352→}

   354→function ContractCreateForm({

   389→    e.preventDefault()

   392→
   393→    if (!brandSlug) {
   394→      setError("Sélectionnez une enseigne cliente.")
   395→      return
   396→    }
   397→    if (takenSlugs.has(brandSlug)) {
   398→      setError("Un contrat existe déjà pour cette enseigne.")
   399→      return
   400→    }
   401→    if (!plan) {
   402→      setError("Sélectionnez un plan du catalogue.")
   403→      return
   404→    }
   405→    const feeNum = parseDecimal(fee)
   406→    if (feeNum === null || feeNum < 1 || feeNum > 100000) {
   407→      setError("Tarif mensuel invalide — saisissez un montant entre 1 et 100 000 DT.")
   408→      return
   409→    }
   410→    if (!/^\d+$/.test(seats.trim()) || Number(seats.trim()) < 1 || Number(seats.trim()) > 500) {
   411→      setError("Nombre de sièges invalide — entier entre 1 et 500.")
   412→      return
   413→    }
   414→    if (!/^\d+$/.test(quota.trim()) || Number(quota.trim()) < 1 || Number(quota.trim()) > 5000) {
   415→      setError("Quota de stations invalide — entier entre 1 et 5 000.")
   416→      return
   417→    }
   418→    if (!renews || Number.isNaN(Date.parse(renews))) {
   419→      setError("Date de renouvellement invalide.")
   420→      return
   421→    }

   423→      setError("Adresse e-mail de contact invalide.")
   424→      return
   425→    }

   437→        }),

   447→      })

   449→      onClose()
   450→    } catch (e) {
   451→      setError(e instanceof Error ? e.message : "Échec de la création du contrat.")

   453→      setSaving(false)
   454→    }

   469→                return (
   470→                  <SelectItem key={b.slug} value={b.slug} disabled={taken}>
   471→                    {b.name}{taken ? " · contrat existant" : ` · ${b.stationCount} stations`}

   484→            <SelectTrigger id="nc-plan" className="w-full">
   485→              <SelectValue placeholder="Sélectionner un plan" />
   486→            </SelectTrigger>
   487→            <SelectContent>
   488→              {plans.map(p => (
   489→                <SelectItem key={p.code} value={p.code} disabled={!p.active}>
   490→                  {p.label} · {p.monthlyFee.toLocaleString("fr-FR")} DT
   491→                </SelectItem>
   492→              ))}
   493→            </SelectContent>
   494→          </Select>
   495→          <p className="text-[11px] text-muted-foreground">La sélection pré-remplit le tarif, les sièges et le quota.</p>
   496→        </div>
   497→        <div className="space-y-1.5">
   498→          <Label htmlFor="nc-fee">Tarif mensuel</Label>
   499→          <div className="relative">
   500→            <Input
   501→              id="nc-fee"
   502→              type="number"
   503→              min={1}
   504→              max={100000}
   505→              step={0.001}
   506→              inputMode="decimal"
