   168→export function EditableCell({ row, field, kind, ariaName, onCommit }: {
   169→  row: DemographicDelegationDTO
   170→  field: string
   171→  kind: EditKind
   172→  ariaName: string
   173→  onCommit: (row: DemographicDelegationDTO, field: string, v: number | null) => void
   174→}) {
   175→  const value = (row as unknown as Record<string, number | null>)[field] ?? null
   176→  const [editing, setEditing] = useState(false)
   177→  const [draft, setDraft] = useState("")
   178→  const inputRef = useRef<HTMLInputElement>(null)
   179→
   180→  useEffect(() => {
   181→    if (editing) {
   182→      inputRef.current?.focus()
   183→      inputRef.current?.select()
   184→    }
   185→  }, [editing])
   186→
   187→  const display = value === null ? "—" : kind === "pct" ? fmtPct(value) : fmtInt(value)
   188→
   189→  const tryCommit = (fromBlur: boolean) => {
   190→    const t = draft.trim()
   191→    if (t === "") {
   192→      setEditing(false)
   193→      if (value !== null) onCommit(row, field, null)
   194→      return
   195→    }
   196→    const v = parseNumFr(t)
   197→    const invalid = !Number.isFinite(v)
   198→      || (kind === "pct" && (v < 0 || v > 100))
   199→      || (kind === "int" && (!Number.isInteger(v) || v < 0))
   200→    if (invalid) {
   201→      if (fromBlur) { setEditing(false); return }
   202→      toast.error(kind === "pct" ? "Le taux doit être un nombre entre 0 et 100" : "Nombre entier positif attendu")
   203→      return
   204→    }
   205→    setEditing(false)
   206→    if (v !== value) onCommit(row, field, v)
   207→  }
   208→
   209→  if (editing) {
   210→    return (
   211→      <input
   212→        ref={inputRef}
   213→        value={draft}
   214→        onChange={e => setDraft(e.target.value)}
   215→        onKeyDown={e => {
   216→          if (e.key === "Enter") { e.preventDefault(); tryCommit(false) }
   217→          if (e.key === "Escape") { e.preventDefault(); setEditing(false) }
   218→        }}
   219→        onBlur={() => tryCommit(true)}
   220→        inputMode={kind === "int" ? "numeric" : "decimal"}
   221→        className="h-8 w-full min-w-0 rounded-md border border-emerald-400 bg-background px-2 text-right font-mono text-xs tabular-nums shadow-sm outline-none focus-visible:ring-2 focus-visible:ring-emerald-500/40 dark:border-emerald-600"
   222→        aria-label={ariaName}
   223→      />
   224→    )
   225→  }
   226→
   227→  return (
   228→    <button
   229→      type="button"
   230→      onClick={() => { setDraft(value === null ? "" : String(value)); setEditing(true) }}
   231→      className={`inline-flex h-8 w-full items-center justify-end rounded-md px-2 font-mono text-xs tabular-nums transition-colors hover:bg-emerald-100/80 hover:font-semibold dark:hover:bg-emerald-950/70 ${value === null ? "text-muted-foreground/50" : "text-foreground"}`}
   232→      title="Cliquer pour modifier"
   233→      aria-label={`${ariaName} : ${value === null ? "non renseigné" : display} — cliquer pour modifier`}
   234→    >
   235→      {display}
   236→    </button>
   237→  )
   238→}

   242→function ChartPie({ items, unit, decimals, total }: {
   243→  items: { label: string; value: number }[]
   244→  unit: string
   245→  decimals: number
   246→  total: number
   247→}) {
   248→  const palette = useChartPalette()
   249→  const sorted = [...items].sort((a, b) => b.value - a.value)
   250→  const top = sorted.slice(0, 8)
   251→  const rest = sorted.slice(8)
   252→  const slices = rest.length > 0
   253→    ? [...top, { label: `Autres (${rest.length})`, value: rest.reduce((s, x) => s + x.value, 0) }]
   254→    : top
   255→  const R = 56
   256→  const C = 2 * Math.PI * R
   257→  // Calcul pur des segments (part cumulée, sans mutation) : camembert = donut
   258→  // SVG en traits circulaires, départ à 12 h (rotation -90°).
   259→  const segments = slices.reduce<{ label: string; value: number; frac: number; start: number }[]>((list, s) => {
   260→    const prev = list.length > 0 ? list[list.length - 1] : null
   261→    const start = prev ? prev.start + prev.frac : 0
   262→    const frac = total > 0 ? s.value / total : 0
   263→    return [...list, { label: s.label, value: s.value, frac, start }]
   264→  }, [])
   265→
   266→  return (
   267→    <div className="flex flex-col items-center gap-6 lg:flex-row lg:gap-10">
   268→      <svg viewBox="0 0 160 160" className="h-44 w-44 shrink-0" role="img"
   269→        aria-label={`Camembert : ${slices.length} parts, total ${fmtChart(total, decimals)} ${unit}`}>
   270→        <g transform="rotate(-90 80 80)">
   271→          {segments.map((s, i) => {
   272→            const len = s.frac * C
   273→            if (len <= 0) return null
   274→            return (
   275→              <motion.circle
   276→                key={s.label}
   277→                cx={80} cy={80} r={R}
   278→                fill="none"
   279→                stroke={palette.color(i + 1)}
   280→                strokeWidth={26}
   281→                strokeDasharray={`${len} ${C - len}`}
   282→                strokeDashoffset={-s.start * C}
   283→                initial={{ opacity: 0 }}
   284→                animate={{ opacity: 1 }}
   285→                transition={{ duration: 0.35, delay: i * 0.05 }}
   286→              />
   287→            )
   288→          })}
   289→        </g>
   290→        <text x={80} y={74} textAnchor="middle" className="fill-foreground text-[13px] font-bold">
   291→          {fmtChart(total, decimals)}
   292→        </text>
   293→        <text x={80} y={90} textAnchor="middle" className="fill-muted-foreground text-[9px]">
   294→          {unit}
   295→        </text>
   296→      </svg>
   297→      <ul className="grid w-full max-w-lg grid-cols-1 gap-x-8 gap-y-1.5 sm:grid-cols-2">
   298→        {slices.map((s, i) => (
   299→          <li key={s.label} className="flex items-center gap-2 text-xs">
   300→            <span className="h-2.5 w-2.5 shrink-0 rounded-full" style={{ backgroundColor: palette.color(i + 1) }} aria-hidden />
   301→            <span className="min-w-0 flex-1 truncate" title={s.label}>{s.label}</span>
   302→            <span className="font-mono tabular-nums text-muted-foreground">{fmtChart(s.value, decimals)}</span>
   303→            <span className="w-14 text-right font-mono tabular-nums text-muted-foreground/70">
   304→              {total > 0 ? `${((s.value / total) * 100).toFixed(1).replace(".", ",")} %` : "—"}
   305→            </span>
   306→          </li>
   307→        ))}
   308→      </ul>
   309→    </div>
   310→  )
   311→}

   315→function ChartBars({ items, unit, decimals }: {
   316→  items: { label: string; value: number }[]
   317→  unit: string
   318→  decimals: number
   319→}) {
   320→  const sorted = [...items].sort((a, b) => b.value - a.value)
   321→  const shown = sorted.slice(0, 12)
   322→  const maxV = shown[0]?.value ?? 0
   323→
   324→  return (
   325→    <div className="space-y-2">
   326→      {shown.map((x, i) => (
   327→        <div key={x.label} className="flex items-center gap-3">
   328→          <span className="w-36 shrink-0 truncate text-right text-xs" title={x.label}>{x.label}</span>
   329→          <div className="relative h-5 min-w-0 flex-1 overflow-hidden rounded-md bg-muted">
   330→            <motion.div
   331→              className="h-full rounded-md bg-emerald-600"
   332→              style={{ opacity: 1 - Math.min(i, 8) * 0.08 }}
   333→              initial={{ width: 0 }}
   334→              animate={{ width: `${maxV > 0 ? Math.max((x.value / maxV) * 100, 2) : 0}%` }}
   335→              transition={{ duration: 0.45, delay: i * 0.04, ease: "easeOut" }}
   336→            />
   337→            <span className="absolute inset-y-0 right-2 flex items-center font-mono text-[10px] tabular-nums text-foreground/80">
   338→              {fmtChart(x.value, decimals)} {unit}
   339→            </span>
   340→          </div>
   341→        </div>
   342→      ))}
   343→      {sorted.length > shown.length && (
   344→        <p className="pl-[10.5rem] text-[11px] text-muted-foreground">… et {sorted.length - shown.length} autres</p>
   345→      )}
   346→    </div>
   347→  )
   348→}

   944→/** Résumé FR du patch pour le toast : « population : 52 400 · taux urbain : 78 % ». */
   945→function describePatch(patch: Record<string, number | null>): string {
   946→  return Object.entries(patch)
   947→    .map(([k, v]) => {
   948→      const label = TOAST_LABELS[k] ?? k
   949→      return v === null ? `${label} effacé` : `${label} : ${k.endsWith("Pct") ? fmtPct(v) : fmtInt(v)}`
   950→    })
   951→    .join(" · ")
   952→}
   953→
   954→/** Résumé FR du patch gouvernorat : « population : 1 056 300 · urbain : 99 % ». */
   955→function describeGouvPatch(patch: Record<string, number | null>): string {
   956→  return Object.entries(patch)
   957→    .map(([k, v]) => {
   958→      const label = GOV_TOAST_LABELS[k] ?? k
   959→      if (v === null) return `${label} effacé`
   960→      if (k === "areaKm2") return `${label} : ${fmtInt(v)} km²`
   961→      if (k === "avgFleetAge") return `${label} : ${v.toLocaleString("fr-FR", { maximumFractionDigits: 1 })} ans`
   962→      if (k.endsWith("Pct") || k === "motorizationRate") return `${label} : ${fmtPct(v)}`
   963→      return `${label} : ${fmtInt(v)}`
   964→    })
   965→    .join(" · ")
   966→}
