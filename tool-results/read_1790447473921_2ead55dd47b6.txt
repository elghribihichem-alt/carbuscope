    45→/** "2026-09" → "sept. 26" (axe X condensé). */
    46→function moisCourt(mois: string): string {
    47→  const [y, m] = mois.split("-").map(Number)
    48→  const s = new Intl.DateTimeFormat("fr-FR", { month: "short" }).format(new Date(y, m - 1, 1))
    49→  return `${s.replace(".", "")}. ${String(y).slice(2)}`
    50→}

   741→/**
   742→ * Ouvre la boîte d'impression du navigateur sur un document HTML autonome
   743→ * monté dans une iframe hors écran (srcdoc). Imprimante → « Enregistrer au
   744→ * format PDF » produit le rapport A4. Aucune collision avec l'impression
   745→ * facture : le CSS print de la page ne s'applique pas dans l'iframe et
   746→ * l'iframe est retirée du DOM après fermeture réelle du dialogue
   747→ * (afterprint), avec garde-fou 60 s — en headless print() est un no-op,
   748→ * l'iframe reste alors montée pour la QA.
   749→ */
   750→function ouvrirRapportImpression(html: string): void {
   751→  const iframe = document.createElement("iframe")
   752→  iframe.title = "Rapport d'activité CarbuScope (impression)"
   753→  iframe.setAttribute("aria-hidden", "true")
   754→  iframe.style.cssText =
   755→    "position:fixed;left:-10000px;top:0;width:210mm;height:297mm;border:0;background:#fff"
   756→  iframe.srcdoc = html
   757→  let nettoye = false
   758→  let garde: number | undefined
   759→  const nettoyer = () => {
   760→    if (nettoye) return
   761→    nettoye = true
   762→    window.clearTimeout(garde)
   763→    iframe.remove()
   764→  }
   765→  iframe.onload = () => {
   766→    try {
   767→      iframe.contentWindow?.focus()
   768→      iframe.contentWindow?.print()
   769→    } catch {
   770→      // Environnement sans dialogue d'impression (headless) : la garde
   771→      // temporelle ci-dessous nettoie — l'iframe reste entre-temps pour la QA.
   772→    }
   773→    try {
   774→      iframe.contentWindow?.addEventListener("afterprint", nettoyer, { once: true })
   775→    } catch {
   776→      // afterprint indisponible : la garde temporelle nettoie l'iframe.
   777→    }
   778→    // Garde-fou 60 s : print() peut être un no-op (headless), on laisse
   779→    // l'iframe montée le temps de la QA puis on la retire du DOM.
   780→    garde = window.setTimeout(nettoyer, 60_000)
   781→  }
   782→  document.body.appendChild(iframe)
   783→}

   799→      const r = await fetch("/api/super/analytics", { signal })
