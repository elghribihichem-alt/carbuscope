   167→   167→export function EditableCell({ row, field, kind, ariaName, onCommit }: {
   168→   168→  row: DemographicDelegationDTO
   169→   169→  field: string
   170→   170→  kind: EditKind
   171→   171→  ariaName: string
   172→   172→  onCommit: (row: DemographicDelegationDTO, field: string, v: number | null) => void
   173→   173→}) {
   174→   174→  const value = (row as unknown as Record<string, number | null>)[field] ?? null
   175→   175→  const [editing, setEditing] = useState(false)
   176→   176→  const [draft, setDraft] = useState("")
   177→   177→  const inputRef = useRef<HTMLInputElement>(null)
   178→   178→
   179→   179→  useEffect(() => {
   180→   180→    if (editing) {
   181→   181→      inputRef.current?.focus()
   182→   182→      inputRef.current?.select()
   183→   183→    }
   184→   184→  }, [editing])
   185→   185→
   186→   186→  const display = value === null ? "—" : kind === "pct" ? fmtPct(value) : fmtInt(value)
   187→   187→
   188→   188→  const tryCommit = (fromBlur: boolean) => {
   189→   189→    const t = draft.trim()
   190→   190→    if (t === "") {
   191→   191→      setEditing(false)
   192→   192→      if (value !== null) onCommit(row, field, null)
   193→   193→      return
   194→   194→    }
   195→   195→    const v = parseNumFr(t)
   196→   196→    const invalid = !Number.isFinite(v)
   197→   197→      || (kind === "pct" && (v < 0 || v > 100))
   198→   198→      || (kind === "int" && (!Number.isInteger(v) || v < 0))
   199→   199→    if (invalid) {
   200→   200→      if (fromBlur) { setEditing(false); return }
   201→   201→      toast.error(kind === "pct" ? "Le taux doit être un nombre entre 0 et 100" : "Nombre entier positif attendu")
   202→   202→      return
   203→   203→    }
   204→   204→    setEditing(false)
   205→   205→    if (v !== value) onCommit(row, field, v)
   206→   206→  }
   207→   207→
   208→   208→  if (editing) {
   209→   209→    return (
   210→   210→      <input
   211→   211→        ref={inputRef}
   212→   212→        value={draft}
   213→   213→        onChange={e => setDraft(e.target.value)}
   214→   214→        onKeyDown={e => {
   215→   215→          if (e.key === "Enter") { e.preventDefault(); tryCommit(false) }
   216→   216→          if (e.key === "Escape") { e.preventDefault(); setEditing(false) }
   217→   217→        }}
   218→   218→        onBlur={() => tryCommit(true)}
   219→   219→        inputMode={kind === "int" ? "numeric" : "decimal"}
   220→   220→        className="h-8 w-full min-w-0 rounded-md border border-emerald-400 bg-background px-2 text-right font-mono text-xs tabular-nums shadow-sm outline-none focus-visible:ring-2 focus-visible:ring-emerald-500/40 dark:border-emerald-600"
   221→   221→        aria-label={ariaName}
   222→   222→      />
   223→   223→    )
   224→   224→  }
   225→   225→
   226→   226→  return (
   227→   227→    <button
   228→   228→      type="button"
   229→   229→      onClick={() => { setDraft(value === null ? "" : String(value)); setEditing(true) }}
   230→   230→      className={`inline-flex h-8 w-full items-center justify-end rounded-md px-2 font-mono text-xs tabular-nums transition-colors hover:bg-emerald-100/80 hover:font-semibold dark:hover:bg-emerald-950/70 ${value === null ? "text-muted-foreground/50" : "text-foreground"}`}
   231→   231→      title="Cliquer pour modifier"
   232→   232→      aria-label={`${ariaName} : ${value === null ? "non renseigné" : display} — cliquer pour modifier`}
   233→   233→    >
   234→   234→      {display}
   235→   235→    </button>
   236→   236→  )
   237→   237→}

   241→   241→function ChartPie({ items, unit, decimals, total }: {
   242→   242→  items: { label: string; value: number }[]
   243→   243→  unit: string
   244→   244→  decimals: number
   245→   245→  total: number
   246→   246→}) {
   247→   247→  const sorted = [...items].sort((a, b) => b.value - a.value)
   248→   248→  const top = sorted.slice(0, 8)
   249→   249→  const rest = sorted.slice(8)
   250→   250→  const slices = rest.length > 0
   251→   251→    ? [...top, { label: `Autres (${rest.length})`, value: rest.reduce((s, x) => s + x.value, 0) }]
   252→   252→    : top
   253→   253→  const R = 56
   254→   254→  const C = 2 * Math.PI * R
   255→   255→  // Calcul pur des segments (part cumulée, sans mutation) : camembert = donut
   256→   256→  // SVG en traits circulaires, départ à 12 h (rotation -90°).
   257→   257→  const segments = slices.reduce<{ label: string; value: number; frac: number; start: number }[]>((list, s) => {
   258→   258→    const prev = list.length > 0 ? list[list.length - 1] : null
   259→   259→    const start = prev ? prev.start + prev.frac : 0
   260→   260→    const frac = total > 0 ? s.value / total : 0
   261→   261→    return [...list, { label: s.label, value: s.value, frac, start }]
   262→   262→  }, [])
   263→   263→
   264→   264→  return (
   265→   265→    <div className="flex flex-col items-center gap-6 lg:flex-row lg:gap-10">
   266→   266→      <svg viewBox="0 0 160 160" className="h-44 w-44 shrink-0" role="img"
   267→   267→        aria-label={`Camembert : ${slices.length} parts, total ${fmtChart(total, decimals)} ${unit}`}>
   268→   268→        <g transform="rotate(-90 80 80)">
   269→   269→          {segments.map((s, i) => {
   270→   270→            const len = s.frac * C
   271→   271→            if (len <= 0) return null
   272→   272→            return (
   273→   273→              <motion.circle
   274→   274→                key={s.label}
   275→   275→                cx={80} cy={80} r={R}
   276→   276→                fill="none"
   277→   277→                stroke={CHART_COLORS[i % CHART_COLORS.length]}
   278→   278→                strokeWidth={26}
   279→   279→                strokeDasharray={`${len} ${C - len}`}
   280→   280→                strokeDashoffset={-s.start * C}
   281→   281→                initial={{ opacity: 0 }}
   282→   282→                animate={{ opacity: 1 }}
   283→   283→                transition={{ duration: 0.35, delay: i * 0.05 }}
   284→   284→              />
   285→   285→            )
   286→   286→          })}
   287→   287→        </g>
   288→   288→        <text x={80} y={74} textAnchor="middle" className="fill-foreground text-[13px] font-bold">
   289→   289→          {fmtChart(total, decimals)}
   290→   290→        </text>
   291→   291→        <text x={80} y={90} textAnchor="middle" className="fill-muted-foreground text-[9px]">
   292→   292→          {unit}
   293→   293→        </text>
   294→   294→      </svg>
   295→   295→      <ul className="grid w-full max-w-lg grid-cols-1 gap-x-8 gap-y-1.5 sm:grid-cols-2">
   296→   296→        {slices.map((s, i) => (
   297→   297→          <li key={s.label} className="flex items-center gap-2 text-xs">
   298→   298→            <span className="h-2.5 w-2.5 shrink-0 rounded-full" style={{ backgroundColor: CHART_COLORS[i % CHART_COLORS.length] }} aria-hidden />
   299→   299→            <span className="min-w-0 flex-1 truncate" title={s.label}>{s.label}</span>
   300→   300→            <span className="font-mono tabular-nums text-muted-foreground">{fmtChart(s.value, decimals)}</span>
   301→   301→            <span className="w-14 text-right font-mono tabular-nums text-muted-foreground/70">
   302→   302→              {total > 0 ? `${((s.value / total) * 100).toFixed(1).replace(".", ",")} %` : "—"}
   303→   303→            </span>
   304→   304→          </li>
   305→   305→        ))}
   306→   306→      </ul>
   307→   307→    </div>
   308→   308→  )
   309→   309→}

   313→   313→function ChartBars({ items, unit, decimals }: {
   314→   314→  items: { label: string; value: number }[]
   315→   315→  unit: string
   316→   316→  decimals: number
   317→   317→}) {
   318→   318→  const sorted = [...items].sort((a, b) => b.value - a.value)
   319→   319→  const shown = sorted.slice(0, 12)
   320→   320→  const maxV = shown[0]?.value ?? 0
   321→   321→
   322→   322→  return (
   323→   323→    <div className="space-y-2">
   324→   324→      {shown.map((x, i) => (
   325→   325→        <div key={x.label} className="flex items-center gap-3">
   326→   326→          <span className="w-36 shrink-0 truncate text-right text-xs" title={x.label}>{x.label}</span>
   327→   327→          <div className="relative h-5 min-w-0 flex-1 overflow-hidden rounded-md bg-muted">
   328→   328→            <motion.div
   329→   329→              className="h-full rounded-md bg-emerald-600"
   330→   330→              style={{ opacity: 1 - Math.min(i, 8) * 0.08 }}
   331→   331→              initial={{ width: 0 }}
   332→   332→              animate={{ width: `${maxV > 0 ? Math.max((x.value / maxV) * 100, 2) : 0}%` }}
   333→   333→              transition={{ duration: 0.45, delay: i * 0.04, ease: "easeOut" }}
   334→   334→            />
   335→   335→            <span className="absolute inset-y-0 right-2 flex items-center font-mono text-[10px] tabular-nums text-foreground/80">
   336→   336→              {fmtChart(x.value, decimals)} {unit}
   337→   337→            </span>
   338→   338→          </div>
   339→   339→        </div>
   340→   340→      ))}
   341→   341→      {sorted.length > shown.length && (
   342→   342→        <p className="pl-[10.5rem] text-[11px] text-muted-foreground">… et {sorted.length - shown.length} autres</p>
   343→   343→      )}
   344→   344→    </div>
   345→   345→  )
   346→   346→}

   942→   942→/** Résumé FR du patch pour le toast : « population : 52 400 · taux urbain : 78 % ». */
   943→   943→function describePatch(patch: Record<string, number | null>): string {
   944→   944→  return Object.entries(patch)
   945→   945→    .map(([k, v]) => {
   946→   946→      const label = TOAST_LABELS[k] ?? k
   947→   947→      return v === null ? `${label} effacé` : `${label} : ${k.endsWith("Pct") ? fmtPct(v) : fmtInt(v)}`
   948→   948→    })
   949→   949→    .join(" · ")
   950→   950→}

   952→   952→/** Résumé FR du patch gouvernorat : « population : 1 056 300 · urbain : 99 % ». */
   953→   953→function describeGouvPatch(patch: Record<string, number | null>): string {
   954→   954→  return Object.entries(patch)
   955→   955→    .map(([k, v]) => {
   956→   956→      const label = GOV_TOAST_LABELS[k] ?? k
   957→   957→      if (v === null) return `${label} effacé`
   958→   958→      if (k === "areaKm2") return `${label} : ${fmtInt(v)} km²`
   959→   959→      if (k === "avgFleetAge") return `${label} : ${v.toLocaleString("fr-FR", { maximumFractionDigits: 1 })} ans`
   960→   960→      if (k.endsWith("Pct") || k === "motorizationRate") return `${label} : ${fmtPct(v)}`
   961→   961→      return `${label} : ${fmtInt(v)}`
   962→   962→    })
   963→   963→    .join(" · ")
   964→   964→}
