    95→let staticPromise: Promise<StaticLayers> | null = null
    96→function loadStatic(): Promise<StaticLayers> {
    97→  if (!staticPromise) {
    98→    staticPromise = (async () => {
    99→      const safe = async (url: string, key: string) => {
   100→        try {
   101→          const r = await fetch(url)
   102→          if (!r.ok) return []
   103→          const j = await r.json()
   104→          return j[key] ?? []
   105→        } catch { return [] }
   106→      }
   107→      const [roads, projects, governorats, services] = await Promise.all([
   108→        safe("/api/roads", "roads"),
   109→        safe("/api/projects", "projects"),
   110→        safe("/api/governorats", "governorats"),
   111→        safe("/api/services", "services"),
   112→      ])
   113→      return { roads, projects, governorats, services }
   114→    })()
   115→  }
   116→  return staticPromise
   117→}

   204→        offX: (r.width - MAP_W * scale) / 2,

   207→        h: r.height,

   221→    const offX = (r.width - MAP_W * scale) / 2
   222→    const offY = (r.height - MAP_H * scale) / 2

   224→  }, [])

   227→    const m = metrics()

   229→    const sx = (e.clientX - m.rect.left - m.offX) / m.scale
   230→    const sy = (e.clientY - m.rect.top - m.offY) / m.scale
   231→    return [sx, sy]
   232→  }, [metrics])

   235→    left: mstate.offX + (p.x * view.k + view.x) * mstate.scale,
   236→    top: mstate.offY + (p.y * view.k + view.y) * mstate.scale,

   241→    const el = svgRef.current
   242→    if (!el) return

   244→      e.preventDefault()

   248→        const factor = e.deltaY < 0 ? 1.18 : 1 / 1.18
   249→        const k = clamp(v.k * factor, 1, 14)
   250→        const mx = (pt[0] - v.x) / v.k
   251→        const my = (pt[1] - v.y) / v.k
   252→        return { k, x: pt[0] - mx * k, y: pt[1] - my * k }
   253→      })
   254→    }
   255→    el.addEventListener("wheel", onWheel, { passive: false })
   256→    return () => el.removeEventListener("wheel", onWheel)
   257→  }, [svgPointFromEvent])

   261→    const cx = MAP_W / 2, cy = MAP_H / 2
   262→    const mx = (cx - v.x) / v.k, my = (cy - v.y) / v.k
   263→    return { k, x: cx - mx * k, y: cy - my * k }
   264→  })
   265→
   266→  const resetView = () => setView({ k: 1, x: 0, y: 0 })
   267→
   268→  const fitBounds = useCallback((sts: StationDTO[]) => {
   269→    if (!sts.length) { resetView(); return }

   271→    const minX = Math.min(...pts.map(p => p.x)), maxX = Math.max(...pts.map(p => p.x))
   272→    const minY = Math.min(...pts.map(p => p.y)), maxY = Math.max(...pts.map(p => p.y))
   273→    const bw = Math.max(maxX - minX, 10), bh = Math.max(maxY - minY, 10)
   274→    const k = clamp(Math.min((MAP_W - 80) / bw, (MAP_H - 80) / bh), 1, 6)
   275→    const cx = (minX + maxX) / 2, cy = (minY + maxY) / 2
   276→    setView({ k, x: MAP_W / 2 - cx * k, y: MAP_H / 2 - cy * k })
   277→  }, [])
   278→
   279→  useEffect(() => {
   280→    if (!(autoFitOnLoad && stations.length)) return
   281→    const raf = requestAnimationFrame(() => fitBounds(stations))

   303→        selectStation(st)

   305→        setView({ k: 6, x: MAP_W / 2 - p.x * 6, y: MAP_H / 2 - p.y * 6 })

   307→    })
   308→    return () => cancelAnimationFrame(raf)
   309→  }, [focusCode, stations, selectStation])
   310→
   311→  // ---------- Détail fiche (dérivé : actif seulement si cohérent avec la sélection) ----------
   312→  const activeDetail = selCode && detailCode === selCode ? detail : null
   313→  useEffect(() => {
   314→    if (!selCode) return
   315→    const st = stations.find(s => s.code === selCode)
   316→    if (!st) return
   317→    if (detailCode === selCode) return

   324→  }, [selCode, stations, detailCode])
   325→
   326→  // ---------- Drag pan ----------
   327→  const dragRef = useRef<{ sx: number; sy: number; vx: number; vy: number } | null>(null)

   338→    if (!m) return

   384→    const set = new Set<string>()
   385→    for (const s of stations) if (pointInPolygon(s.lng, s.lat, poly)) set.add(s.code)

   387→  }, [polygonPts, polygonClosed, stations, tool])

   395→    let ext: string
   396→    if (format === "csv") {

   399→        [s.code, s.name, s.brandName, s.gouvernorat, s.delegation, s.status, s.traffic, s.lat, s.lng, s.services.join("|")].join(";")

   402→      ext = "csv"
   403→    } else {
   404→      const fc = {
   405→        type: "FeatureCollection",
   406→        features: selected.map(s => ({
   407→          type: "Feature",
   408→          geometry: { type: "Point", coordinates: [s.lng, s.lat] },
   409→          properties: { code: s.code, nom: s.name, enseigne: s.brandName, gouvernorat: s.gouvernorat, statut: s.status, trafic: s.traffic, services: s.services },
   410→        })),
   411→      }
   412→      blob = new Blob([JSON.stringify(fc, null, 2)], { type: "application/geo+json" })
   413→      ext = "geojson"
   414→    }
   415→    const a = document.createElement("a")
   416→    a.href = URL.createObjectURL(blob)
   417→    a.download = `carbuscope-selection.${ext}`

   419→    URL.revokeObjectURL(a.href)
   420→    toast.success(`${selected.length} stations exportées (${ext.toUpperCase()})`)
   421→  }
   422→
   423→  const clearTools = () => {
   424→    setTool(null); setMeasurePts([]); setRadiusCenter(null); setPolygonPts([]); setPolygonClosed(false)
   425→  }
   426→
   427→  const measureTotal = useMemo(() => {
   428→    if (measurePts.length < 2) return 0
   429→    let t = 0
   430→    for (let i = 1; i < measurePts.length; i++) {
   431→      const a = unproject(measurePts[i - 1][0], measurePts[i - 1][1])
   432→      const b = unproject(measurePts[i][0], measurePts[i][1])
   433→      t += haversineKm(a.lat, a.lng, b.lat, b.lng)
   434→    }
   435→    return t
   436→  }, [measurePts])
   437→
   438→  // ---------- Rendu stations (mémoïsé) ----------
   439→  const stationNodes = useMemo(() => {
   440→    return stations.map(s => {
   441→      const p = project(s.lng, s.lat)
   442→      const isOwn = ownBrandSlug && s.brandSlug === ownBrandSlug
   443→      const isSel = s.code === selCode
   444→      const isHover = s.code === hoverCode
   445→      const inRad = insideRadius?.has(s.code)
   446→      const inPoly = insidePolygon?.has(s.code)
   447→      const isPinned = compareSet.has(s.code)
   448→      const highlighted = isOwn || isSel || isHover || inRad || inPoly || isPinned
   449→      const r = isOwn ? 3.6 : 2.7
   450→      if (s.status === "CLOSED") {

   861→function brandLegend(stations: StationDTO[], ownSlug?: string | null): [string, string, number][] {
   862→  const m = new Map<string, [string, number]>()
   863→  for (const s of stations) {
   864→    const cur = m.get(s.brandName)
   865→    m.set(s.brandName, [s.brandColor, (cur?.[1] ?? 0) + 1])
   866→  }
   867→  return [...m.entries()]
   868→    .sort((a, b) => b[1][1] - a[1][1])
   869→    .slice(0, 8)
   870→    .map(([name, [color, count]]) => [ownSlug && stations.find(s => s.brandName === name)?.brandSlug === ownSlug ? `${name} (moi)` : name, color, count])
   871→}
