    37→type IconType = React.ComponentType<{ className?: string }>
    38→
    39→interface NavChild {

    47→interface NavItem {
    48→  id: string
    49→  icon: IconType
    50→  label: string
    51→  onSelect: () => void
    52→  active: boolean

    98→function useMenu(mode: Mode): MenuModel {
    99→  const { t } = useI18n()
   100→  const tab = useAppStore(s => s.tab)
   101→  const role = useAppStore(s => s.role)
   102→  const userId = useAppStore(s => s.userId)
   103→  const nav = useAppStore(s => s.nav)
   104→  const setTab = useAppStore(s => s.setTab)
   105→  const setNav = useAppStore(s => s.setNav)
   106→  const setMobileNavOpen = useAppStore(s => s.setMobileNavOpen)
   107→
   108→  const closeMobile = React.useCallback(() => setMobileNavOpen(false), [setMobileNavOpen])
   109→  const cur = React.useCallback((key: string) => nav[key] ?? NAV_DEFAULTS[key] ?? "", [nav])
   110→
   111→  // Onglet Démographie conditionnel (T37) : visible si le compte démo connecté
   112→  // dispose d'un accès démographie (sonde /api/demographie/acces).
   113→  // L'accès n'est écrit que dans les callbacks asynchrones (règle
   114→  // react-hooks/set-state-in-effect) ; l'absence d'e-mail dérive `false`.
   115→  const email = userId ? DEMO_ACCOUNTS_MAP[userId]?.email ?? null : null
   116→  const [demographieAccess, setDemographieAccess] = useState(false)
   117→  useEffect(() => {
   118→    if (!email) return
   119→    let alive = true
   120→    fetch(`/api/demographie/acces?email=${encodeURIComponent(email)}`)
   121→      .then(r => r.json())
   122→      .then(j => { if (alive) setDemographieAccess(Boolean(j?.access)) })
   123→      .catch(() => {})
   124→    return () => { alive = false }
   125→  }, [email])
   126→  const hasDemographie = email !== null && demographieAccess
   127→
   128→  // Badges du back-office (file de modération + missions ouvertes) — écrits
   129→  // uniquement dans les callbacks asynchrones ; hors mode back-office les
   130→  // badges ne sont simplement pas rendus par le menu.
   131→  const [boPending, setBoPending] = useState<number | null>(null)
   132→  const [boMissions, setBoMissions] = useState<number | null>(null)
   133→  useEffect(() => {
   134→    // T50 : les badges back-office sont chargés dès l'espace enseigne (le
   135→    // groupe G7 y est rendu), plus seulement dans l'ancien mode back-office.
   136→    if (mode !== "portail") return
   137→    let alive = true
   138→    fetch("/api/contributions?status=PENDING&limit=50")
   139→      .then(r => r.json())
   140→      .then(j => { if (alive) setBoPending((j.contributions ?? []).length) })
   141→      .catch(() => {})
   142→    fetch("/api/missions")
   143→      .then(r => r.json())
   144→      .then(j => {
   145→        if (!alive) return
   146→        setBoMissions((j.missions ?? []).filter((m: { status: string }) => m.status === "PLANIFIEE" || m.status === "EN_COURS").length)
   147→      })
   148→      .catch(() => {})
   149→    return () => { alive = false }
   150→  }, [mode])
   151→
   152→  return useMemo(() => {
   153→    // -- helpers -------------------------------------------------------------
   154→    const leaf = (
   155→      id: string, icon: IconType, label: string,
   156→      active: boolean, onSelect: () => void, extra?: Partial<NavItem>,
   157→    ): NavItem => ({ id, icon, label, active, onSelect, ...extra })
   158→
   159→    const branch = (
   160→      id: string, icon: IconType, label: string,
   161→      parentActive: boolean, parentSelect: () => void,
   162→      navKey: string, children: { id: string; icon: IconType; label: string }[],
   163→      locked?: boolean,
   164→    ): NavItem => ({
   165→      id, icon, label, active: parentActive, locked,
   166→      onSelect: parentSelect,
   167→      children: children.map(c => ({
   168→        id: c.id,
   169→        icon: c.icon,
   170→        label: c.label,
   171→        active: parentActive && cur(navKey) === c.id,
   172→        onSelect: () => {
   173→          if (locked) return
   174→          parentSelect()
   175→          setNav(navKey, c.id)
   176→          closeMobile()
   177→        },
   178→      })),
   179→    })
   180→
   181→    // -- ESPACE ENSEIGNE FUSIONNÉ (T50 — blueprint 8 groupes) -------------------
   182→    // Un seul espace « portail » embarque : observatoire (G1, G2, G5, G8),
   183→    // portail enseigne (G3), veille terrain + agents (G4), Shadow Market (G6),
   184→    // back-office plateforme (G7). Le module actif est porté par nav.espace.
   185→    if (mode === "portail") {
   186→      const espace = cur("espace")
   187→      const locked = (tb: Tab) => !TAB_ACCESS[tb].includes(role)
   188→      const allowed = (tb: Tab) => !locked(tb)
   189→      const go = (tb: Tab) => () => { setNav("espace", "observatoire"); setTab(tb); closeMobile() }
   190→
   191→      const obsTab = (tb: Tab, icon: IconType, labelKey: DictKey): NavItem =>
   192→        leaf(tb, icon, t(labelKey), espace === "observatoire" && tab === tb, allowed(tb) ? go(tb) : () => {}, { locked: locked(tb) })
   193→
   194→      const obsBranch = (
   195→        tb: Tab, icon: IconType, labelKey: DictKey, navKey: string,
   196→        children: { id: string; icon: IconType; label: string }[],
   197→      ): NavItem => {
   198→        const isLocked = locked(tb)
   199→        const select = () => { if (!isLocked) { setNav("espace", "observatoire"); setTab(tb); closeMobile() } }
   200→        return branch(tb, icon, t(labelKey), espace === "observatoire" && tab === tb && !isLocked, select, navKey, children, isLocked)
   201→      }
   202→
   203→      // G3 — onglets internes du Portail enseigne.
   204→      const pTab = (id: string, icon: IconType, label: string): NavItem =>
   205→        leaf(id, icon, label, espace === "portail" && cur("portail") === id, () => { setNav("espace", "portail"); setNav("portail", id); closeMobile() })
   206→
   207→      // G6 — sections Shadow Market (accordéons).
   208→      const sBranch = (
   209→        id: string, icon: IconType, label: string,
   210→        children: { id: string; icon: IconType; label: string }[],
   211→      ): NavItem =>
   212→        branch(
   213→          id, icon, label, espace === "shadow" && cur("shadow") === id,
   214→          () => { setNav("espace", "shadow"); setNav("shadow", id); closeMobile() },
   215→          `shadow-${id}`, children,
   216→        )
   217→
   218→      // G7 — outils du back-office plateforme.
   219→      const boItem = (id: string, icon: IconType, label: string, extra?: Partial<NavItem>): NavItem =>
   220→        leaf(id, icon, label, espace === "backoffice" && cur("backoffice") === id, () => { setNav("espace", "backoffice"); setNav("backoffice", id); closeMobile() }, extra)
   221→
   222→      const analyseItems: NavItem[] = [
   223→        obsTab("competition", Crosshair, "rm.tabs.competition"),
   224→        obsBranch("implantation", Target, "rm.tabs.implantation", "implantation", [
   225→          { id: "zones", icon: MapPinPlus, label: t("sb.sub.implantation.zones") },
   226→          { id: "isochrones", icon: Clock, label: t("sb.sub.implantation.isochrones") },
   227→          { id: "comparateur", icon: GitCompareArrows, label: t("sb.sub.implantation.comparateur") },
   228→          { id: "cannibalisation", icon: Swords, label: t("sb.sub.implantation.cannibalisation") },
   229→        ]),
   230→        obsBranch("calculators", Calculator, "rm.tabs.calculators", "calculators",
   231→          CALC_DEFS.map(def => ({
   232→            id: def.id,
   233→            icon: def.icon,
   234→            label: t(`sb.sub.calculators.${def.id}` as DictKey),
   235→          })),
   236→        ),
   237→      ]
   238→      if (hasDemographie && email) {
   239→        analyseItems.push(obsTab("demographie", Users, "rm.tabs.demographie"))
   240→      }
   241→
   242→      // Blueprint 8 groupes — G6 et G7 sont conditionnés aux rôles (T50).
   243→      const groups: NavGroup[] = [
   244→        {
   245→          id: "f-pilotage", label: t("sb.groups.pilotage"), items: [
   246→            obsTab("dashboard", LayoutDashboard, "rm.tabs.dashboard"),
   247→            obsTab("map", Map, "rm.tabs.map"),
   248→          ],
   249→        },
   250→        {
   251→          id: "f-reseau", label: t("sb.groups.reseau"), items: [
   252→            obsTab("stations", Fuel, "rm.tabs.stations"),
   253→            obsTab("prices", Banknote, "rm.tabs.prices"),
   254→            obsTab("veille", Radar, "rm.tabs.veille"),
   255→          ],
   256→        },
   257→        {
   258→          id: "f-enseigne", label: "Espace enseigne", items: [
   259→            pTab("dashboard", LayoutDashboard, "Tableau de bord enseigne"),
   260→            pTab("stations", Store, "Mes stations"),
   261→            pTab("reviews", Star, "Avis & réputation"),
   262→            pTab("api", PlugZap, "API & promotions"),
   263→          ],
   264→        },
   265→        {
   266→          id: "f-terrain", label: "Veille terrain", items: [
   267→            branch(
   268→              "terrain", Footprints, "Missions terrain",
   269→              espace === "portail" && cur("portail") === "terrain",
   270→              () => { setNav("espace", "portail"); setNav("portail", "terrain"); closeMobile() },
   271→              "portal-terrain", [
   272→                { id: "missions", icon: ClipboardList, label: "Mes missions" },
   273→                { id: "own", icon: ShieldCheck, label: "Fiche propriétaire" },
   274→                { id: "competitors", icon: Eye, label: "Fiches concurrents" },
   275→              ],
   276→            ),
   277→            leaf("agents", Smartphone, "Application agents terrain", espace === "agents", () => { setNav("espace", "agents"); closeMobile() }),
   278→          ],
   279→        },
   280→        { id: "f-analyse", label: t("sb.groups.analyse"), items: analyseItems },
   281→      ]
   282→
   283→      if (SHADOW_MODULE_ROLES.includes(role)) {
   284→        groups.push({
   285→          id: "f-shadow", label: "Shadow Market", items: [
   286→            sBranch("overview", BarChart3, "Vue d'Ensemble", [
   287→              { id: "kpis", icon: BarChart3, label: "Indicateurs Clés" },
   288→              { id: "alerts", icon: TriangleAlert, label: "Alertes" },
   289→              { id: "top", icon: Trophy, label: "Top 10 Menaces" },
   290→              { id: "calendar", icon: CalendarDays, label: "Calendrier" },
   291→            ]),
   292→            sBranch("cartography", Map, "Cartographie", [
   293→              { id: "national", icon: Globe, label: "Carte Nationale" },
   294→              { id: "district", icon: Target, label: "Par District" },
   295→              { id: "station", icon: MapPin, label: "Par Station" },
   296→              { id: "road", icon: Route, label: "Par Axe Routier" },
   297→              { id: "redzones", icon: TriangleAlert, label: "Zones Rouges" },
   298→            ]),
   299→            sBranch("recensement", ClipboardList, "Recensement", [
   300→              { id: "nouveau", icon: Plus, label: "Nouveau Point" },
   301→              { id: "liste", icon: List, label: "Liste des Points" },
   302→              { id: "maj", icon: RefreshCw, label: "Mises à Jour" },
   303→              { id: "audit", icon: Users, label: "Équipe d'Audit" },
   304→              { id: "validation", icon: CircleCheck, label: "Validation" },
   305→              { id: "galerie", icon: Camera, label: "Galerie Photos" },
   306→            ]),
   307→            sBranch("impact", TrendingDown, "Analyse d'Impact", [
   308→              { id: "station", icon: Landmark, label: "Perte par Station" },
   309→              { id: "region", icon: Map, label: "Perte par Région" },
   310→              { id: "cannibalisation", icon: Scale, label: "Cannibalisation" },
   311→              { id: "tendance", icon: TrendingUp, label: "Tendance" },
   312→              { id: "simulation", icon: SlidersHorizontal, label: "Simulation" },
   313→            ]),
   314→            sBranch("reports", FileText, "Rapports", [
   315→              { id: "monthly", icon: CalendarDays, label: "Rapport Mensuel" },
   316→              { id: "brand", icon: Building2, label: "Rapport par Enseigne" },
   317→              { id: "custom", icon: CalendarRange, label: "Rapport Personnalisé" },
   318→              { id: "export", icon: Share2, label: "Export / Partage" },
   319→            ]),
   320→            sBranch("settings", Settings, "Paramètres", [
   321→              { id: "access", icon: KeyRound, label: "Accès & Droits" },
   322→              { id: "radii", icon: Ruler, label: "Rayons & Seuils" },
   323→              { id: "typologies", icon: Tags, label: "Typologies" },
   324→              { id: "notifications", icon: Bell, label: "Notifications" },
   325→            ]),
   326→          ],
   327→        })
   328→      }
   329→
   330→      if (PLATFORM_MODULE_ROLES.includes(role)) {
   331→        groups.push({
   332→          id: "f-backoffice", label: "Back-office plateforme", items: [
   333→            boItem("moderation", ShieldAlert, "Modération", { badge: boPending, badgeTone: "danger" }),
   334→            boItem("sovereign", Database, "Données souveraines"),
   335→            boItem("freshness", TimerReset, "Fraîcheur"),
   336→            boItem("agents", UserCog, "Agents terrain"),
   337→            boItem("alerts", Radar, "Veille & alertes"),
   338→            boItem("stats", BarChart3, "Stats plateforme"),
   339→            boItem("decrees", Banknote, "Prix & décrets"),
   340→            boItem("missions", ClipboardList, "Missions", { badge: boMissions, badgeTone: "accent" }),
   341→          ],
   342→        })
   343→      }
   344→
   345→      groups.push({
   346→        id: "f-gestion", label: t("sb.groups.gestion"), items: [
   347→          obsTab("reports", FileText, "rm.tabs.reports"),
   348→          obsTab("settings", Settings, "rm.tabs.settings"),
   349→        ],
   350→      })
   351→
   352→      return {
   353→        title: t("chrome.modePortail"),
   354→        icon: Building,
   355→        groups,
   356→      }
   357→    }
   358→
   359→    // -- SUPER ADMIN ------------------------------------------------------------
   360→    if (mode === "superadmin") {
   361→      const saItem = (id: string, icon: IconType, label: string): NavItem =>
   362→        leaf(id, icon, label, cur("superadmin") === id, () => { setNav("superadmin", id); closeMobile() })
   363→      return {
   364→        title: t("chrome.modeSuperadmin"),
   365→        icon: Crown,
   366→        groups: [{
   367→          id: "superadmin", label: "Super Administration", items: [
   368→            saItem("overview", LayoutDashboard, "Vue d'ensemble"),
   369→            saItem("contracts", FileSignature, "Enseignes & contrats"),

   372→            saItem("users", UsersRound, "Utilisateurs"),
   373→            saItem("demo", BarChart3, "Démographie"),

   375→            saItem("carto", MapPinned, "Cartographie"),
   376→            saItem("audit", ScrollText, "Journal d'audit"),
   377→          ],
   378→        }],
   379→      }
   380→    }
   381→
   382→    // -- MOBILE B2C (aperçu téléphone) ------------------------------------------
   383→    if (mode === "mobile") {
   384→      const mItem = (id: string, icon: IconType, label: string): NavItem =>
   385→        leaf(id, icon, label, cur("mobile") === id, () => { setNav("mobile", id); closeMobile() })
   386→      return {
   387→        title: t("chrome.modeMobile"),
   388→        icon: Smartphone,
   389→        groups: [{
   390→          id: "mobile", label: t("sb.mobile.title"), items: [
   391→            mItem("map", Map, "Carte"),
   392→            mItem("search", Search, "Recherche"),

   401→    }
   402→
   403→    // Repli défensif (union exhaustive : portail | mobile | superadmin) —
   404→    // rend un menu vide plutôt qu'un crash si un mode inconnu apparaissait.
   405→    return { title: t("chrome.modePortail"), icon: Building, groups: [] }
   406→  }, [mode, tab, role, nav, hasDemographie, email, boPending, boMissions, t, setTab, setNav, closeMobile, cur])
   407→}

   412→
   413→function NavRow({
   414→  item, open, onToggleOpen, dir,
   415→}: {
   416→  item: NavItem
   417→  open: boolean
   418→  onToggleOpen: () => void
   419→  dir: string
   420→}) {
   421→  const { t } = useI18n()
   422→  const hasChildren = (item.children?.length ?? 0) > 0
   423→
   424→  const btn = (
   425→    <button
   426→      type="button"
   427→      onClick={() => {
   428→        if (item.locked) return
   429→        item.onSelect()
   430→        if (hasChildren) onToggleOpen()
   431→      }}
   432→      disabled={item.locked}
   433→      aria-current={item.active ? "page" : undefined}
   434→      aria-expanded={hasChildren ? open : undefined}
   435→      className={cn(
   436→        "group flex w-full items-center gap-2.5 rounded-lg px-2.5 py-2 text-[13px] font-medium transition-colors",
   437→        item.locked
   438→          ? "cursor-not-allowed text-[var(--c-ink-ghost)]"
   439→          : item.active
   440→            ? "bg-[var(--c-brown)] text-[var(--primary-foreground)] shadow-sm"
   441→            : "text-[var(--c-ink)] hover:bg-[var(--c-tint)]",
   442→      )}
   443→    >
   444→      {item.locked ? (
   445→        <Lock className="h-[17px] w-[17px] shrink-0" aria-hidden />
   446→      ) : (
   447→        <item.icon className="h-[17px] w-[17px] shrink-0" aria-hidden />
   448→      )}
   449→      <span className="min-w-0 flex-1 truncate text-start">{item.label}</span>
   450→      {item.badge != null && item.badge > 0 && (
   451→        <span
   452→          className={cn(
   453→            "flex h-[18px] min-w-[18px] shrink-0 items-center justify-center rounded-full px-1 text-[10px] font-bold tabular-nums",
   454→            item.badgeTone === "danger"
   455→              ? "bg-[var(--c-terra)] text-white"
   456→              : "bg-[var(--c-amber)]/25 text-[var(--c-amber)]",
   457→            item.active && "ring-1 ring-white/40",
   458→          )}
   459→        >
   460→          {item.badge > 99 ? "99+" : item.badge}
   461→        </span>
   462→      )}
   463→      {hasChildren && (
   464→        <ChevronRight
   465→          className="h-3.5 w-3.5 shrink-0 opacity-60 transition-transform duration-200"
   466→          style={{ transform: open ? "rotate(90deg)" : dir === "rtl" ? "rotate(180deg)" : "rotate(0deg)" }}
   467→          aria-hidden
   468→        />
   469→      )}
   470→    </button>
   471→  )
   472→
   473→  return (
   474→    <li>
   475→      {item.locked ? (
   476→        <Tooltip>
   477→          <TooltipTrigger asChild>{btn}</TooltipTrigger>
   478→          <TooltipContent side="right">
   479→            <p>{t("sb.lockedHint")}</p>
   480→            <p className="text-[11px] text-muted-foreground">{t("sb.lockedChange")}</p>
   481→          </TooltipContent>
   482→        </Tooltip>
   483→      ) : btn}
   484→      <AnimatePresence initial={false}>
   485→        {open && hasChildren && (
   486→          <motion.div
   487→            key="submenu"
   488→            initial={{ height: 0, opacity: 0 }}
   489→            animate={{ height: "auto", opacity: 1 }}
   490→            exit={{ height: 0, opacity: 0 }}
   491→            transition={{ duration: 0.2, ease: [0.32, 0.72, 0, 1] }}
   492→            className="overflow-hidden"
   493→          >
   494→            <ul className="mx-3 my-1 space-y-0.5 border-s border-[var(--c-line)] ps-2">
   495→              {item.children!.map(child => (
   496→                <li key={child.id}>
   497→                  <button
   498→                    type="button"
   499→                    onClick={child.onSelect}
   500→                    disabled={item.locked}
   501→                    aria-current={child.active ? "true" : undefined}
   502→                    className={cn(
   503→                      "flex w-full items-center gap-2 rounded-md px-2 py-1.5 text-xs transition-colors",
   504→                      item.locked
   505→                        ? "cursor-not-allowed text-[var(--c-ink-ghost)]"
   506→                        : child.active
   507→                          ? "bg-[var(--c-brown)]/12 font-semibold text-[var(--c-brown)]"
   508→                          : "text-[var(--c-ink-muted)] hover:bg-[var(--c-tint)] hover:text-[var(--c-ink)]",
   509→                    )}
   510→                  >
   511→                    <child.icon className="h-3.5 w-3.5 shrink-0" aria-hidden />
   512→                    <span className="min-w-0 truncate text-start">{child.label}</span>
   513→                  </button>
   514→                </li>
   515→              ))}
   516→            </ul>
   517→          </motion.div>
   518→        )}
   519→      </AnimatePresence>
   520→    </li>
   521→  )
   522→}

   575→  const toggleOpen = (id: string) => {
   576→    setOpenMap(m => {
   577→      const current = m[id] ?? menu.groups.flatMap(g => g.items).find(i => i.id === id)?.active ?? false
   578→      return { ...m, [id]: !current }
   579→    })

   583→  useEffect(() => {

   585→      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === "b") {
   586→        e.preventDefault()

   588→      }

   590→    window.addEventListener("keydown", onKey)
