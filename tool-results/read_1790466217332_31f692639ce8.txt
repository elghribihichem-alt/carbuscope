   414→   944→   944→/** Résumé FR du patch pour le toast : « population : 52 400 · taux urbain : 78 % ». */
   415→   945→   945→function describePatch(patch: Record<string, number | null>): string {
   416→   946→   946→  return Object.entries(patch)
   417→   947→   947→    .map(([k, v]) => {
   418→   948→   948→      const label = TOAST_LABELS[k] ?? k
   419→   949→   949→      return v === null ? `${label} effacé` : `${label} : ${k.endsWith("Pct") ? fmtPct(v) : fmtInt(v)}`
   420→   950→   950→    })
   421→   951→   951→    .join(" · ")
   422→   952→   952→}

   426→   956→   956→  return Object.entries(patch)

   435→   965→   965→    .join(" · ")
