    74→    74→function sortField(key: SortKey): string {
    75→    75→  return key.slice(0, key.lastIndexOf("-"))
    76→    76→}

    78→    78→function toggleSort(cur: SortKey, target: SortKey): SortKey {
    79→    79→  if (sortField(cur) !== sortField(target)) return target
    80→    80→  const field = sortField(cur)
    81→    81→  return cur.endsWith("-asc") ? (`${field}-desc` as SortKey) : (`${field}-asc` as SortKey)
    82→    82→}

    84→    84→function compareStations(a: StationDTO, b: StationDTO, sort: SortKey): number {
    85→    85→  const field = sortField(sort)
    86→    86→  const mul = sort.endsWith("-asc") ? 1 : -1
    87→    87→  let r = 0
    88→    88→  if (field === "code") r = a.code.localeCompare(b.code, "fr")
    89→    89→  else if (field === "name") r = a.name.localeCompare(b.name, "fr")
    90→    90→  else if (field === "gov") r = a.gouvernorat.localeCompare(b.gouvernorat, "fr") || a.delegation.localeCompare(b.delegation, "fr")
    91→    91→  else if (field === "traffic") r = a.traffic - b.traffic
    92→    92→  else if (field === "verified") r = new Date(a.lastVerified).getTime() - new Date(b.lastVerified).getTime()
    93→    93→  else if (field === "status") r = (STATUS_ORDER[a.status] ?? 9) - (STATUS_ORDER[b.status] ?? 9)
    94→    94→  if (r === 0) return a.code.localeCompare(b.code, "fr")
    95→    95→  return r * mul
    96→    96→}

    98→    98→function SortTh({
    99→    99→  label, target, sort, onSort, className,
   100→   100→}: {
   101→   101→  label: string
   102→   102→  target: SortKey
   103→   103→  sort: SortKey
   104→   104→  onSort: (k: SortKey) => void
   105→   105→  className?: string
   106→   106→}) {
   107→   107→  const isActive = sortField(sort) === sortField(target)
   108→   108→  const asc = sort.endsWith("-asc")
   109→   109→  return (
   110→   110→    <th scope="col" className={cn("h-10 whitespace-nowrap px-2 text-left align-middle font-medium", className)}>
   111→   111→      <button
   112→   112→        type="button"
   113→   113→        onClick={() => onSort(target)}
   114→   114→        className="inline-flex items-center gap-1 text-[11px] font-medium uppercase tracking-wide text-muted-foreground transition-colors hover:text-foreground focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-ring"
   115→   115→        aria-label={`Trier par ${label}${isActive ? (asc ? " (croissant)" : " (décroissant)") : ""}`}
   116→   116→      >
   117→   117→        {label}
   118→   118→        {isActive
   119→   119→          ? (asc ? <ArrowUp className="h-3 w-3" aria-hidden /> : <ArrowDown className="h-3 w-3" aria-hidden />)
   120→   120→          : <ArrowUpDown className="h-3 w-3 opacity-40" aria-hidden />}
   121→   121→      </button>
   122→   122→    </th>
   123→   123→  )
   124→   124→}

   134→   134→function TrafficCell({ station, maxTraffic }: { station: StationDTO; maxTraffic: number }) {
   135→   135→  const pct = maxTraffic > 0 ? Math.min(100, Math.round((station.traffic / maxTraffic) * 100)) : 0
   136→   136→  return (
   137→   137→    <div className="flex items-center gap-2">
   138→   138→      <span className="w-14 text-right font-medium tabular-nums">{num(station.traffic)}</span>
   139→   139→      <div className="h-1.5 w-14 shrink-0 overflow-hidden rounded-full bg-muted" aria-hidden="true">
   140→   140→        <div
   141→   141→          className={cn("h-full rounded-full", station.status === "ACTIVE" && station.traffic < TENSION_THRESHOLD ? "bg-amber-500" : "bg-primary")}
   142→   142→          style={{ width: `${pct}%` }}
   143→   143→        />
   144→   144→      </div>
   145→   145→    </div>
   146→   146→  )
   147→   147→}
