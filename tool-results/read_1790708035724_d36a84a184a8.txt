    40→function csvEscape(v: string | number): string {
    41→  const s = String(v ?? "")
    42→  return /[",;\n\r]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s
    43→}
    44→
    45→function csvRow(cells: (string | number)[]): string {
    46→  return cells.map(csvEscape).join(";")
    47→}
    48→
    49→function downloadCsv(filename: string, content: string): void {
    50→  const blob = new Blob([`\ufeff${content}`], { type: "text/csv;charset=utf-8;" })
    51→  const url = URL.createObjectURL(blob)
    52→  const a = document.createElement("a")
    53→  a.href = url
    54→  a.download = filename
    55→  document.body.appendChild(a)
    56→  a.click()
    57→  document.body.removeChild(a)
    58→  URL.revokeObjectURL(url)
    59→}

    68→function exportContractsCsv(rows: Row[]): void {
    69→  const lines: string[] = [
    70→    csvRow([
    71→      "Enseigne", "Plan", "Statut", "Tarif mensuel (DT)", "Sièges actifs",
    72→      "Quota sièges", "Stations", "Contact", "Renouvellement", "Impayé (DT)",
    73→    ]),
    74→  ]
    75→  for (const c of rows) {
    76→    lines.push(csvRow([
    77→      c.brandName,
    78→      PLAN_LABELS[c.plan] ?? c.plan,
    79→      CONTRACT_STATUS_META[c.status]?.label ?? c.status,
    80→      fmtCsvAmount(c.monthlyFee),
    81→      c.activeUsers,
    82→      c.seats,
    83→      `${c.stationCount} / ${c.stationQuota}`,
    84→      `${c.contactName} (${c.contactEmail})`,
    85→      new Date(c.renewsAt).toLocaleDateString("fr-FR"),
    86→      fmtCsvAmount(c.unpaidTotal),
    87→    ]))
    88→  }
    89→  downloadCsv(`carbuscope-contrats-${todayStamp()}.csv`, lines.join("\r\n"))
    90→}

    94→/** ISO → valeur d'un input type=date (YYYY-MM-DD), fuseau local. */
    95→function isoToLocalDateInput(iso: string): string {
    96→  const d = new Date(iso)
    97→  const p = (n: number) => String(n).padStart(2, "0")
    98→  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())}`
    99→}

   101→/** Décimal français accepté : « 890 » ou « 890,5 ». null si vide/invalide. */
   102→function parseDecimal(raw: string): number | null {
   103→  const t = raw.trim().replace(",", ".")
   104→  if (t === "") return null
   105→  const v = Number(t)
   106→  return Number.isFinite(v) ? v : null
   107→}

   109→function ContractEditForm({
   110→  contract, actor, onClose, onSaved,
   111→}: {
   112→  contract: Row
   113→  actor: string
   114→  onClose: () => void
   115→  onSaved: (contract: Row, plan: string, fee: number) => Promise<void>
   116→}) {
   117→  const [plan, setPlan] = useState<string>(contract.plan)
   118→  const [fee, setFee] = useState(contract.monthlyFee > 0 ? String(contract.monthlyFee) : "")
   119→  const [seats, setSeats] = useState(String(contract.seats))
   120→  const [quota, setQuota] = useState(String(contract.stationQuota))
   121→  const [renews, setRenews] = useState(isoToLocalDateInput(contract.renewsAt))
   122→  const [saving, setSaving] = useState(false)
   123→  const [error, setError] = useState<string | null>(null)
   124→
   125→  async function submit(e: React.FormEvent<HTMLFormElement>) {
   126→    e.preventDefault()
   127→    if (saving) return
   128→    setError(null)
   129→
   130→    const feeNum = parseDecimal(fee)
   131→    if (feeNum === null || feeNum < 1 || feeNum > 100000) {
   132→      setError("Tarif mensuel invalide — saisissez un montant entre 1 et 100 000 DT.")
   133→      return
   134→    }
   135→    if (!/^\d+$/.test(seats.trim()) || Number(seats.trim()) < 1 || Number(seats.trim()) > 500) {
   136→      setError("Nombre de sièges invalide — entier entre 1 et 500.")
   137→      return
   138→    }
   139→    if (!/^\d+$/.test(quota.trim()) || Number(quota.trim()) < 1 || Number(quota.trim()) > 5000) {
   140→      setError("Quota de stations invalide — entier entre 1 et 5 000.")
   141→      return
   142→    }
   143→    if (!renews || Number.isNaN(Date.parse(renews))) {
   144→      setError("Date de renouvellement invalide.")
   145→      return
   146→    }
   147→    const seatsNum = Number(seats.trim())
   148→    const quotaNum = Number(quota.trim())
   149→
   150→    // N'envoyer que les champs réellement modifiés (pattern station-form-dialog).
   151→    const data: Record<string, unknown> = {}
   152→    if (plan !== contract.plan) data.plan = plan
   153→    if (feeNum !== contract.monthlyFee) data.monthlyFee = feeNum
   154→    if (seatsNum !== contract.seats) data.seats = seatsNum
   155→    if (quotaNum !== contract.stationQuota) data.stationQuota = quotaNum
   156→    if (renews !== isoToLocalDateInput(contract.renewsAt)) data.renewsAt = renews
   157→
   158→    if (Object.keys(data).length === 0) {
   159→      toast.info("Aucune modification à enregistrer.")
   160→      onClose()
   161→      return
   162→    }
   163→
   164→    setSaving(true)
   165→    try {
   166→      const r = await fetch(`/api/super/contracts/${contract.id}`, {
   167→        method: "PATCH",
   168→        headers: { "Content-Type": "application/json" },
   169→        body: JSON.stringify({ ...data, actor }),
   170→      })
   171→      if (!r.ok) {
   172→        const j = await r.json().catch(() => null)
   173→        throw new Error(j?.error ?? `Erreur ${r.status}`)
   174→      }
   175→      await onSaved(contract, plan, feeNum)
   176→      onClose()
   177→    } catch (e) {
   178→      setError(e instanceof Error ? e.message : "Échec de la modification du contrat.")
   179→    } finally {
   180→      setSaving(false)
   181→    }
   182→  }
   183→
   184→  return (
   185→    // noValidate : la validation française (role="alert") gère tous les cas,
   186→    // y compris les valeurs vides/négatives, sans tooltip natif du navigateur.
   187→    <form onSubmit={submit} noValidate className="space-y-4">
   188→      <div className="grid gap-3 sm:grid-cols-2">
   189→        <div className="space-y-1.5">
   190→          <Label htmlFor="sa-plan">Plan</Label>
   191→          <Select value={plan} onValueChange={setPlan}>
   192→            <SelectTrigger id="sa-plan" className="w-full">
   193→              <SelectValue placeholder="Sélectionner un plan" />
   194→            </SelectTrigger>
   195→            <SelectContent>
   196→              {PLAN_OPTIONS.map(p => (
   197→                <SelectItem key={p} value={p}>{PLAN_LABELS[p] ?? p}</SelectItem>
   198→              ))}
   199→            </SelectContent>
   200→          </Select>
   201→        </div>
   202→        <div className="space-y-1.5">
   203→          <Label htmlFor="sa-fee">Tarif mensuel</Label>
   204→          <div className="relative">
   205→            <Input
   206→              id="sa-fee"
   207→              type="number"
   208→              min={1}
   209→              max={100000}
   210→              step={0.001}
   211→              inputMode="decimal"
   212→              value={fee}
   213→              onChange={e => setFee(e.target.value)}
   214→              placeholder="Ex. 890"
   215→              className="pr-10 tabular-nums"
   216→              aria-describedby="sa-fee-hint"
   217→            />
   218→            <span className="pointer-events-none absolute inset-y-0 right-3 flex items-center text-xs font-medium text-muted-foreground" aria-hidden>DT</span>
   219→          </div>
   220→          <p id="sa-fee-hint" className="text-[11px] text-muted-foreground">DT par mois, jusqu&apos;à 3 décimales.</p>
   221→        </div>
   222→        <div className="space-y-1.5">
   223→          <Label htmlFor="sa-seats">Sièges (utilisateurs inclus)</Label>
   224→          <Input
   225→            id="sa-seats"
   226→            type="number"
   227→            min={1}
   228→            max={500}
   229→            step={1}
   230→            inputMode="numeric"
   231→            value={seats}
   232→            onChange={e => setSeats(e.target.value)}
   233→            className="tabular-nums"
   234→          />
   235→        </div>
   236→        <div className="space-y-1.5">
   237→          <Label htmlFor="sa-quota">Quota stations</Label>
   238→          <Input
   239→            id="sa-quota"
   240→            type="number"
   241→            min={1}
   242→            max={5000}
   243→            step={1}
   244→            inputMode="numeric"
   245→            value={quota}
   246→            onChange={e => setQuota(e.target.value)}
   247→            className="tabular-nums"
   248→          />
   249→        </div>
   250→        <div className="space-y-1.5 sm:col-span-2">
   251→          <Label htmlFor="sa-renews">Renouvellement</Label>
   252→          <Input
   253→            id="sa-renews"
   254→            type="date"
   255→            value={renews}
   256→            onChange={e => setRenews(e.target.value)}
   257→          />
   258→          <p className="text-[11px] text-muted-foreground">
   259→            Prochaine échéance annuelle du contrat — actuellement le {fmtDate(contract.renewsAt)}.
   260→          </p>
   261→        </div>
   262→      </div>
   263→
   264→      {error && (
   265→        <p role="alert" className="rounded-md border border-destructive/40 bg-destructive/10 px-3 py-2 text-xs font-medium text-destructive">
   266→          {error}
   267→        </p>
   268→      )}
   269→
   270→      <DialogFooter className="gap-2">
   271→        <Button type="button" variant="outline" size="sm" disabled={saving} onClick={onClose}>
   272→          Annuler
   273→        </Button>
   274→        <Button type="submit" size="sm" className="gap-1.5" disabled={saving}>
   275→          {saving ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <Save className="h-4 w-4" aria-hidden />}
   276→          Enregistrer les modifications
   277→        </Button>
   278→      </DialogFooter>
   279→    </form>
   280→  )
   281→}

   318→/** Valeur d'un input type=date pour aujourd'hui + n jours, fuseau local. */
   319→function localDateInDays(n: number): string {
   320→  const d = new Date(Date.now() + n * 86_400_000)
   321→  const p = (x: number) => String(x).padStart(2, "0")
   322→  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())}`
   323→}

   331→  actor: string
   332→  onClose: () => void
   333→  onCreated: () => Promise<void>

   335→  const [brandSlug, setBrandSlug] = useState<string>("")

   339→  const [quota, setQuota] = useState("")

   341→  const [contactName, setContactName] = useState("")
   342→  const [contactEmail, setContactEmail] = useState("")
   343→  const [saving, setSaving] = useState(false)

   346→  const freeBrands = brands.filter(b => !takenSlugs.has(b.slug))

   348→  function pickPlan(code: string) {

   363→
   364→    if (!brandSlug) {
   365→      setError("Sélectionnez une enseigne cliente.")

   370→      return

   372→    if (!plan) {
   373→      setError("Sélectionnez un plan du catalogue.")
   374→      return
   375→    }
   376→    const feeNum = parseDecimal(fee)
   377→    if (feeNum === null || feeNum < 1 || feeNum > 100000) {

   382→      setError("Nombre de sièges invalide — entier entre 1 et 500.")
   383→      return
   384→    }
   385→    if (!/^\d+$/.test(quota.trim()) || Number(quota.trim()) < 1 || Number(quota.trim()) > 5000) {
   386→      setError("Quota de stations invalide — entier entre 1 et 5 000.")

   388→    }
   389→    if (!renews || Number.isNaN(Date.parse(renews))) {
   390→      setError("Date de renouvellement invalide.")
   391→      return
   392→    }
   393→    if (contactEmail.trim() && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(contactEmail.trim())) {
   394→      setError("Adresse e-mail de contact invalide.")
   395→      return
   396→    }

   398→    const quotaNum = Number(quota.trim())

   403→        method: "POST",
   404→        headers: { "Content-Type": "application/json" },

   406→          brandSlug, plan, monthlyFee: feeNum, seats: seatsNum, stationQuota: quotaNum,

   413→      }

   420→      onClose()

   422→      setError(e instanceof Error ? e.message : "Échec de la création du contrat.")

   568→function NewContractDialog({
   569→  open, takenSlugs, actor, onClose, onCreated,
   570→}: {
   571→  open: boolean
   572→  takenSlugs: Set<string>
   573→  actor: string
   574→  onClose: () => void
   575→  onCreated: () => Promise<void>
   576→}) {
   577→  const [brands, setBrands] = useState<BrandOption[] | null>(null)
   578→  const [plans, setPlans] = useState<SaasPlanDTO[] | null>(null)
   579→  const [loading, setLoading] = useState(false)
   580→  const [loadError, setLoadError] = useState<string | null>(null)
   581→
   582→  // Chargement à l'ouverture (pattern reload : états gérés dans la fonction).
   583→  const loadOptions = useCallback(async (open: boolean) => {
   584→    if (!open) return
   585→    setLoading(true)
   586→    setBrands(null)
   587→    setPlans(null)
   588→    setLoadError(null)
   589→    try {
   590→      const [rb, rp] = await Promise.all([
   591→        fetch("/api/brands").then(r => (r.ok ? r.json() : Promise.reject(new Error(`HTTP ${r.status}`)))),
   592→        fetch("/api/super/plans").then(r => (r.ok ? r.json() : Promise.reject(new Error(`HTTP ${r.status}`)))),
   593→      ]) as [{ brands: BrandOption[] }, { plans: SaasPlanDTO[] }]
   594→      setBrands(rb.brands)
   595→      setPlans(rp.plans)
   596→    } catch {
   597→      setLoadError("Impossible de charger les enseignes et les plans.")
   598→    } finally {
   599→      setLoading(false)
   600→    }
   601→  }, [])
   602→
   603→  useEffect(() => {
   604→    loadOptions(open)
   605→  }, [open, loadOptions])
   606→
   607→  const freeCount = brands ? brands.filter(b => !takenSlugs.has(b.slug)).length : 0
   608→
   609→  return (
   610→    <Dialog open={open} onOpenChange={o => { if (!o) onClose() }}>
   611→      <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-lg">
   612→        <DialogHeader>
   613→          <DialogTitle>Nouveau contrat</DialogTitle>
   614→          <DialogDescription>
   615→            Ouvre un essai de 30 jours pour une enseigne encore sans contrat — création journalisée dans l&apos;audit.
   616→          </DialogDescription>
   617→        </DialogHeader>
   618→        {loadError && (
   619→          <p role="alert" className="rounded-md border border-destructive/40 bg-destructive/10 px-3 py-2 text-xs font-medium text-destructive">
   620→            {loadError}
   621→          </p>
   622→        )}
   623→        {!brands || !plans || loading ? (
   624→          <p className="flex items-center gap-2 py-4 text-sm text-muted-foreground">
   625→            <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> Chargement des enseignes et des plans…
   626→          </p>
   627→        ) : freeCount === 0 ? (
   628→          <p className="py-4 text-sm text-muted-foreground">
   629→            Toutes les enseignes du répertoire ont déjà un contrat — aucun nouveau client disponible.
   630→          </p>
   631→        ) : (
   632→          <ContractCreateForm
   633→            key={brands.map(b => b.slug).join(",")}
   634→            brands={brands}
   635→            plans={plans}
   636→            takenSlugs={takenSlugs}
   637→            actor={actor}
   638→            onClose={onClose}
   639→            onCreated={onCreated}
   640→          />
   641→        )}
   642→      </DialogContent>
   643→    </Dialog>
   644→  )
   645→}
