    87→const monthBounds = (month: string): [string, string] => {
    88→  const [y, m] = month.split("-").map(Number)
    89→  const last = new Date(y, m, 0).getDate()
    90→  const mm = String(m).padStart(2, "0")
    91→  return [`${month}-01`, `${month}-${String(last).padStart(2, "0")}`]
    92→}

   100→
   101→export function CustomReportTab({ ctx }: { ctx: ShadowReportsCtx }) {
   102→  const { reports, overview, scopeLabel } = ctx
   103→
   104→  // Étape courante + champs du constructeur
   105→  const [step, setStep] = useState(1)
   106→  const [from, setFrom] = useState(monthStart())
   107→  const [to, setTo] = useState(todayStr())
   108→  const [preset, setPreset] = useState("custom")
   109→  const [scopeType, setScopeType] = useState<ScopeType>("national")
   110→  const [brandSlug, setBrandSlug] = useState(reports.brands[0]?.brandSlug ?? "")
   111→  const [gouvernorat, setGouvernorat] = useState(overview.govLosses[0]?.gouvernorat ?? "")
   112→  const [indicators, setIndicators] = useState<IndicatorKey[]>(DEFAULT_INDICATORS)
   113→
   114→  // Génération + modèles
   115→  const [generating, setGenerating] = useState(false)
   116→  const [generated, setGenerated] = useState<GeneratedReport | null>(null)
   117→  const [mapCache, setMapCache] = useState<ShadowMapResponse | null>(null)
   118→  const [models, setModels] = useState<ReportModel[]>([])
   119→  const [modelName, setModelName] = useState("")
   120→
   121→  // Modèles sauvegardés (lecture différée — setState asynchrone, hors règle).
   122→  useEffect(() => {
   123→    let alive = true
   124→    Promise.resolve()
   125→      .then(() => {
   126→        if (!alive) return
   127→        try {
   128→          const raw = localStorage.getItem(MODELS_KEY)
   129→          const parsed = raw ? (JSON.parse(raw) as ReportModel[]) : []
   130→          if (Array.isArray(parsed)) setModels(parsed.filter(m => m && m.id && m.from && m.to))
   131→        } catch {
   132→          setModels([])
   133→        }
   134→      })
   135→    return () => {
   136→      alive = false
   137→    }
   138→  }, [])
   139→
   140→  const persistModels = (next: ReportModel[]) => {
   141→    setModels(next)
   142→    try {
   143→      localStorage.setItem(MODELS_KEY, JSON.stringify(next))
   144→    } catch {
   145→      toast.error("Sauvegarde impossible", { description: "Le stockage local est indisponible (mode privé ?)." })
   146→    }
   147→  }
   148→
   149→  const periodValid = from <= to
   150→  const scopeValid =
   151→    scopeType !== "brand" || !!brandSlug
   152→  const gouvValid = scopeType !== "gouv" || !!gouvernorat
   153→  const indicatorsValid = indicators.length > 0
   154→  const canGenerate = periodValid && scopeValid && gouvValid && indicatorsValid && !generating
   155→
   156→  // -------------------------------------------------------------------
   157→  // Assemblage du rapport (client-side, depuis overview + map) — les
   158→  // paramètres sont explicites (chargement d'un modèle = génération directe).
   159→  // -------------------------------------------------------------------
   160→
   161→  const buildReport = async (params: {
   162→    from: string
   163→    to: string
   164→    scopeType: ScopeType
   165→    brandSlug: string
   166→    gouvernorat: string
   167→    indicators: IndicatorKey[]
   168→  }): Promise<void> => {
   169→    const { from: pFrom, to: pTo, scopeType: pScopeType, brandSlug: pBrandSlug, gouvernorat: pGouvernorat, indicators: pIndicators } = params
   170→    setGenerating(true)
   171→    try {
   172→      let map = mapCache
   173→      if (!map) {
   174→        const r = await fetch("/api/shadow/map")
   175→        if (!r.ok) throw new Error("map")
   176→        map = (await r.json()) as ShadowMapResponse
   177→        setMapCache(map)
   178→      }
   179→
   180→      const fromMonth = pFrom.slice(0, 7)
   181→      const toMonth = pTo.slice(0, 7)
   182→      const inRange = overview.monthly.filter(m => m.month >= fromMonth && m.month <= toMonth)
   183→      const rangeStats = inRange.length
   184→        ? inRange.reduce((acc, m) => ({ newPoints: acc.newPoints + m.newPoints, closed: acc.closed + m.closedPoints }), { newPoints: 0, closed: 0 })
   185→        : { newPoints: 0, closed: 0 }
   186→      const lastInRange = inRange[inRange.length - 1] ?? overview.monthly[overview.monthly.length - 1]
   187→
   188→      // ---- Périmètre ----
   189→      let pPoints = overview.kpis.activePoints
   190→      let pLoss = overview.kpis.estLossTnd
   191→      let pStations = overview.kpis.stationsThreatened
   192→      let scopeName = `National — ${scopeLabel}`
   193→      let scopedTypoSource = map.points
   194→      let scopedStations = map.stations.filter(s => s.lossTnd > 0)
   195→
   196→      if (pScopeType === "gouv" && pGouvernorat) {
   197→        const gov = overview.govLosses.find(g => g.gouvernorat === pGouvernorat)
   198→        pPoints = gov?.points ?? 0
   199→        pLoss = gov?.lossTnd ?? 0
   200→        pStations = gov?.stationsThreatened ?? 0
   201→        scopeName = `Gouvernorat — ${pGouvernorat}`
   202→        scopedTypoSource = map.points.filter(p => p.gouvernorat === pGouvernorat)
   203→        scopedStations = scopedStations.filter(s => s.gouvernorat === pGouvernorat)
   204→      } else if (pScopeType === "brand" && pBrandSlug) {
   205→        const brand = reports.brands.find(b => b.brandSlug === pBrandSlug)
   206→        pLoss = brand?.lossTnd ?? 0
   207→        pStations = brand?.stationsThreatened ?? 0
   208→        pPoints = brand?.points ?? 0
   209→        scopeName = `Enseigne — ${brand?.brandName ?? pBrandSlug}`
   210→        scopedTypoSource = map.points.filter(p => p.brandThreat === pBrandSlug)
   211→        scopedStations = scopedStations.filter(s => s.brandSlug === pBrandSlug)
   212→      }
   213→
   214→      const typoByCode = new Map(map.typologies.map(t => [t.code, t]))
   215→      const typoStats = new Map<string, { count: number; volumeLJ: number }>()
   216→      for (const p of scopedTypoSource) {
   217→        if (["REJETE", "DOUBLON"].includes(p.status)) continue
   218→        const cur = typoStats.get(p.typologyCode) ?? { count: 0, volumeLJ: 0 }
   219→        cur.count += 1
   220→        cur.volumeLJ += p.volumeLJ
   221→        typoStats.set(p.typologyCode, cur)
   222→      }
   223→      const typoTotal = [...typoStats.values()].reduce((s, t) => s + t.count, 0) || 1
   224→      const typologies: PaperTypo[] = [...typoStats.entries()]
   225→        .map(([code, t]) => ({
   226→          code,
   227→          label: typoByCode.get(code)?.label ?? code,
   228→          color: typoByCode.get(code)?.color ?? "#9E9E9E",
   229→          count: t.count,
   230→          pct: Math.round((t.count / typoTotal) * 1000) / 10,
   231→        }))
   232→        .sort((a, b) => b.count - a.count)
   233→
   234→      const threats: PaperThreat[] = [...scopedStations]
   235→        .sort((a, b) => b.lossTnd - a.lossTnd)
   236→        .slice(0, 10)
   237→        .map(s => ({
   238→          code: s.code,
   239→          name: s.name,
   240→          brandName: s.brandName,
   241→          gouvernorat: s.gouvernorat,
   242→          delegation: s.delegation,
   243→          points: s.points,
   244→          volumeStolenLJ: Math.round(s.lossTnd / (map.config.marginTndPerL * 30.4)),
   245→          lossTnd: Math.round(s.lossTnd),
   246→          level: s.level,
   247→          worstDistanceM: null,
   248→        }))
   249→
   250→      const gov: PaperGov[] =
   251→        pScopeType === "national"
   252→          ? topGov(overview.govLosses, 8).map(g => ({ gouvernorat: g.gouvernorat, points: g.points, lossTnd: g.lossTnd, level: g.level }))
   253→          : pScopeType === "gouv"
   254→            ? overview.govLosses
   255→                .filter(g => g.gouvernorat === pGouvernorat)
   256→                .map(g => ({ gouvernorat: g.gouvernorat, points: g.points, lossTnd: g.lossTnd, level: g.level }))
   257→            : govLossesForBrandStations(scopedStations)
   258→                .slice(0, 8)
   259→                .map(g => ({
   260→                  gouvernorat: g.gouvernorat,
   261→                  points: g.points,
   262→                  lossTnd: Math.round(g.lossTnd),
   263→                  level: levelFromLoss(g.lossTnd, map.config.lossRedTnd, map.config.lossOrangeTnd) as ShadowLevel,
   264→                }))
   265→
   266→      // ---- Rédaction ----
   267→      const periodLabel = `du ${dateFr(pFrom)} au ${dateFr(pTo)}`
   268→      const synthesis: string[] = []
   269→      synthesis.push(
   270→        `Rapport personnalisé du Shadow Market — période ${periodLabel}, périmètre « ${scopeName} ». ` +
   271→        `État au ${lastInRange?.label ?? reports.monthLabel} : ${pPoints.toLocaleString("fr-FR")} points informels actifs, ` +
   272→        `${formatTnd(pLoss)} de marge estimée perdue par mois, ${pStations.toLocaleString("fr-FR")} stations menacées.`,
   273→      )
   274→      if (inRange.length) {
   275→        synthesis.push(
   276→          `Sur la période (${inRange.length} mois couverts) : ${rangeStats.newPoints} apparitions de points et ` +
   277→          `${rangeStats.closed} neutralisations constatées. Les indicateurs « état courant » mesurent la situation ` +
   278→          `constatée à la date du rapport.`,
   279→        )
   280→      } else {
   281→        synthesis.push(
   282→          "Aucune statistique mensuelle archivée ne couvre cette période — les indicateurs portent l'état courant du recensement.",
   283→        )
   284→      }
   285→
   286→      let recommendations: string[] | null = null
   287→      if (pIndicators.includes("recommendations")) {
   288→        if (pScopeType === "brand" && pBrandSlug) {
   289→          const brand = reports.brands.find(b => b.brandSlug === pBrandSlug)
   290→          recommendations = brandRecommendations(brand?.brandName ?? pBrandSlug, pLoss, brand?.sharePct ?? 0, pStations, threats[0] ?? null)
   291→        } else if (pScopeType === "gouv" && pGouvernorat) {
   292→          const shareNat = overview.kpis.activePoints > 0 ? Math.round((pPoints / overview.kpis.activePoints) * 100) : 0
   293→          recommendations = [
   294→            `${pGouvernorat} concentre ${shareNat} % des points informels nationaux (${pPoints.toLocaleString("fr-FR")} points, ` +
   295→            `${formatTnd(pLoss)}/mois de perte) : intensifier les passages conjoints douanes / brigade sur les zones rouges.`,
   296→            threats[0]
   297→              ? `Priorité locale : ${threats[0].name} (${threats[0].gouvernorat}) — ${threats[0].points} points informels, ${formatTnd(threats[0].lossTnd)} de perte mensuelle.`
   298→              : "Aucune station fortement cannibalisée dans ce périmètre — maintenir la veille de recensement.",
   299→            "Étendre le recensement aux délégations limitrophes pour sécuriser la mesure de la recrudescence.",
   300→          ]
   301→        } else {
   302→          recommendations = reports.recommendations
   303→        }
   304→      }
   305→
   306→      const kpis: PaperKpi[] = []
   307→      if (pIndicators.includes("activePoints")) {
   308→        kpis.push({ label: "Points actifs", value: pPoints.toLocaleString("fr-FR"), sub: `+${rangeStats.newPoints} sur la période`, tone: "neutral" })
   309→      }
   310→      if (pIndicators.includes("estLoss")) {
   311→        kpis.push({ label: "Perte estimée", value: formatTnd(pLoss), sub: "marge / mois", tone: "RED" })
   312→      }
   313→      if (pIndicators.includes("stationsThreatened")) {
   314→        const reds = scopedStations.filter(s => s.level === "RED").length
   315→        kpis.push({ label: "Stations menacées", value: pStations.toLocaleString("fr-FR"), sub: `dont ${reds} critiques`, tone: "ORANGE" })
   316→      }
   317→      if (kpis.length === 0) kpis.push({ label: "Périmètre", value: scopeName.split("—")[1]?.trim() ?? scopeName, sub: periodLabel, tone: "neutral" })
   318→
   319→      const csvMeta: [string, string | number][] = [
   320→        ["Rapport personnalisé Shadow Market", scopeName],
   321→        ["Période", `du ${pFrom} au ${pTo}`],
   322→        ["Généré le", new Date().toLocaleString("fr-FR")],
   323→      ]
   324→
   325→      setGenerated({
   326→        docSubtitle: `${scopeName} · ${periodLabel}`,
   327→        generatedAt: new Date().toISOString(),
   328→        synthesis,
   329→        kpis,
   330→        threats: pIndicators.includes("topThreats") ? threats : null,
   331→        gov: pIndicators.includes("gouvernorats") ? gov : null,
   332→        trend: pIndicators.includes("trend12")
   333→          ? overview.monthly.map(m => ({ label: m.label, activePoints: m.activePoints, estLossTnd: m.estLossTnd }))
   334→          : null,
   335→        typologies: pIndicators.includes("typologies") ? typologies : null,
   336→        recommendations,
   337→        csvMeta,
   338→        csvThreats: threats.map(t => [t.code, t.name, t.brandName, t.gouvernorat, t.delegation, t.points, t.volumeStolenLJ, t.lossTnd, t.level]),
   339→        csvGov: gov.map(g => [g.gouvernorat, g.points, g.lossTnd, g.level]),
   340→      })
   341→      toast.success("Rapport personnalisé assemblé", {
   342→        description: `${scopeName} · ${periodLabel} — ${pIndicators.length} indicateurs inclus. Aperçu papier prêt à imprimer.`,
   343→      })
   344→    } catch {
   345→      toast.error("Assemblage impossible", { description: "Les données de carte n'ont pas pu être chargées. Réessayez." })
   346→    } finally {
   347→      setGenerating(false)
   348→    }
   349→  }
   350→
   351→  const toggleIndicator = (k: IndicatorKey) => {
   352→    setIndicators(prev => (prev.includes(k) ? prev.filter(x => x !== k) : [...prev, k]))
   353→  }
   354→
   355→  const saveModel = () => {
   356→    const name = modelName.trim() || `Modèle ${new Date().toLocaleDateString("fr-FR")}`
   357→    const model: ReportModel = {
   358→      id: `m-${Date.now()}`,
   359→      name,
   360→      from, to, scopeType, brandSlug, gouvernorat,
   361→      indicators: [...indicators],
   362→      savedAt: new Date().toISOString(),
   363→    }
   364→    persistModels([model, ...models].slice(0, 12))
   365→    setModelName("")
   366→    toast.success("Modèle enregistré", {
   367→      description: `« ${name} » sauvegardé localement — rechargeable en un clic depuis cette liste (12 modèles max).`,
   368→    })
   369→  }
   370→
   371→  const loadModel = (m: ReportModel) => {
   372→    // Application du modèle + génération directe avec ses paramètres explicites
   373→    // (aucun délai, aucun closure périmée — buildReport est paramétré).
   374→    setFrom(m.from)
   375→    setTo(m.to)
   376→    setPreset("custom")
   377→    setScopeType(m.scopeType)
   378→    setBrandSlug(m.brandSlug || reports.brands[0]?.brandSlug || "")
   379→    setGouvernorat(m.gouvernorat || overview.govLosses[0]?.gouvernorat || "")
   380→    setIndicators(m.indicators?.length ? m.indicators : DEFAULT_INDICATORS)
   381→    setStep(3)
   382→    toast.info("Modèle chargé", { description: `« ${m.name} » — génération lancée.` })
   383→    void buildReport({
   384→      from: m.from,
   385→      to: m.to,
   386→      scopeType: m.scopeType,
   387→      brandSlug: m.brandSlug || reports.brands[0]?.brandSlug || "",
   388→      gouvernorat: m.gouvernorat || overview.govLosses[0]?.gouvernorat || "",
   389→      indicators: m.indicators?.length ? m.indicators : DEFAULT_INDICATORS,
   390→    })
   391→  }
   392→
   393→  const monthsOptions = overview.monthly.map(m => m.month).reverse()
   394→
   395→  const stepCard = (n: number) => {
   396→    const s = STEPS[n - 1]
   397→    const Icon = s.icon
   398→    return (
   399→      <div className="flex items-center gap-2.5">
   400→        <span
   401→          className={cn(
   402→            "flex h-9 w-9 shrink-0 items-center justify-center rounded-full border-2 font-black transition-colors",
   403→            step === n
   404→              ? "border-[#E53935] bg-[#E53935] text-white"
   405→              : step > n
   406→                ? "border-[#4CAF50] bg-[#4CAF50]/10 text-[#4CAF50]"
   407→                : "border-stone-300 text-stone-400 dark:border-stone-700",
   408→          )}
   409→          aria-hidden
   410→        >
   411→          {step > n ? <Check className="h-4 w-4" /> : <Icon className="h-4 w-4" />}
   412→        </span>
   413→        <div className="min-w-0">
   414→          <p className={cn("text-[13px] font-bold leading-tight", step >= n ? "text-stone-900 dark:text-stone-50" : "text-stone-400")}>
   415→            Étape {n} · {s.label}
   416→          </p>
   417→          {step === n && <p className="text-[10.5px] text-muted-foreground">en cours</p>}
   418→        </div>
   419→      </div>
   420→    )
   421→  }
   422→
   423→  return (
   424→    <div className="space-y-4">
   425→      {/* Assistant 3 étapes */}
   426→      <Card className="border-stone-200 bg-white/95 dark:border-stone-800 dark:bg-stone-900/95">
   427→        <CardContent className="space-y-5 p-4 sm:p-6">
   428→          <div className="flex flex-wrap items-center justify-between gap-3">
   429→            <h3 className="flex items-center gap-2 text-sm font-semibold text-stone-900 dark:text-stone-50">
   430→              <Sparkles className="h-4 w-4 text-[#E53935]" aria-hidden />
   431→              Assistant de rapport personnalisé
   432→            </h3>
   433→            <Badge variant="outline" className="text-[10px]">Étape {step}/3</Badge>
   434→          </div>
   435→
   436→          {/* Stepper visuel */}
   437→          <div className="flex items-center gap-2 sm:gap-4">
   438→            {stepCard(1)}
   439→            <span className="h-0.5 flex-1 rounded-full bg-stone-200 dark:bg-stone-800" aria-hidden />
   440→            {stepCard(2)}
   441→            <span className="h-0.5 flex-1 rounded-full bg-stone-200 dark:bg-stone-800" aria-hidden />
   442→            {stepCard(3)}
   443→          </div>
   444→
   445→          {/* Étape 1 — Période */}
   446→          {step === 1 && (
   447→            <div className="space-y-4">
   448→              <div className="grid gap-3 sm:grid-cols-2">
   449→                <div className="space-y-1.5">
   450→                  <Label className="text-xs font-medium">Mois prédéfini</Label>
   451→                  <Select
   452→                    value={preset}
   453→                    onValueChange={v => {
   454→                      setPreset(v)
   455→                      if (v !== "custom") {
   456→                        const [f, t] = monthBounds(v)
   457→                        setFrom(f)
   458→                        setTo(t)
   459→                      }
   460→                    }}
   461→                  >
   462→                    <SelectTrigger aria-label="Mois prédéfini">
   463→                      <SelectValue />
   464→                    </SelectTrigger>
   465→                    <SelectContent className="max-h-64">
   466→                      <SelectItem value="custom">Période libre (dates ci-dessous)</SelectItem>
   467→                      {monthsOptions.map(m => (
   468→                        <SelectItem key={m} value={m}>{monthLabelFr(m)}</SelectItem>
   469→                      ))}
   470→                    </SelectContent>
   471→                  </Select>
   472→                </div>
   473→                <div />
   474→                <div className="space-y-1.5">
   475→                  <Label className="text-xs font-medium">Du</Label>
   476→                  <Input type="date" value={from} max={todayStr()} onChange={e => { setFrom(e.target.value); setPreset("custom") }} aria-label="Date de début" />
   477→                </div>
   478→                <div className="space-y-1.5">
   479→                  <Label className="text-xs font-medium">Au</Label>
   480→                  <Input type="date" value={to} max={todayStr()} onChange={e => { setTo(e.target.value); setPreset("custom") }} aria-label="Date de fin" />
   481→                </div>
   482→              </div>
   483→              {!periodValid && (
   484→                <p className="text-xs font-semibold text-[#E53935]">La date de début doit précéder la date de fin.</p>
   485→              )}
   486→            </div>
   487→          )}
   488→
   489→          {/* Étape 2 — Périmètre */}
   490→          {step === 2 && (
   491→            <div className="space-y-4">
   492→              <div role="radiogroup" aria-label="Type de périmètre" className="grid grid-cols-3 gap-2">
   493→                {([
   494→                  { v: "national", label: "National", icon: MapPin },
   495→                  { v: "brand", label: "Enseigne", icon: Building2 },
   496→                  { v: "gouv", label: "Gouvernorat", icon: CalendarRange },
   497→                ] as { v: ScopeType; label: string; icon: typeof MapPin }[]).map(o => {
   498→                  const Icon = o.icon
   499→                  const active = scopeType === o.v
   500→                  return (
   501→                    <button
   502→                      key={o.v}
   503→                      type="button"
   504→                      role="radio"
   505→                      aria-checked={active}
   506→                      onClick={() => setScopeType(o.v)}
   507→                      className={cn(
   508→                        "flex flex-col items-center gap-1.5 rounded-xl border p-3 transition-all",
   509→                        active
   510→                          ? "border-[#E53935] bg-[#E53935]/5 ring-1 ring-[#E53935]"
   511→                          : "border-stone-200 hover:border-stone-400 dark:border-stone-800",
   512→                      )}
   513→                    >
   514→                      <Icon className={cn("h-5 w-5", active ? "text-[#E53935]" : "text-stone-400")} aria-hidden />
   515→                      <span className={cn("text-xs font-bold", active ? "text-stone-900 dark:text-stone-50" : "text-muted-foreground")}>{o.label}</span>
   516→                    </button>
   517→                  )
   518→                })}
   519→              </div>
   520→              {scopeType === "brand" && (
   521→                <div className="space-y-1.5">
   522→                  <Label className="text-xs font-medium">Enseigne</Label>
   523→                  <Select value={brandSlug} onValueChange={setBrandSlug}>
   524→                    <SelectTrigger className="w-full sm:w-[260px]" aria-label="Enseigne du périmètre">
   525→                      <SelectValue />
   526→                    </SelectTrigger>
   527→                    <SelectContent>
   528→                      {reports.brands.map(b => (
   529→                        <SelectItem key={b.brandSlug} value={b.brandSlug}>
   530→                          {b.brandName} — {b.stationsThreatened} st. menacées
   531→                        </SelectItem>
   532→                      ))}
   533→                    </SelectContent>
   534→                  </Select>
   535→                </div>
   536→              )}
   537→              {scopeType === "gouv" && (
   538→                <div className="space-y-1.5">
   539→                  <Label className="text-xs font-medium">Gouvernorat</Label>
   540→                  <Select value={gouvernorat} onValueChange={setGouvernorat}>
   541→                    <SelectTrigger className="w-full sm:w-[260px]" aria-label="Gouvernorat du périmètre">
   542→                      <SelectValue />
   543→                    </SelectTrigger>
   544→                    <SelectContent className="max-h-64">
   545→                      {overview.govLosses.map(g => (
   546→                        <SelectItem key={g.gouvernorat} value={g.gouvernorat}>
   547→                          {g.gouvernorat} — {g.points} pts
   548→                        </SelectItem>
   549→                      ))}
   550→                    </SelectContent>
   551→                  </Select>
   552→                </div>
   553→              )}
   554→              {!scopeValid && <p className="text-xs font-semibold text-[#E53935]">Sélectionnez une enseigne.</p>}
   555→              {!gouvValid && <p className="text-xs font-semibold text-[#E53935]">Sélectionnez un gouvernorat.</p>}
   556→            </div>
   557→          )}
   558→
   559→          {/* Étape 3 — Indicateurs */}
   560→          {step === 3 && (
   561→            <div className="space-y-3">
   562→              <p className="text-[11px] text-muted-foreground">Cochez les blocs à inclure dans le rapport (au moins un).</p>
   563→              <div className="grid grid-cols-2 gap-2 lg:grid-cols-4">
   564→                {INDICATORS.map(ind => {
   565→                  const checked = indicators.includes(ind.key)
   566→                  return (
   567→                    <label
   568→                      key={ind.key}
   569→                      className={cn(
   570→                        "flex cursor-pointer items-start gap-2.5 rounded-xl border p-3 transition-all",
   571→                        checked ? "border-[#4CAF50] bg-[#4CAF50]/5" : "border-stone-200 hover:border-stone-400 dark:border-stone-800",
   572→                      )}
   573→                    >
   574→                      <Checkbox checked={checked} onCheckedChange={() => toggleIndicator(ind.key)} aria-label={ind.label} />
   575→                      <span className="min-w-0">
   576→                        <span className="block text-[12.5px] font-bold leading-tight text-stone-800 dark:text-stone-100">{ind.label}</span>
   577→                        <span className="block text-[10.5px] leading-snug text-muted-foreground">{ind.desc}</span>
   578→                      </span>
   579→                    </label>
   580→                  )
   581→                })}
   582→              </div>
   583→              {!indicatorsValid && <p className="text-xs font-semibold text-[#E53935]">Sélectionnez au moins un indicateur.</p>}
   584→            </div>
   585→          )}
   586→
   587→          {/* Navigation de l'assistant */}
   588→          <div className="flex flex-wrap items-center justify-between gap-2 border-t border-stone-200 pt-4 dark:border-stone-800">
   589→            <Button variant="outline" className="gap-1.5" disabled={step === 1} onClick={() => setStep(s => Math.max(1, s - 1))}>
   590→              <ChevronLeft className="h-4 w-4" aria-hidden /> Précédent
   591→            </Button>
   592→            {step < 3 ? (
   593→              <Button
   594→                className="gap-1.5"
   595→                disabled={(step === 1 && !periodValid) || (step === 2 && (!scopeValid || !gouvValid))}
   596→                onClick={() => setStep(s => Math.min(3, s + 1))}
   597→              >
   598→                Suivant <ChevronRight className="h-4 w-4" aria-hidden />
   599→              </Button>
   600→            ) : (
   601→              <Button className="gap-1.5" disabled={!canGenerate} onClick={() => void buildReport({ from, to, scopeType, brandSlug, gouvernorat, indicators })}>
   602→                {generating ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <Wand2 className="h-4 w-4" aria-hidden />}
   603→                {generating ? "Assemblage…" : "Générer le rapport"}
   604→              </Button>
   605→            )}
   606→          </div>
   607→        </CardContent>
   608→      </Card>
   609→
   610→      {/* Modèles sauvegardés */}
   611→      <Card className="border-stone-200 bg-white/95 dark:border-stone-800 dark:bg-stone-900/95">
   612→        <CardContent className="space-y-3 p-4">
   613→          <h3 className="flex items-center gap-2 text-sm font-semibold text-stone-900 dark:text-stone-50">
   614→            <BookMarked className="h-4 w-4 text-[#FF9800]" aria-hidden />
   615→            Modèles enregistrés
   616→            <Badge variant="outline" className="text-[10px]">{models.length}</Badge>
   617→          </h3>
   618→          {models.length === 0 ? (
   619→            <p className="rounded-lg border border-dashed border-stone-300 p-3 text-xs text-muted-foreground dark:border-stone-700">
   620→              Aucun modèle sauvegardé — générez un rapport puis « Enregistrer le modèle » (stockage local de ce navigateur).
   621→            </p>
   622→          ) : (
   623→            <div className="max-h-48 space-y-1.5 overflow-y-auto pr-1 [scrollbar-width:thin]">
   624→              {models.map(m => (
   625→                <div key={m.id} className="flex flex-wrap items-center gap-2 rounded-lg border border-stone-200 bg-white p-2.5 dark:border-stone-800 dark:bg-stone-900">
   626→                  <div className="min-w-0 flex-1">
   627→                    <p className="truncate text-[13px] font-semibold text-stone-800 dark:text-stone-100">{m.name}</p>
   628→                    <p className="text-[11px] text-muted-foreground">
   629→                      {m.scopeType === "national" ? "National" : m.scopeType === "brand" ? reports.brands.find(b => b.brandSlug === m.brandSlug)?.brandName ?? "Enseigne" : m.gouvernorat}
   630→                      {" · "}{dateFr(m.from)} → {dateFr(m.to)} · {m.indicators.length} indicateurs
   631→                    </p>
   632→                  </div>
   633→                  <Button size="sm" variant="outline" className="h-8 gap-1.5" onClick={() => loadModel(m)}>
   634→                    <Wand2 className="h-3.5 w-3.5" aria-hidden /> Charger &amp; générer
   635→                  </Button>
   636→                  <Button
   637→                    size="icon"
   638→                    variant="ghost"
   639→                    className="h-8 w-8 text-muted-foreground hover:text-red-600"
   640→                    aria-label={`Supprimer ${m.name}`}
   641→                    onClick={() => {
   642→                      persistModels(models.filter(x => x.id !== m.id))
   643→                      toast.info("Modèle supprimé", { description: m.name })
   644→                    }}
   645→                  >
   646→                    <Trash2 className="h-3.5 w-3.5" aria-hidden />
   647→                  </Button>
   648→                </div>
   649→              ))}
   650→            </div>
   651→          )}
   652→        </CardContent>
   653→      </Card>
   654→
   655→      {/* Rapport assemblé */}
   656→      {generated && (
   657→        <div className="space-y-2">
   658→          <div className="flex flex-wrap items-center gap-2">
   659→            <h3 className="flex items-center gap-2 text-sm font-semibold text-stone-900 dark:text-stone-50">
   660→              <span aria-hidden>🧾</span> Rapport assemblé
   661→            </h3>
   662→            <Button
   663→              variant="outline"
   664→              size="sm"
   665→              className="gap-1.5"
   666→              onClick={() => {
   667→                downloadTableCsv(`shadow-perso-${from}_${to}.csv`, generated.csvMeta, [
   668→                  {
   669→                    title: "Top menaces",
   670→                    header: ["Code", "Station", "Enseigne", "Gouvernorat", "Délégation", "Points", "L/j volés", "Perte DT/mois", "Niveau"],
   671→                    rows: generated.csvThreats,
   672→                  },
   673→                  { title: "Pertes par gouvernorat", header: ["Gouvernorat", "Points", "Perte DT/mois", "Niveau"], rows: generated.csvGov },
   674→                ])
   675→                toast.success("Export CSV téléchargé", { description: `shadow-perso-${from}_${to}.csv` })
   676→              }}
   677→            >
   678→              <FileDown className="h-3.5 w-3.5" aria-hidden /> Télécharger CSV
   679→            </Button>
   680→            <div className="flex items-center gap-1.5">
   681→              <Input
   682→                value={modelName}
   683→                onChange={e => setModelName(e.target.value)}
   684→                placeholder="Nom du modèle (ex. Sud-Est trimestriel)"
   685→                className="h-9 w-56 text-xs"
   686→                aria-label="Nom du modèle"
   687→              />
   688→              <Button variant="outline" size="sm" className="gap-1.5" onClick={saveModel}>
   689→                <Save className="h-3.5 w-3.5" aria-hidden /> Enregistrer le modèle
   690→              </Button>
   691→            </div>
   692→          </div>
   693→          <ReportPaper
   694→            docTitle="Rapport Personnalisé"
   695→            docSubtitle={generated.docSubtitle}
   696→            generatedAt={generated.generatedAt}
   697→            synthesis={generated.synthesis}
   698→            kpis={generated.kpis}
   699→            threats={generated.threats}
   700→            threatsTitle="Top 10 menaces du périmètre"
   701→            gov={generated.gov}
   702→            govTitle="Perte estimée par gouvernorat"
   703→            trend={generated.trend}
   704→            typologies={generated.typologies}
   705→            recommendations={generated.recommendations}
   706→            printHint="Rapport assemblé à la demande — seuls les indicateurs cochés apparaissent."
   707→          />
   708→        </div>
