    58→    58→function csvEscape(v: string | number): string {
    59→    59→  const s = String(v ?? "")
    60→    60→  return /[",;\n\r]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s
    61→    61→}
    62→    62→
    63→    63→function csvRow(cells: (string | number)[]): string {
    64→    64→  return cells.map(csvEscape).join(";")
    65→    65→}

    67→    67→function downloadCsv(filename: string, content: string): void {
    68→    68→  const blob = new Blob([`\ufeff${content}`], { type: "text/csv;charset=utf-8;" })
    69→    69→  const url = URL.createObjectURL(blob)
    70→    70→  const a = document.createElement("a")
    71→    71→  a.href = url
    72→    72→  a.download = filename
    73→    73→  document.body.appendChild(a)
    74→    74→  a.click()
    75→    75→  document.body.removeChild(a)
    76→    76→  URL.revokeObjectURL(url)
    77→    77→}

    86→    86→function exportContractsCsv(rows: Row[], statusFilter: string, planFilter: string): string {
    87→    87→  const lines: string[] = [

    89→    89→      "Enseigne", "Plan", "Statut", "Tarif mensuel (DT)", "Sièges actifs",

    91→    91→    ]),

    97→    97→      CONTRACT_STATUS_META[c.status]?.label ?? c.status,
    98→    98→      fmtCsvAmount(c.monthlyFee),
    99→    99→      c.activeUsers,
   100→   100→      c.seats,
   101→   101→      `${c.stationCount} / ${c.stationQuota}`,
   102→   102→      `${c.contactName} (${c.contactEmail})`,
   103→   103→      new Date(c.renewsAt).toLocaleDateString("fr-FR"),
   104→   104→      fmtCsvAmount(c.unpaidTotal),
   105→   105→    ]))
   106→   106→  }

   124→   124→function isoToLocalDateInput(iso: string): string {
   125→   125→  const d = new Date(iso)
   126→   126→  const p = (n: number) => String(n).padStart(2, "0")

   130→   130→/** Décimal français accepté : « 890 » ou « 890,5 ». null si vide/invalide. */
   131→   131→function parseDecimal(raw: string): number | null {
   132→   132→  const t = raw.trim().replace(",", ".")
   133→   133→  if (t === "") return null
   134→   134→  const v = Number(t)
   135→   135→  return Number.isFinite(v) ? v : null
   136→   136→}
